    13→let _zaiInstance: any = null;
    14→async function createZAI() {
    15→  if (_zaiInstance) return _zaiInstance;
    16→  
    17→  // Check if env vars are available (Vercel deployment)
    18→  const baseUrl = process.env.ZAI_BASE_URL;
    19→  const apiKey = process.env.ZAI_API_KEY;
    20→  const chatId = process.env.ZAI_CHAT_ID;
    21→  const token = process.env.ZAI_TOKEN;
    22→  const userId = process.env.ZAI_USER_ID;
    23→
    24→  if (baseUrl && apiKey && chatId && token && userId) {
    25→    console.log("[ZAI] Creating with environment config (new ZAI(config))");
    26→    _zaiInstance = new ZAI({ baseUrl, apiKey, chatId, token, userId });
    27→  } else {
    28→    // Fallback to .z-ai-config file (local development)
    29→    console.log("[ZAI] Creating with default config (file-based)");
    30→    _zaiInstance = await ZAI.create();
    31→  }
    32→  return _zaiInstance;

    52→ * search because it reads the actual page content.

    55→ * follows the redirect and renders the page. The product data (including

   134→                const originalPrice = priceFields.marketPrice || priceFields.origPrice || null;

   337→/* ───────────────────────────────────────────────────────────────────
   338→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   339→ *
   340→ * Temu share URLs include everything we need:
   341→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   342→ *   - top_gallery_url    = URL-encoded product image URL
   343→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   344→ *
   345→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   346→ * ─────────────────────────────────────────────────────────────────── */
   347→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   348→  let parsed: URL;
   349→  try {
   350→    parsed = new URL(originalUrl);
   351→  } catch {
   352→    return null;
   353→  }
   354→
   355→  let price: number | null = null;
   356→  let currency = "USD";
   357→  let priceSource = "";
   358→
   359→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   360→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   361→  if (hint) {
   362→    try {
   363→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   364→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   365→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   366→      if (cents > 0 && cents < 10000000) {
   367→        const usd = cents / 100;
   368→        if (usd >= 0.01 && usd < 100000) {
   369→          price = usd;
   370→          currency = "USD";
   371→          priceSource = "url-hint";
   372→        }
   373→      }
   374→    } catch { /* not valid base64 */ }
   375→  }
   376→
   377→  // Product image from top_gallery_url
   378→  let image: string | null = null;
   379→  const topGallery = parsed.searchParams.get("top_gallery_url");
   380→  if (topGallery) {
   381→    try {
   382→      // Validate it's a real URL
   383→      const imgUrl = new URL(topGallery);
   384→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   385→        image = topGallery;
   386→      }
   387→    } catch { /* not a valid URL */ }
   388→  }
   389→
   390→  // Product name from URL slug
   391→  let productName: string | null = null;
   392→  const segments = parsed.pathname.split("/").filter(Boolean);
   393→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   394→  const nameFromSlug = slug
   395→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   396→    .replace(/\.html?$/i, "")
   397→    .replace(/-/g, " ")
   398→    .replace(/\s+/g, " ")
   399→    .trim();
   400→  if (nameFromSlug && nameFromSlug.length > 3) {
   401→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   402→  }
   403→
   404→  // If we got at least the price (the most important piece), return success
   405→  if (price) {
   406→    return {
   407→      price,
   408→      currency,
   409→      productName,
   410→      productDescription: null,
   411→      canonicalUrl: null,
   412→      originalPrice: null,
   413→      image,
   414→      source: priceSource,
   415→      antiBotDetected: false,
   416→    };
   417→  }
   418→
   419→  // No price from URL — return null so caller tries HTML-based strategies
   420→  return null;
   421→}

   479→async function fetchPriceWithWebSearch(
   480→  goodsId: string | null,
   481→  itemId: string | null,
   482→  originalUrl: string,
   483→): Promise<TemuProductData | null> {
   484→  try {
   485→    const zai = await createZAI();
   486→
   487→    // Build search query — try different strategies
   488→    let searchQuery = "";
   489→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   490→      // Item ID like TV10922608 — search broadly on Temu
   491→      searchQuery = `temu ${itemId}`;
   492→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   493→      // Numeric goods_id — search with the -g- pattern
   494→      searchQuery = `site:temu.com "g-${goodsId}"`;
   495→    } else {
   496→      // Fallback: search with the URL
   497→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   498→      searchQuery = `site:temu.com ${cleanUrl}`;
   499→    }
   500→
   501→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   502→
   503→    // Step 1: Web search to find the product
   504→    const searchResults = await (zai as any).invokeFunction("web_search", {
   505→      query: searchQuery,
   506→      num: 5,
   507→    });
   508→
   509→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   510→      console.log("[WebSearch] No results found, trying broader search...");
   511→
   512→      // Broader search without site: restriction
   513→      const broadQuery = itemId
   514→        ? `temu ${itemId} price`
   515→        : goodsId
   516→          ? `temu ${goodsId} price`
   517→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   518→
   519→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   520→      const broadResults = await (zai as any).invokeFunction("web_search", {
   521→        query: broadQuery,
   522→        num: 5,
   523→      });
   524→
   525→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   526→        console.log("[WebSearch] No results from broader search either");
   527→        return null;
   528→      }
   529→
   530→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   531→    }
   532→
   533→    // Step 2: Try to parse price from initial search results
   534→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   535→    if (initialPrice?.price && initialPrice.price > 0) {
   536→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   537→      return initialPrice;
   538→    }
   539→
   540→    // Step 3: If initial search found product names but no prices,
   541→    // try a more targeted search with the product name to find prices
   542→    const productNameFromResults = searchResults
   543→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   544→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   545→      .find((n: string) => n && n.length > 10);
   546→
   547→    if (productNameFromResults) {
   548→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   549→      try {
   550→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   551→        const nameResults = await (zai as any).invokeFunction("web_search", {
   552→          query: nameQuery,
   553→          num: 5,
   554→        });
   555→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   556→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   557→          if (namePrice?.price && namePrice.price > 0) {
   558→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   559→            return namePrice;
   560→          }
   561→          // Merge results and try LLM on combined results
   562→          const combinedResults = [...searchResults, ...nameResults];
   563→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   564→        }
   565→      } catch (err) {
   566→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   567→      }
   568→    }
   569→
   570→    // Step 4: Use LLM on whatever results we have
   571→    const llmResult = await extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   572→    if (llmResult?.price && llmResult.price > 0) {
   573→      return llmResult;
   574→    }
   575→
   576→    // Step 5: If LLM didn't find price, try reading Temu search result pages
   577→    // via AllOrigins proxy — sometimes the proxy can access the product page
   578→    // when direct access is blocked
   579→    const temuResults = searchResults
   580→      .filter((r: any) => r.url?.includes("temu.com") && r.url?.includes("-g-"))
   581→      .slice(0, 2);
   582→
   583→    for (const result of temuResults) {
   584→      try {
   585→        console.log(`[WebSearch] Trying AllOrigins for search result: ${result.url.slice(0, 80)}...`);
   586→        // Try both /raw and /get endpoints
   587→        const proxyUrls = [
   588→          `https://api.allorigins.win/raw?url=${encodeURIComponent(result.url)}`,
   589→          `https://api.allorigins.win/get?url=${encodeURIComponent(result.url)}`,
   590→        ];
   591→
   592→        let proxyHtml: string | null = null;
   593→
   594→        for (const pUrl of proxyUrls) {
   595→          try {
   596→            const controller = new AbortController();
   597→            const timeout = setTimeout(() => controller.abort(), 15000);
   598→            const proxyRes = await fetch(pUrl, { signal: controller.signal });
   599→            clearTimeout(timeout);
   600→            if (!proxyRes.ok) continue;
   601→
   602→            const isRaw = pUrl.includes("/raw?");
   603→            if (isRaw) {
   604→              const text = await proxyRes.text();
   605→              if (text && text.length > 5000 && text.includes("<")) {
   606→                proxyHtml = text;
   607→                break;
   608→              }
   609→            } else {
   610→              const data = await proxyRes.json();
   611→              const contents = typeof data === "string" ? data : data?.contents;
   612→              if (contents && typeof contents === "string" && contents.length > 5000) {
   613→                proxyHtml = contents;
   614→                break;
   615→              }
   616→            }
   617→          } catch { /* try next */ }
   618→        }
   619→
   620→        if (!proxyHtml || proxyHtml.length < 5000) continue;
   621→
   622→        // Check if we got the real product page (has OG title)
   623→        const hasOGTitle = /<meta[^>]*property=["']og:title["']/i.test(proxyHtml);
   624→        if (!hasOGTitle) continue;
   625→
   626→        // Extract price using extractProductInfo
   627→        const proxyResult = extractProductInfo(proxyHtml, result.url);
   628→        if (proxyResult.price && proxyResult.price > 0) {
   629→          console.log(`[WebSearch] ✓ Got price $${proxyResult.price} from AllOrigins search result`);
   630→          return proxyResult;
   631→        }
   632→
   633→        // Also try direct OG price extraction
   634→        const ogPriceMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   635→        const ogCurrencyMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   636→        if (ogPriceMatch) {
   637→          const priceVal = parseFloat(ogPriceMatch[1]);
   638→          const cur = ogCurrencyMatch?.[1] || "USD";
   639→          if (priceVal > 0 && priceVal < 100000) {
   640→            let priceUSD = priceVal;
   641→            if (cur !== "USD" && CURRENCY_TO_USD[cur]) {
   642→              priceUSD = Math.round(priceVal * CURRENCY_TO_USD[cur] * 100) / 100;
   643→            }
   644→            const ogTitle = proxyHtml.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1];
   645→            const ogImage = proxyHtml.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1];
   646→            const foundGoodsId = result.url.match(/-g-(\d{10,})/)?.[1] || goodsId;
   647→            console.log(`[WebSearch] ✓ Got OG price ${priceVal} ${cur} = $${priceUSD} from AllOrigins search result`);
   648→            return {
   649→              price: priceUSD,
   650→              currency: "USD",
   651→              productName: ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null,
   652→              productDescription: null,
   653→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   654→              originalPrice: null,
   655→              image: ogImage || null,
   656→              source: `websearch-allorigins-og(${cur})`,
   657→              antiBotDetected: false,
   658→            };
   659→          }
   660→        }
   661→      } catch (err) {
   662→        console.log(`[WebSearch] AllOrigins for search result failed:`, String(err).slice(0, 80));
   663→      }
   664→    }
   665→
   666→    return llmResult; // Return whatever the LLM found (may be null)
   667→  } catch (err) {
   668→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   669→    return null;
   670→  }
   671→}

   673→async function extractPriceFromSearchResults(
   674→  zai: any,
   675→  searchResults: any[],
   676→  goodsId: string | null,
   677→  itemId: string | null,
   678→): Promise<TemuProductData | null> {
   679→  // Step 2: Try to parse prices directly from search snippets first (fast, no LLM cost)
   680→  const snippetPrice = parsePriceFromSnippets(searchResults, goodsId);
   681→  if (snippetPrice?.price && snippetPrice.price > 0) {
   682→    console.log(`[WebSearch] ✓ Found price from snippet: ${snippetPrice.price} USD (source: ${snippetPrice.source})`);
   683→    return snippetPrice;
   684→  }
   685→
   686→  // Step 3: If snippets don't have a clear price, use LLM to extract
   687→  console.log("[WebSearch] Snippets don't have clear price, using LLM...");
   688→
   689→  const searchContext = searchResults
   690→    .slice(0, 5)
   691→    .map(
   692→      (r: any, i: number) =>
   693→        `${i + 1}. ${r.name || "No title"}\n   URL: ${r.url}\n   Snippet: ${r.snippet || "No snippet"}`,
   694→    )
   695→    .join("\n\n");
   696→
   697→  try {
   698→    const completion = await zai.createChatCompletion({
   699→      messages: [
   700→        {
   701→          role: "system",
   702→          content:
   703→            'You are a price extraction assistant for Temu products. Extract the product price from the search results. ' +
   704→            'Return ONLY a JSON object with: {"price_usd": <number_in_USD>, "name": "<product_name>", "currency": "<original_currency>", "original_price": <number_before_conversion>, "confidence": <high|medium|low>}. ' +
   705→            'If the price is in a non-USD currency, CONVERT it to USD using approximate exchange rates. ' +
   706→            'Common conversions: MUR→USD ÷47, OMR→USD ×2.60, BHD→USD ×2.65, PKR→USD ÷278, EUR→USD ×1.08, GBP→USD ×1.27, SAR→USD ×0.27, AED→USD ×0.27. ' +
   707→            'If you cannot find a clear price, return {"price_usd": null, "name": "<best_guess_name>", "confidence": "low"}. ' +
   708→            'ALWAYS return valid JSON. The price_usd must be a number, not a string. ' +
   709→            'IMPORTANT: Do NOT estimate or guess the price. If the search results do not contain a clear price for this specific product, return null. ' +
   710→            'Do NOT confuse "delivery guarantee" prices, "credit for delay" amounts, or "30% OFF" discount percentages with the product price. ' +
   711→            'Do NOT use prices from unrelated products. Only extract the price that clearly corresponds to the product with the given goods_id or Item ID.',
   712→        },
   713→        {
   714→          role: "user",
   715→          content:
   716→            `Product goods_id: ${goodsId || "unknown"}\n` +
   717→            `Item ID: ${itemId || "unknown"}\n\n` +
   718→            `Search Results:\n${searchContext}\n\n` +
   719→            `Extract the product price in USD from these results. Return JSON only.`,
   720→        },
   721→      ],
   722→    });
   723→
   724→    const aiResponse = completion.choices?.[0]?.message?.content || "";
   725→    console.log("[WebSearch] LLM response:", aiResponse.slice(0, 300));
   726→
   727→    // Parse AI response — extract JSON
   728→    const jsonMatch = aiResponse.match(/\{[\s\S]*?\}/);
   729→    if (!jsonMatch) {
   730→      console.log("[WebSearch] No JSON in LLM response");
   731→      return null;
   732→    }
   733→
   734→    const parsed = JSON.parse(jsonMatch[0]);
   735→    const priceUSD = typeof parsed.price_usd === "number"
   736→      ? parsed.price_usd
   737→      : parseFloat(String(parsed.price_usd));
   738→
   739→    if (priceUSD && priceUSD > 0 && priceUSD < 100000) {
   740→      console.log(`[WebSearch] ✓ LLM found price: $${priceUSD} (confidence: ${parsed.confidence})`);
   741→      return {
   742→        price: priceUSD,
   743→        currency: "USD",
   744→        productName: parsed.name || null,
   745→        productDescription: null,
   746→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   747→        originalPrice: null,
   748→        image: null, // Image will come from share URL params
   749→        source: `web-search-llm(${parsed.confidence})`,
   750→        antiBotDetected: false,
   751→      };
   752→    }
   753→
   754→    // LLM couldn't find price either — return what we can
   755→    if (parsed.name) {
   756→      return {
   757→        price: null,
   758→        currency: "USD",
   759→        productName: parsed.name,
   760→        productDescription: null,
   761→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   762→        originalPrice: null,
   763→        image: null,
   764→        source: "web-search-llm(no-price)",
   765→        antiBotDetected: false,
   766→      };
   767→    }
   768→
   769→    return null;
   770→  } catch (err) {
   771→    console.error("[WebSearch] LLM error:", String(err).slice(0, 150));
   772→    return null;
   773→  }

   776→/**
   777→ * Try to parse prices directly from search result snippets.
   778→ * This avoids the cost of an LLM call and is often sufficient
   779→ * because Google includes price info in Temu search snippets.
   780→ */
   781→function parsePriceFromSnippets(
   782→  results: any[],
   783→  goodsId: string | null,
   784→): TemuProductData | null {
   785→  // Look for Temu results that have price info in snippets
   786→  for (const result of results) {
   787→    if (!result.url?.includes("temu.com")) continue;
   788→    if (!result.snippet) continue;
   789→
   790→    const snippet = result.snippet;
   791→
   792→    // Extract the goods_id from the URL if we don't have it
   793→    let foundGoodsId = goodsId;
   794→    if (!foundGoodsId) {
   795→      const gMatch = result.url.match(/-g-(\d{10,})/);
   796→      if (gMatch) foundGoodsId = gMatch[1];
   797→    }
   798→
   799→    // Extract product name from search result
   800→    const productName = result.name
   801→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   802→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   803→      .trim() || null;
   804→
   805→    // Determine the local currency from the Temu URL locale prefix
   806→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   807→    let localCurrency = "USD";
   808→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   809→    if (localeMatch) {
   810→      const locale = localeMatch[1].toLowerCase();
   811→      const localeToCurrency: Record<string, string> = {
   812→        mu: "MUR",       // Mauritius
   813→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   814→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   815→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   816→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   817→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   818→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   819→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   820→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   821→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   822→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   823→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   824→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   825→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   826→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   827→        np: "NPR", "np-en": "NPR", // Nepal
   828→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   829→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   830→        ph: "PHP", "ph-en": "PHP", // Philippines
   831→        br: "BRL", "br-pt": "BRL", // Brazil
   832→        mx: "MXN", "mx-es": "MXN", // Mexico
   833→      };
   834→      const found = localeToCurrency[locale];
   835→      if (found) localCurrency = found;
   836→    }
   837→
   838→    // Try to find a price in the snippet
   839→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   840→    // Common patterns:
   841→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   842→    //   "$6.99"                  → USD
   843→    //   "OMR 1.200"             → Omani Rial
   844→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   845→    //   "€5.99"                  → EUR
   846→    //   "£4.99"                  → GBP
   847→
   848→    // Determine if this locale uses European comma format (comma = decimal separator)
   849→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   850→    const europeanCommaLocales = new Set([
   851→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",
   852→      "ma-fr", "tn-fr", // French-speaking locales
   853→    ]);
   854→    const usesEuropeanComma = europeanCommaLocales.has(localCurrency === "USD" && localeMatch ? localeMatch[1].toLowerCase() : "") ||
   855→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   856→
   857→    // Helper: parse price string respecting locale format
   858→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   859→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   860→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   861→      if (isEuropean) {
   862→        // European: remove dots (thousands), replace comma with dot (decimal)
   863→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   864→      } else {
   865→        // English: remove commas (thousands)
   866→        return parseFloat(priceStr.replace(/,/g, ""));
   867→      }
   868→    };
   869→
   870→    // First, try explicit currency patterns (most reliable)
   871→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   872→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   873→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   874→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   875→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   876→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   877→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   878→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   879→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   880→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   881→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   882→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   883→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   884→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   885→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   886→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   887→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   888→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   889→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   890→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   891→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   892→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   893→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   894→    ];
   895→
   896→    for (const { pattern, currency } of explicitPatterns) {
   897→      const match = snippet.match(pattern);
   898→      if (match) {
   899→        // For USD prices on European locale pages (like Ecuador), use comma-as-decimal parsing
   900→        const isEuroPrice = currency === "USD" && usesEuropeanComma;
   901→        const localPrice = parsePriceString(match[1], isEuroPrice);
   902→        if (localPrice > 0 && localPrice < 10000000) {
   903→          const rate = CURRENCY_TO_USD[currency];
   904→          if (rate) {
   905→            const usdPrice = Math.round(localPrice * rate * 100) / 100;

  1534→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1535→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1536→  // 1. OG meta tags (work even on anti-bot page)
  1537→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1538→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1539→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1540→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1541→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1542→
  1543→  // Decode HTML entities in title
  1544→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1545→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1546→
  1547→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1548→  let price: number | null = null;
  1549→  let currency = "USD";
  1550→  let originalPrice: number | null = null;
  1551→  let priceSource = "";
  1552→
  1553→  // ─────────────────────────────────────────────────────────────
  1554→  // 2-preferred. URL-embedded price hint.
  1555→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1556→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1557→  // → 1.23 USD. This is the most reliable signal because it comes
  1558→  // straight from the share URL the user copied.
  1559→  // ─────────────────────────────────────────────────────────────
  1560→  try {
  1561→    const parsedUrl = new URL(originalUrl);
  1562→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1563→    if (hint) {
  1564→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1565→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1566→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1567→      // Strip non-digits
  1568→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1569→      if (cents > 0 && cents < 10000000) {
  1570→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1571→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1572→        const usd = cents / 100;
  1573→        if (usd >= 0.01 && usd < 100000) {
  1574→          price = usd;
  1575→          currency = "USD";
  1576→          priceSource = "url-hint";
  1577→        }
  1578→      }
  1579→    }
  1580→  } catch { /* not a parseable URL */ }
  1581→
  1582→  // 2a. JSON-LD structured data
  1583→  if (!price) {
  1584→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1585→    for (const match of jsonLdMatches) {
  1586→      try {
  1587→        const data = JSON.parse(match[1]);
  1588→        const schemas = Array.isArray(data) ? data : [data];
  1589→        for (const schema of schemas) {
  1590→          if (schema["@type"] === "Product" && schema.offers) {
  1591→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1592→            for (const offer of offers) {
  1593→              if (offer.price !== undefined) {
  1594→                price = parseFloat(offer.price);
  1595→                currency = offer.priceCurrency || "USD";
  1596→                priceSource = "json-ld";
  1597→                break;
  1598→              }
  1599→            }
  1600→          }
  1601→        }
  1602→      } catch { /* skip */ }
  1603→      if (price) break;
  1604→    }
  1605→  }
  1606→
  1607→  // 2b. OG product:price:amount meta
  1608→  if (!price) {
  1609→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1610→    if (ogPrice) {
  1611→      const p = parseFloat(ogPrice[1]);
  1612→      if (p > 0) {
  1613→        price = p;
  1614→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1615→        currency = ogCur?.[1] || "USD";
  1616→        priceSource = "og-meta";
  1617→      }
  1618→    }
  1619→  }
  1620→
  1621→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1622→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1623→  //     one per "similar product", one per cross-sell, etc.). The previous
  1624→  //     implementation picked the FIRST match, which often returned a
  1625→  //     cross-sell product's price (e.g. $30) instead of the actual
  1626→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1627→  //     pick the LOWEST plausible one, since the actual product the user
  1628→  //     is viewing is almost always the cheapest among the rendered
  1629→  //     priceInfo blocks.
  1630→  if (!price) {
  1631→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1632→    const candidates: Candidate[] = [];
  1633→
  1634→    // priceInfo with explicit priceStr (most reliable)
  1635→    const priceInfosWithStr = [...html.matchAll(
  1636→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1637→    )];
  1638→    for (const pi of priceInfosWithStr) {
  1639→      const cents = parseInt(pi[1]);
  1640→      const cur = pi[2];
  1641→      const priceStr = pi[3];
  1642→      // Skip i18n placeholder strings
  1643→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1644→      if (cents <= 0 || cents >= 100000000) continue;
  1645→      const usd = cents / 100;
  1646→      if (usd < 0.01 || usd > 100000) continue;
  1647→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1648→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1649→      candidates.push({ usd, currency: cur, marketUsd });
  1650→    }
  1651→
  1652→    // priceInfo without priceStr (broader fallback)
  1653→    if (candidates.length === 0) {
  1654→      const priceInfosAll = [...html.matchAll(
  1655→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1656→      )];
  1657→      for (const pi of priceInfosAll) {
  1658→        const cents = parseInt(pi[1]);
  1659→        const cur = pi[2];
  1660→        if (cents <= 50 || cents >= 100000000) continue;
  1661→        const usd = cents / 100;
  1662→        if (usd < 0.50 || usd > 100000) continue;
  1663→        candidates.push({ usd, currency: cur });
  1664→      }
  1665→    }
  1666→
  1667→    if (candidates.length > 0) {
  1668→      // Pick the LOWEST price — the user is always viewing the cheapest
  1669→      // variant of the actual product, not the cross-sells.
  1670→      candidates.sort((a, b) => a.usd - b.usd);
  1671→      const best = candidates[0];
  1672→      price = best.usd;
  1673→      currency = best.currency;
  1674→      if (best.marketUsd && best.marketUsd > best.usd) {
  1675→        originalPrice = best.marketUsd;
  1676→      }
  1677→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1678→    }
  1679→  }
  1680→
  1681→  // 2d. Embedded JSON price fields (broad regex sweep)
  1682→  if (!price) {
  1683→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1684→    const found: { value: number; field: string }[] = [];
  1685→    for (const f of fields) {
  1686→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1687→      for (const m of html.matchAll(re)) {
  1688→        const v = parseFloat(m[1]);
  1689→        if (v > 0 && v < 100000) {
  1690→          found.push({ value: v, field: f });
  1691→        }
  1692→      }
  1693→    }
  1694→    if (found.length > 0) {
  1695→      // Sort and pick lowest (likely sale price)
  1696→      found.sort((a, b) => a.value - b.value);
  1697→      const best = found[0];
  1698→      // Heuristic: if value > 100, probably in cents → divide by 100
  1699→      price = best.value > 100 ? best.value / 100 : best.value;
  1700→      priceSource = `embedded-${best.field}`;
  1701→    }
  1702→  }
  1703→
  1704→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1705→  if (!price) {
  1706→    const text = html.replace(/<[^>]*>/g, " ");
  1707→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1708→    const prices: { value: number; currency: string }[] = [];
  1709→    for (const m of matches) {
  1710→      const raw = m[1].replace(/,/g, "");
  1711→      const v = parseFloat(raw);
  1712→      if (v > 1.5 && v < 5000) {
  1713→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1714→        prices.push({ value: v, currency: cur });
  1715→      }
  1716→    }
  1717→    if (prices.length > 0) {
  1718→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1719→      const sorted = prices.sort((a, b) => a.value - b.value);
  1720→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1721→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1722→      if (filtered.length > 0) {
  1723→        price = filtered[0].value;
  1724→        currency = filtered[0].currency;
  1725→        priceSource = "text";
  1726→      }
  1727→    }
  1728→  }
  1729→
  1730→  return {
  1731→    price,
  1732→    currency,
  1733→    productName,
  1734→    productDescription,
  1735→    canonicalUrl: canonical,
  1736→    originalPrice,
  1737→    image: ogImage,
  1738→    source: priceSource || (productName ? "og-only" : "none"),
  1739→  };
  1740→}

  1771→/* ───────────────────────────────────────────────────────────────────
  1772→ * Strategy -2: ZAI LLM Direct Price Query (MOST RELIABLE for share URLs)
  1773→ *
  1774→ * Temu aggressively blocks all server-side scraping with anti-bot
  1775→ * challenges. Even the page_reader gets the login page instead of
  1776→ * the product page. This strategy uses the ZAI LLM with web search
  1777→ * to directly find the product price.
  1778→ *
  1779→ * How it works:
  1780→ * 1. Search the web for the product (by goods_id or Item ID)
  1781→ * 2. Extract prices from search result snippets (in various currencies)
  1782→ * 3. Use the LLM to interpret and convert prices to USD
  1783→ *
  1784→ * This is the most reliable strategy because:
  1785→ * - It doesn't depend on scraping (which Temu blocks)
  1786→ * - Search engines have already indexed Temu pages with prices
  1787→ * - The LLM can handle different currencies and formats
  1788→ * ─────────────────────────────────────────────────────────────────── */
  1789→async function fetchPriceWithLLMDirect(
  1790→  goodsId: string | null,
  1791→  itemId: string | null,
  1792→  productNameHint: string | null,
  1793→  userLocale: string | null = null,
  1794→): Promise<TemuProductData | null> {
  1795→  try {
  1796→    const zai = await createZAI();
  1797→
  1798→    // Step 1: Web search to find the product with prices
  1799→    // Try multiple search queries to maximize the chance of finding a price
  1800→    const searchQueries: string[] = [];
  1801→
  1802→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
  1803→      // Item ID search
  1804→      searchQueries.push(`temu "${itemId}" price`);
  1805→      searchQueries.push(`site:temu.com ${itemId}`);
  1806→    }
  1807→    if (goodsId && /^\d{10,}$/.test(goodsId)) {
  1808→      // Goods ID search - include "OFF" to find pages with discount prices
  1809→      searchQueries.push(`site:temu.com ${goodsId}`);
  1810→      // Also search with product name hint if available
  1811→      if (productNameHint) {
  1812→        const shortName = productNameHint.replace(/\s+/g, " ").trim().slice(0, 60);
  1813→        searchQueries.push(`temu ${shortName} price`);
  1814→      }
  1815→    }
  1816→
  1817→    if (searchQueries.length === 0) return null;
  1818→
  1819→    // Collect search results from multiple queries
  1820→    const allResults: { name: string; url: string; snippet: string }[] = [];
  1821→    const seenUrls = new Set<string>();
  1822→
  1823→    for (const query of searchQueries) {
  1824→      try {
  1825→        console.log(`[LLM-Direct] Searching: ${query}`);
  1826→        const results = await (zai as any).invokeFunction("web_search", {
  1827→          query,
  1828→          num: 8,
  1829→        });
  1830→
  1831→        if (Array.isArray(results)) {
  1832→          for (const r of results) {
  1833→            if (r.url && !seenUrls.has(r.url)) {
  1834→              seenUrls.add(r.url);
  1835→              allResults.push({
  1836→                name: r.name || "",
  1837→                url: r.url,
  1838→                snippet: r.snippet || "",
  1839→              });
  1840→            }
  1841→          }
  1842→        }
  1843→      } catch (err) {
  1844→        console.log(`[LLM-Direct] Search error:`, String(err).slice(0, 80));
  1845→      }
  1846→    }
  1847→
  1848→    if (allResults.length === 0) {
  1849→      console.log("[LLM-Direct] No search results found");
  1850→      return null;
  1851→    }
  1852→
  1853→    console.log(`[LLM-Direct] Collected ${allResults.length} search results`);
  1854→
  1855→    // Step 2: Try to extract prices directly from snippets first (fast, no LLM cost)

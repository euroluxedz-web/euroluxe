  2518→function isSuspiciousPrice(priceUSD: number, source: string): boolean {

  2521→  if (priceUSD === 30.00 || priceUSD === 30) {

  2523→    return true;
  2524→  }
  2525→  // Also check for $30.xx variations
  2526→  if (priceUSD >= 29.90 && priceUSD <= 30.10) {

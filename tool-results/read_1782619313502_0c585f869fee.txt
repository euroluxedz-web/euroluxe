   100→export default function CalculateurPage() {
   101→  const [productUrl, setProductUrl] = useState("");
   102→  const [manualPrice, setManualPrice] = useState("");
   103→  const [loading, setLoading] = useState(false);
   104→  const [result, setResult] = useState<PriceResult | null>(null);
   105→  const [error, setError] = useState("");
   106→  const [copied, setCopied] = useState(false);
   107→  const [addedToCart, setAddedToCart] = useState(false);
   108→  const [detectedCode, setDetectedCode] = useState<string | null>(null);
   109→  const [temuLink, setTemuLink] = useState<string | null>(null);
   110→  const [detectedProduct, setDetectedProduct] = useState<DetectedProduct | null>(null);
   111→  const priceInputRef = useRef<HTMLInputElement>(null);
   112→  const { t, isArabic } = useLanguage();
   113→  const { user, profile } = useAuth();
   114→  const router = useRouter();
   115→  const addItemToStore = useCartStore((s) => s.addItem);
   116→
   117→  // Checkout state
   118→  const [showCheckout, setShowCheckout] = useState(false);
   119→  const [orderSuccess, setOrderSuccess] = useState(false);
   120→  const [submitting, setSubmitting] = useState(false);
   121→  const [saveInfo, setSaveInfo] = useState(true);
   122→  const [useSaved, setUseSaved] = useState(false);
   123→  const [shippingError, setShippingError] = useState("");
   124→
   125→  const [shipping, setShipping] = useState<ShippingInfo>({
   126→    fullName: "",
   127→    phone: "",
   128→    wilaya: "",
   129→    commune: "",
   130→    codePostal: "",
   131→    address: "",
   132→    notes: "",
   133→  });
   134→
   135→  // Available communes based on selected wilaya
   136→  const [availableCommunes, setAvailableCommunes] = useState<Commune[]>([]);
   137→
   138→  const isAuthenticated = !!user;
   139→
   140→  // Pre-fill shipping info from user profile
   141→  useEffect(() => {
   142→    if (profile && isAuthenticated) {
   143→      const hasSavedInfo = profile.phone || profile.wilaya || profile.address;
   144→      if (hasSavedInfo) {
   145→        setUseSaved(true);
   146→        const communes = getCommunesForWilaya(profile.wilaya || "");
   147→        setAvailableCommunes(communes);
   148→        setShipping({
   149→          fullName: profile.name || "",
   150→          phone: profile.phone || "",
   151→          wilaya: profile.wilaya || "",
   152→          commune: profile.commune || "",
   153→          codePostal: profile.codePostal || "",
   154→          address: profile.address || "",
   155→          notes: "",
   156→        });
   157→      } else {
   158→        setShipping((prev) => ({
   159→          ...prev,
   160→          fullName: profile.name || "",
   161→          phone: profile.phone || "",
   162→        }));
   163→      }
   164→    }
   165→  }, [profile, isAuthenticated]);
   166→
   167→  // Detect Temu product code / Item ID in real-time
   168→  useEffect(() => {
   169→    const input = productUrl.trim();
   170→    const isTemuCode = /^[a-zA-Z0-9]{6,30}$/.test(input);
   171→    if (isTemuCode) {
   172→      setDetectedCode(input);
   173→      // Item IDs (like TV10922608) don't work in -g- URL format, use search instead
   174→      if (/^[A-Z]{2}\d+/i.test(input)) {
   175→        setTemuLink(`https://www.temu.com/search_result.html?search_key=${input}`);
   176→      } else {
   177→        setTemuLink(`https://www.temu.com/-g-${input}.html`);
   178→      }
   179→    } else if (input.includes("temu.com")) {
   180→      setDetectedCode(null);
   181→      setTemuLink(input.startsWith("http") ? input : `https://${input}`);
   182→    } else {
   183→      setDetectedCode(null);
   184→      setTemuLink(null);
   185→    }
   186→  }, [productUrl]);
   187→
   188→  // Extract product name from URL
   189→  const extractProductName = (url: string): string | null => {
   190→    try {
   191→      const parsed = new URL(url.startsWith("http") ? url : `https://${url}`);
   192→      const segments = parsed.pathname.split("/").filter(Boolean);
   193→      const slug =
   194→        segments.find((s) => s.includes("-g-") && s.length > 10) ||
   195→        segments.find((s) => s.includes("-") && s.length > 10) ||
   196→        segments[segments.length - 1] ||
   197→        "";
   198→      const name = slug
   199→        .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   200→        .replace(/\.html?$/i, "")
   201→        .replace(/-/g, " ")
   202→        .trim();
   203→      if (name && name.length > 3) {
   204→        return name.replace(/\b\w/g, (l) => l.toUpperCase());
   205→      }
   206→    } catch { /* skip */ }
   207→    return null;
   208→  };
   209→
   210→  // AUTO-EXTRACT
   211→  const handleAutoExtract = async () => {
   212→    setError("");
   213→    setResult(null);
   214→    setShowCheckout(false);
   215→    setOrderSuccess(false);
   216→    setDetectedProduct(null);
   217→
   218→    if (!productUrl.trim()) {
   219→      setError(t("calc.error.empty"));
   220→      return;
   221→    }
   222→
   223→    const isTemuProductId = /^[a-zA-Z0-9]{6,30}$/.test(productUrl.trim());
   224→    let finalUrl = productUrl.trim();
   225→
   226→    if (isTemuProductId) {
   227→      // Item ID format (e.g. TV10922608) — pass as-is, the API will use web search
   228→      // Numeric goods_id — construct -g- URL
   229→      if (/^[A-Z]{2}\d+/i.test(productUrl.trim())) {
   230→        finalUrl = productUrl.trim(); // Pass Item ID directly
   231→      } else {
   232→        finalUrl = `https://www.temu.com/-g-${productUrl.trim()}.html`;
   233→      }
   234→    } else {
   235→      try {
   236→        new URL(finalUrl);
   237→      } catch {
   238→        setError(t("calc.error.invalidUrl"));
   239→        return;
   240→      }
   241→    }
   242→
   243→    setLoading(true);
   244→
   245→    try {
   246→      const response = await fetch("/api/scrape-price", {
   247→        method: "POST",
   248→        headers: { "Content-Type": "application/json" },
   249→        body: JSON.stringify({ url: finalUrl }),
   250→      });
   251→
   252→      const data = await response.json();
   253→
   254→      // Case 1: Auto-extracted price found
   255→      if (data.success && data.price && data.price > 0) {
   256→        setTemuLink(finalUrl);
   257→        setResult({
   258→          usd: data.price,
   259→          dzd: data.dzd || data.price * 270,
   260→          breakdown: data.breakdown,
   261→          productName: data.productName,
   262→          originalPrice: data.originalPrice || null,
   263→          image: data.productImage || data.image || null,
   264→          estimated: data.estimated || false,
   265→          manual: data.manual || false,
   266→          source: data.source || "auto",
   267→          itemId: data.itemId || undefined,
   268→        });
   269→      }
   270→      // Case 2: Product detected but needs manual price entry
   271→      else if (data.success && data.requiresManualPrice && data.productName) {
   272→        setTemuLink(data.productUrl || finalUrl);
   273→        setDetectedProduct({
   274→          name: data.productName,
   275→          description: data.productDescription,
   276→          image: data.productImage || null,
   277→          url: data.productUrl || finalUrl,
   278→          antiBotDetected: data.antiBotDetected,
   279→          message: data.message,
   280→        });
   281→        setError(
   282→          data.message ||
   283→            (isArabic
   284→              ? "تم العثور على المنتج! يرجى إدخال السعر المعروض على Temu في الحقل أدناه."
   285→              : "Produit trouvé ! Veuillez saisir le prix affiché sur Temu dans le champ ci-dessous.")
   286→        );
   287→        setTimeout(() => priceInputRef.current?.focus(), 300);
   288→      }
   289→      // Case 3: Failure
   290→      else {
   291→        setTemuLink(finalUrl);
   292→        setError(
   293→          data.error ||
   294→            (isArabic
   295→              ? "لم نتمكن من استخراج السعر تلقائياً. يرجى إدخاله يدوياً."
   296→              : "Extraction automatique indisponible. Veuillez entrer le prix manuellement.")
   297→        );
   298→        setTimeout(() => priceInputRef.current?.focus(), 300);
   299→      }
   300→    } catch {
   301→      setError(
   302→        isArabic
   303→          ? "يرجى إدخال السعر يدوياً في الحقل أدناه."
   304→          : "Veuillez entrer le prix manuellement dans le champ ci-dessous."
   305→      );
   306→      setTimeout(() => priceInputRef.current?.focus(), 300);
   307→    } finally {
   308→      setLoading(false);
   309→    }
   310→  };
   311→
   312→  // Manual price calculation - uses API for proper breakdown
   313→  const handleManualCalculate = async () => {
   314→    setError("");
   315→    setResult(null);
   316→    setShowCheckout(false);
   317→    setOrderSuccess(false);
   318→
   319→    const normalized = manualPrice.trim().replace(/,/g, ".");
   320→    const priceStr = normalized.replace(/[^\d.]/g, "");
   321→    const parts = priceStr.split(".");
   322→    const cleaned = parts.length > 1
   323→      ? parts[0] + "." + parts.slice(1).join("")
   324→      : priceStr;
   325→    const price = parseFloat(cleaned);
   326→
   327→    if (!price || price <= 0) {
   328→      setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   329→      return;
   330→    }
   331→
   332→    setLoading(true);
   333→    try {
   334→      // Pass the detected product info + URL so the API can use it
   335→      const payload: Record<string, string> = { manualPrice: manualPrice.trim() };
   336→      if (detectedProduct?.name) payload.productName = detectedProduct.name;
   337→      if (detectedProduct?.image) payload.productImage = detectedProduct.image;
   338→      if (productUrl.trim()) payload.url = productUrl.trim();
   339→
   340→      const response = await fetch("/api/scrape-price", {
   341→        method: "POST",
   342→        headers: { "Content-Type": "application/json" },
   343→        body: JSON.stringify(payload),
   344→      });
   345→      const data = await response.json();
   346→
   347→      if (data.success && data.price) {
   348→        const productName = detectedProduct?.name
   349→          || (productUrl.trim() ? extractProductName(productUrl) : null)
   350→          || data.productName;
   351→        setResult({
   352→          usd: data.price,
   353→          dzd: data.dzd,
   354→          breakdown: data.breakdown,
   355→          productName,
   356→          image: detectedProduct?.image || null,
   357→          estimated: false,
   358→          manual: true,
   359→          source: "manual",
   360→        });
   361→      } else {
   362→        setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   363→      }
   364→    } catch {
   365→      setError(isArabic ? "حدث خطأ، حاول مرة أخرى" : "Une erreur est survenue, réessayez");
   366→    } finally {
   367→      setLoading(false);
   368→    }
   369→  };
   370→
   371→  const handleCopyResult = () => {
   372→    if (result) {
   373→      const text = result.productName
   374→        ? `${result.productName} — ${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`
   375→        : `${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`;
   376→      navigator.clipboard.writeText(text);
   377→      setCopied(true);
   378→      setTimeout(() => setCopied(false), 2000);
   379→    }
   380→  };
   381→
   382→  const handleAddToCart = () => {
   383→    if (!result) return;
   384→    const cartItem = {
   385→      id: `calc-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   386→      productId: detectedCode || undefined,
   387→      name: result.productName || (isArabic ? "منتج" : "Produit"),
   388→      image: result.image || undefined,
   389→      price: result.usd,
   390→      quantity: 1,
   391→      url: productUrl.trim() || undefined,
   392→    };
   393→    addItemToStore(cartItem);
   394→    syncAddToServer(cartItem);
   395→    setAddedToCart(true);
   396→    setTimeout(() => setAddedToCart(false), 2000);
   397→  };
   398→
   399→  // Validate phone
   400→  const isValidPhone = (phone: string) => /^(05|06|07)\d{8}$/.test(phone.trim());
   401→
   402→  // Handle checkout submission
   403→  const handleSubmitOrder = async () => {
   404→    setShippingError("");
   405→
   406→    // Validate
   407→    if (!shipping.fullName.trim()) {
   408→      setShippingError(isArabic ? "يرجى إدخال الاسم الكامل" : "Veuillez entrer votre nom complet");
   409→      return;
   410→    }
   411→    if (!isValidPhone(shipping.phone)) {
   412→      setShippingError(t("calc.checkout.errorPhone"));
   413→      return;
   414→    }
   415→    if (!shipping.wilaya) {
   416→      setShippingError(isArabic ? "يرجى اختيار الولاية" : "Veuillez sélectionner votre wilaya");
   417→      return;
   418→    }
   419→    if (!shipping.commune) {
   420→      setShippingError(isArabic ? "يرجى اختيار البلدية" : "Veuillez sélectionner votre commune");
   421→      return;
   422→    }
   423→    if (!shipping.address.trim()) {
   424→      setShippingError(isArabic ? "يرجى إدخال العنوان" : "Veuillez entrer votre adresse");
   425→      return;
   426→    }
   427→
   428→    setSubmitting(true);
   429→
   430→    try {
   431→      // Save shipping info to user profile if requested
   432→      if (isAuthenticated && saveInfo && user) {
   433→        try {
   434→          await updateUserData(user.uid, {
   435→            name: shipping.fullName,
   436→            phone: shipping.phone,
   437→            wilaya: shipping.wilaya,
   438→            commune: shipping.commune,
   439→            codePostal: shipping.codePostal,
   440→            address: shipping.address,
   441→          });
   442→        } catch (e) {
   443→          console.error("Failed to save shipping info:", e);
   444→        }
   445→      }
   446→
   447→      // Create the order
   448→      if (isAuthenticated && user) {
   449→        const orderItems = [{
   450→          name: result?.productName || (isArabic ? "منتج" : "Produit"),
   451→          price: result?.dzd || 0,
   452→          quantity: 1,
   453→          image: result?.image || undefined,
   454→          url: productUrl.trim() || undefined,
   455→          productId: detectedCode || undefined,
   456→        }];
   457→
   458→        const { auth } = await import("@/lib/firebase");
   459→        const token = auth.currentUser ? await auth.currentUser.getIdToken() : null;
   460→
   461→        const res = await fetch("/api/orders", {
   462→          method: "POST",
   463→          headers: {
   464→            "Content-Type": "application/json",
   465→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   466→          },
   467→          body: JSON.stringify({
   468→            items: orderItems,
   469→            total: result?.dzd || 0,
   470→            wilaya: shipping.wilaya,
   471→            commune: shipping.commune,
   472→            codePostal: shipping.codePostal,
   473→            address: shipping.address,
   474→            phone: shipping.phone,
   475→            notes: shipping.notes,
   476→          }),
   477→        });
   478→
   479→        if (res.ok) {
   480→          setOrderSuccess(true);
   481→        } else {
   482→          setShippingError(isArabic ? "حدث خطأ، يرجى المحاولة مرة أخرى" : "Une erreur est survenue, veuillez réessayer");
   483→        }
   484→      }
   485→    } catch (err) {
   486→      console.error("Order error:", err);
   487→      setShippingError(isArabic ? "حدث خطأ في الاتصال" : "Erreur de connexion");
   488→    } finally {
   489→      setSubmitting(false);
   490→    }
   491→  };
   492→
   493→  return (
   494→    <div className="relative min-h-screen flex flex-col bg-transparent text-foreground overflow-x-hidden">
   495→      <Navbar />
   496→      <main className="flex-1 pt-16 sm:pt-20">
   497→        <section className="relative py-20 sm:py-32 overflow-hidden min-h-[80vh]">
   498→          {/* Background Effects */}
   499→          <div className="absolute inset-0 bg-gradient-to-b from-brand-blue/20 via-brand-blue-light/15 to-white/60" />
   500→          <div className="absolute top-0 left-0 w-72 h-72 bg-brand-pink/8 rounded-full blur-3xl" />
   501→          <div className="absolute bottom-0 right-0 w-72 h-72 bg-brand-blue/8 rounded-full blur-3xl" />
   502→
   503→          <div className="relative z-10 max-w-4xl mx-auto px-4">
   504→            {/* Section Header */}
   505→            <motion.div
   506→              initial={{ opacity: 0, y: 30 }}
   507→              animate={{ opacity: 1, y: 0 }}
   508→              transition={{ duration: 0.6 }}
   509→              className="text-center mb-12 relative"
   510→            >
   511→              <div className="hidden lg:block">
   512→                <ImgPlaceholder number={32} className="absolute -left-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[-8deg]" />
   513→                <ImgPlaceholder number={33} className="absolute -right-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[8deg]" />
   514→              </div>
   515→
   516→              <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-brand-pink/10 border border-brand-pink/20 text-brand-pink text-sm font-medium mb-4 font-display">
   517→                <Zap className="w-4 h-4" />
   518→                {t("calc.badge")}
   519→              </div>
   520→              <h1 className="text-3xl sm:text-5xl lg:text-6xl font-black mb-4 font-heading">
   521→                <span className="text-brand-dark">{t("calc.titleCalc")}</span>{" "}
   522→                <span className="bg-brand-gold/30 px-2 py-1 rounded-md text-brand-dark">
   523→                  {t("calc.titleProduct")}
   524→                </span>
   525→              </h1>
   526→              <p className="text-brand-muted-text text-lg max-w-xl mx-auto font-sans">
   527→                {t("calc.subtitle")}
   528→              </p>
   529→            </motion.div>
   530→
   531→            {/* Calculator Card */}
   532→            <motion.div
   533→              initial={{ opacity: 0, y: 30 }}
   534→              animate={{ opacity: 1, y: 0 }}
   535→              transition={{ duration: 0.6, delay: 0.2 }}
   536→              className="bg-white/80 backdrop-blur-xl rounded-3xl p-6 sm:p-10 border border-brand-pink/15 shadow-lg"
   537→            >
   538→              {/* ──── Product URL / Code Input ──── */}
   539→              <div className="mb-6">
   540→                <label className="block text-brand-dark/80 text-sm font-medium mb-2 font-sans">
   541→                  <Link2 className={`w-4 h-4 inline ${isArabic ? "ml-1" : "mr-1"}`} />
   542→                  {t("calc.label")}
   543→                </label>
   544→                <div className="flex gap-3">
   545→                  <div className="relative flex-1">
   546→                    <Input
   547→                      type="text"
   548→                      placeholder={t("calc.placeholder")}
   549→                      value={productUrl}
   550→                      onChange={(e) => {
   551→                        setProductUrl(e.target.value);
   552→                        setResult(null);
   553→                        setError("");
   554→                        setShowCheckout(false);
   555→                        setDetectedProduct(null);
   556→                      }}

   332→/* ───────────────────────────────────────────────────────────────────
   333→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   334→ *
   335→ * Temu share URLs include everything we need:
   336→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   337→ *   - top_gallery_url    = URL-encoded product image URL
   338→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   339→ *
   340→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   341→ * ─────────────────────────────────────────────────────────────────── */
   342→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   343→  let parsed: URL;
   344→  try {
   345→    parsed = new URL(originalUrl);
   346→  } catch {
   347→    return null;
   348→  }
   349→
   350→  let price: number | null = null;
   351→  let currency = "USD";
   352→  let priceSource = "";
   353→
   354→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   355→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   356→  if (hint) {
   357→    try {
   358→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   359→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   360→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   361→      if (cents > 0 && cents < 10000000) {
   362→        const usd = cents / 100;
   363→        if (usd >= 0.01 && usd < 100000) {
   364→          price = usd;
   365→          currency = "USD";
   366→          priceSource = "url-hint";
   367→        }
   368→      }
   369→    } catch { /* not valid base64 */ }
   370→  }
   371→
   372→  // Product image from top_gallery_url
   373→  let image: string | null = null;
   374→  const topGallery = parsed.searchParams.get("top_gallery_url");
   375→  if (topGallery) {
   376→    try {
   377→      // Validate it's a real URL
   378→      const imgUrl = new URL(topGallery);
   379→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   380→        image = topGallery;
   381→      }
   382→    } catch { /* not a valid URL */ }
   383→  }
   384→
   385→  // Product name from URL slug
   386→  let productName: string | null = null;
   387→  const segments = parsed.pathname.split("/").filter(Boolean);
   388→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   389→  const nameFromSlug = slug
   390→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   391→    .replace(/\.html?$/i, "")
   392→    .replace(/-/g, " ")
   393→    .replace(/\s+/g, " ")
   394→    .trim();
   395→  if (nameFromSlug && nameFromSlug.length > 3) {
   396→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   397→  }
   398→
   399→  // If we got at least the price (the most important piece), return success
   400→  if (price) {
   401→    return {
   402→      price,
   403→      currency,
   404→      productName,
   405→      productDescription: null,
   406→      canonicalUrl: null,
   407→      originalPrice: null,
   408→      image,
   409→      source: priceSource,
   410→      antiBotDetected: false,
   411→    };
   412→  }
   413→
   414→  // No price from URL — return null so caller tries HTML-based strategies
   415→  return null;
   416→}

   476→  itemId: string | null,
   477→  originalUrl: string,
   478→): Promise<TemuProductData | null> {
   479→  try {
   480→    const zai = await createZAI();
   481→
   482→    // Build search query — try different strategies
   483→    let searchQuery = "";
   484→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   485→      // Item ID like TV10922608 — search broadly on Temu
   486→      searchQuery = `temu ${itemId}`;
   487→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   488→      // Numeric goods_id — search with the -g- pattern
   489→      searchQuery = `site:temu.com "g-${goodsId}"`;
   490→    } else {
   491→      // Fallback: search with the URL
   492→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   493→      searchQuery = `site:temu.com ${cleanUrl}`;
   494→    }
   495→
   496→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   497→
   498→    // Step 1: Web search to find the product
   499→    const searchResults = await (zai as any).invokeFunction("web_search", {
   500→      query: searchQuery,
   501→      num: 5,
   502→    });
   503→
   504→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   505→      console.log("[WebSearch] No results found, trying broader search...");
   506→
   507→      // Broader search without site: restriction
   508→      const broadQuery = itemId
   509→        ? `temu ${itemId} price`
   510→        : goodsId
   511→          ? `temu ${goodsId} price`
   512→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   513→
   514→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   515→      const broadResults = await (zai as any).invokeFunction("web_search", {
   516→        query: broadQuery,
   517→        num: 5,
   518→      });
   519→
   520→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   521→        console.log("[WebSearch] No results from broader search either");
   522→        return null;
   523→      }
   524→
   525→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   526→    }
   527→
   528→    // Step 2: Try to parse price from initial search results
   529→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   530→    if (initialPrice?.price && initialPrice.price > 0) {
   531→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   532→      return initialPrice;
   533→    }
   534→
   535→    // Step 3: If initial search found product names but no prices,
   536→    // try a more targeted search with the product name to find prices
   537→    const productNameFromResults = searchResults
   538→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   539→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   540→      .find((n: string) => n && n.length > 10);
   541→
   542→    if (productNameFromResults) {
   543→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   544→      try {
   545→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   546→        const nameResults = await (zai as any).invokeFunction("web_search", {
   547→          query: nameQuery,
   548→          num: 5,
   549→        });
   550→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   551→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   552→          if (namePrice?.price && namePrice.price > 0) {
   553→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   554→            return namePrice;
   555→          }
   556→          // Merge results and try LLM on combined results
   557→          const combinedResults = [...searchResults, ...nameResults];
   558→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   559→        }
   560→      } catch (err) {
   561→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   562→      }
   563→    }
   564→
   565→    // Step 4: Use LLM on whatever results we have
   566→    const llmResult = await extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   567→    if (llmResult?.price && llmResult.price > 0) {
   568→      return llmResult;
   569→    }
   570→
   571→    // Step 5: If LLM didn't find price, try reading Temu search result pages
   572→    // via AllOrigins proxy — sometimes the proxy can access the product page
   573→    // when direct access is blocked
   574→    const temuResults = searchResults
   575→      .filter((r: any) => r.url?.includes("temu.com") && r.url?.includes("-g-"))
   576→      .slice(0, 2);
   577→
   578→    for (const result of temuResults) {
   579→      try {
   580→        console.log(`[WebSearch] Trying AllOrigins for search result: ${result.url.slice(0, 80)}...`);
   581→        // Try both /raw and /get endpoints
   582→        const proxyUrls = [
   583→          `https://api.allorigins.win/raw?url=${encodeURIComponent(result.url)}`,
   584→          `https://api.allorigins.win/get?url=${encodeURIComponent(result.url)}`,
   585→        ];
   586→
   587→        let proxyHtml: string | null = null;
   588→
   589→        for (const pUrl of proxyUrls) {
   590→          try {
   591→            const controller = new AbortController();
   592→            const timeout = setTimeout(() => controller.abort(), 15000);
   593→            const proxyRes = await fetch(pUrl, { signal: controller.signal });
   594→            clearTimeout(timeout);
   595→            if (!proxyRes.ok) continue;
   596→
   597→            const isRaw = pUrl.includes("/raw?");
   598→            if (isRaw) {
   599→              const text = await proxyRes.text();
   600→              if (text && text.length > 5000 && text.includes("<")) {
   601→                proxyHtml = text;
   602→                break;
   603→              }
   604→            } else {
   605→              const data = await proxyRes.json();
   606→              const contents = typeof data === "string" ? data : data?.contents;
   607→              if (contents && typeof contents === "string" && contents.length > 5000) {
   608→                proxyHtml = contents;
   609→                break;
   610→              }
   611→            }
   612→          } catch { /* try next */ }
   613→        }
   614→
   615→        if (!proxyHtml || proxyHtml.length < 5000) continue;
   616→
   617→        // Check if we got the real product page (has OG title)
   618→        const hasOGTitle = /<meta[^>]*property=["']og:title["']/i.test(proxyHtml);
   619→        if (!hasOGTitle) continue;
   620→
   621→        // Extract price using extractProductInfo
   622→        const proxyResult = extractProductInfo(proxyHtml, result.url);
   623→        if (proxyResult.price && proxyResult.price > 0) {
   624→          console.log(`[WebSearch] ✓ Got price $${proxyResult.price} from AllOrigins search result`);
   625→          return proxyResult;
   626→        }
   627→
   628→        // Also try direct OG price extraction
   629→        const ogPriceMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   630→        const ogCurrencyMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   631→        if (ogPriceMatch) {
   632→          const priceVal = parseFloat(ogPriceMatch[1]);
   633→          const cur = ogCurrencyMatch?.[1] || "USD";
   634→          if (priceVal > 0 && priceVal < 100000) {
   635→            let priceUSD = priceVal;
   636→            if (cur !== "USD" && CURRENCY_TO_USD[cur]) {
   637→              priceUSD = Math.round(priceVal * CURRENCY_TO_USD[cur] * 100) / 100;
   638→            }
   639→            const ogTitle = proxyHtml.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1];
   640→            const ogImage = proxyHtml.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1];
   641→            const foundGoodsId = result.url.match(/-g-(\d{10,})/)?.[1] || goodsId;
   642→            console.log(`[WebSearch] ✓ Got OG price ${priceVal} ${cur} = $${priceUSD} from AllOrigins search result`);
   643→            return {
   644→              price: priceUSD,
   645→              currency: "USD",
   646→              productName: ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null,
   647→              productDescription: null,
   648→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   649→              originalPrice: null,
   650→              image: ogImage || null,
   651→              source: `websearch-allorigins-og(${cur})`,
   652→              antiBotDetected: false,
   653→            };
   654→          }
   655→        }
   656→      } catch (err) {
   657→        console.log(`[WebSearch] AllOrigins for search result failed:`, String(err).slice(0, 80));
   658→      }
   659→    }
   660→
   661→    return llmResult; // Return whatever the LLM found (may be null)
   662→  } catch (err) {
   663→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   664→    return null;
   665→  }
   666→}

   668→async function extractPriceFromSearchResults(
   669→  zai: any,
   670→  searchResults: any[],
   671→  goodsId: string | null,
   672→  itemId: string | null,
   673→): Promise<TemuProductData | null> {
   674→  // Step 2: Try to parse prices directly from search snippets first (fast, no LLM cost)
   675→  const snippetPrice = parsePriceFromSnippets(searchResults, goodsId);
   676→  if (snippetPrice?.price && snippetPrice.price > 0) {
   677→    console.log(`[WebSearch] ✓ Found price from snippet: ${snippetPrice.price} USD (source: ${snippetPrice.source})`);
   678→    return snippetPrice;
   679→  }
   680→
   681→  // Step 3: If snippets don't have a clear price, use LLM to extract
   682→  console.log("[WebSearch] Snippets don't have clear price, using LLM...");
   683→
   684→  const searchContext = searchResults
   685→    .slice(0, 5)
   686→    .map(
   687→      (r: any, i: number) =>
   688→        `${i + 1}. ${r.name || "No title"}\n   URL: ${r.url}\n   Snippet: ${r.snippet || "No snippet"}`,
   689→    )
   690→    .join("\n\n");
   691→
   692→  try {
   693→    const completion = await zai.createChatCompletion({
   694→      messages: [
   695→        {
   696→          role: "system",
   697→          content:
   698→            'You are a price extraction assistant for Temu products. Extract the product price from the search results. ' +
   699→            'Return ONLY a JSON object with: {"price_usd": <number_in_USD>, "name": "<product_name>", "currency": "<original_currency>", "original_price": <number_before_conversion>, "confidence": <high|medium|low>}. ' +
   700→            'If the price is in a non-USD currency, CONVERT it to USD using approximate exchange rates. ' +
   701→            'Common conversions: MUR→USD ÷47, OMR→USD ×2.60, BHD→USD ×2.65, PKR→USD ÷278, EUR→USD ×1.08, GBP→USD ×1.27, SAR→USD ×0.27, AED→USD ×0.27. ' +
   702→            'If you cannot find a clear price, return {"price_usd": null, "name": "<best_guess_name>", "confidence": "low"}. ' +
   703→            'ALWAYS return valid JSON. The price_usd must be a number, not a string. ' +
   704→            'IMPORTANT: Do NOT estimate or guess the price. If the search results do not contain a clear price for this specific product, return null. ' +
   705→            'Do NOT confuse "delivery guarantee" prices, "credit for delay" amounts, or "30% OFF" discount percentages with the product price. ' +
   706→            'Do NOT use prices from unrelated products. Only extract the price that clearly corresponds to the product with the given goods_id or Item ID.',
   707→        },
   708→        {
   709→          role: "user",
   710→          content:
   711→            `Product goods_id: ${goodsId || "unknown"}\n` +
   712→            `Item ID: ${itemId || "unknown"}\n\n` +
   713→            `Search Results:\n${searchContext}\n\n` +
   714→            `Extract the product price in USD from these results. Return JSON only.`,
   715→        },
   716→      ],
   717→    });
   718→
   719→    const aiResponse = completion.choices?.[0]?.message?.content || "";
   720→    console.log("[WebSearch] LLM response:", aiResponse.slice(0, 300));
   721→
   722→    // Parse AI response — extract JSON
   723→    const jsonMatch = aiResponse.match(/\{[\s\S]*?\}/);
   724→    if (!jsonMatch) {
   725→      console.log("[WebSearch] No JSON in LLM response");
   726→      return null;
   727→    }
   728→
   729→    const parsed = JSON.parse(jsonMatch[0]);
   730→    const priceUSD = typeof parsed.price_usd === "number"
   731→      ? parsed.price_usd
   732→      : parseFloat(String(parsed.price_usd));
   733→
   734→    if (priceUSD && priceUSD > 0 && priceUSD < 100000) {
   735→      console.log(`[WebSearch] ✓ LLM found price: $${priceUSD} (confidence: ${parsed.confidence})`);
   736→      return {
   737→        price: priceUSD,
   738→        currency: "USD",
   739→        productName: parsed.name || null,
   740→        productDescription: null,
   741→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   742→        originalPrice: null,
   743→        image: null, // Image will come from share URL params
   744→        source: `web-search-llm(${parsed.confidence})`,
   745→        antiBotDetected: false,
   746→      };
   747→    }
   748→
   749→    // LLM couldn't find price either — return what we can
   750→    if (parsed.name) {
   751→      return {
   752→        price: null,
   753→        currency: "USD",
   754→        productName: parsed.name,
   755→        productDescription: null,
   756→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   757→        originalPrice: null,
   758→        image: null,
   759→        source: "web-search-llm(no-price)",
   760→        antiBotDetected: false,
   761→      };
   762→    }
   763→
   764→    return null;
   765→  } catch (err) {
   766→    console.error("[WebSearch] LLM error:", String(err).slice(0, 150));
   767→    return null;
   768→  }

   771→/**
   772→ * Try to parse prices directly from search result snippets.
   773→ * This avoids the cost of an LLM call and is often sufficient
   774→ * because Google includes price info in Temu search snippets.
   775→ */
   776→function parsePriceFromSnippets(
   777→  results: any[],
   778→  goodsId: string | null,
   779→): TemuProductData | null {
   780→  // Look for Temu results that have price info in snippets
   781→  for (const result of results) {
   782→    if (!result.url?.includes("temu.com")) continue;
   783→    if (!result.snippet) continue;
   784→
   785→    const snippet = result.snippet;
   786→
   787→    // Extract the goods_id from the URL if we don't have it
   788→    let foundGoodsId = goodsId;
   789→    if (!foundGoodsId) {
   790→      const gMatch = result.url.match(/-g-(\d{10,})/);
   791→      if (gMatch) foundGoodsId = gMatch[1];
   792→    }
   793→
   794→    // Extract product name from search result
   795→    const productName = result.name
   796→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   797→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   798→      .trim() || null;
   799→
   800→    // Determine the local currency from the Temu URL locale prefix
   801→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   802→    let localCurrency = "USD";
   803→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   804→    if (localeMatch) {
   805→      const locale = localeMatch[1].toLowerCase();
   806→      const localeToCurrency: Record<string, string> = {
   807→        mu: "MUR",       // Mauritius
   808→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   809→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   810→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   811→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   812→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   813→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   814→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   815→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   816→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   817→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   818→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   819→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   820→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   821→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   822→        np: "NPR", "np-en": "NPR", // Nepal
   823→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   824→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   825→        ph: "PHP", "ph-en": "PHP", // Philippines
   826→        br: "BRL", "br-pt": "BRL", // Brazil
   827→        mx: "MXN", "mx-es": "MXN", // Mexico
   828→      };
   829→      const found = localeToCurrency[locale];
   830→      if (found) localCurrency = found;
   831→    }
   832→
   833→    // Try to find a price in the snippet
   834→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   835→    // Common patterns:
   836→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   837→    //   "$6.99"                  → USD
   838→    //   "OMR 1.200"             → Omani Rial
   839→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   840→    //   "€5.99"                  → EUR
   841→    //   "£4.99"                  → GBP
   842→
   843→    // Determine if this locale uses European comma format (comma = decimal separator)
   844→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   845→    const europeanCommaLocales = new Set([
   846→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",
   847→      "ma-fr", "tn-fr", // French-speaking locales
   848→    ]);
   849→    const usesEuropeanComma = europeanCommaLocales.has(localCurrency === "USD" && localeMatch ? localeMatch[1].toLowerCase() : "") ||
   850→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   851→
   852→    // Helper: parse price string respecting locale format
   853→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   854→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   855→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   856→      if (isEuropean) {
   857→        // European: remove dots (thousands), replace comma with dot (decimal)
   858→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   859→      } else {
   860→        // English: remove commas (thousands)
   861→        return parseFloat(priceStr.replace(/,/g, ""));
   862→      }
   863→    };
   864→
   865→    // First, try explicit currency patterns (most reliable)
   866→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   867→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   868→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   869→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   870→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   871→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   872→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   873→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   874→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   875→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   876→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   877→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   878→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   879→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   880→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   881→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   882→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   883→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   884→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   885→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   886→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   887→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   888→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   889→    ];
   890→
   891→    for (const { pattern, currency } of explicitPatterns) {
   892→      const match = snippet.match(pattern);
   893→      if (match) {
   894→        // For USD prices on European locale pages (like Ecuador), use comma-as-decimal parsing
   895→        const isEuroPrice = currency === "USD" && usesEuropeanComma;
   896→        const localPrice = parsePriceString(match[1], isEuroPrice);
   897→        if (localPrice > 0 && localPrice < 10000000) {
   898→          const rate = CURRENCY_TO_USD[currency];
   899→          if (rate) {
   900→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   901→            if (usdPrice >= 0.10 && usdPrice <= 5000) {
   902→              console.log(`[WebSearch] Found explicit price: ${localPrice} ${currency} (European format: ${isEuroPrice}) = $${usdPrice} USD`);
   903→              return {
   904→                price: usdPrice,
   905→                currency: "USD",
   906→                productName,
   907→                productDescription: null,

  1405→        await new Promise(r => setTimeout(r, 1000));
  1406→        continue;
  1407→      }
  1408→
  1409→      const html = await response.text();
  1410→      console.log(`[ScrapingBee attempt ${attempt}] HTML length: ${html.length}, cost: ${response.headers.get("Spb-cost")} credits`);
  1411→
  1412→      const result = extractProductInfo(html, url);
  1413→
  1414→      // If we got the real product page (not anti-bot), use it
  1415→      // Heuristic: anti-bot page is ~310-325KB and has many "verify" mentions
  1416→      const isAntiBot = html.length < 450000 && (html.match(/verify/gi) || []).length > 100;
  1417→
  1418→      if (result.productName || result.price) {
  1419→        return {
  1420→          ...result,
  1421→          antiBotDetected: isAntiBot,
  1422→          source: isAntiBot ? "scrapingbee-og" : "scrapingbee-full",
  1423→        };
  1424→      }
  1425→
  1426→      // Got a page but no useful data — retry
  1427→      lastError = "Page returned but no product info extracted";
  1428→      await new Promise(r => setTimeout(r, 1000));
  1429→    } catch (err) {
  1430→      lastError = String(err).slice(0, 150);
  1431→      console.error(`[ScrapingBee attempt ${attempt}] Error:`, lastError);
  1432→      await new Promise(r => setTimeout(r, 1000));
  1433→    }
  1434→  }
  1435→
  1436→  console.error("[ScrapingBee] All attempts failed. Last error:", lastError);
  1437→  return null;
  1438→}

  1440→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1441→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1442→  // 1. OG meta tags (work even on anti-bot page)
  1443→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1444→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1445→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1446→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1447→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1448→
  1449→  // Decode HTML entities in title
  1450→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1451→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1452→
  1453→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1454→  let price: number | null = null;
  1455→  let currency = "USD";
  1456→  let originalPrice: number | null = null;
  1457→  let priceSource = "";
  1458→
  1459→  // ─────────────────────────────────────────────────────────────
  1460→  // 2-preferred. URL-embedded price hint.
  1461→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1462→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1463→  // → 1.23 USD. This is the most reliable signal because it comes
  1464→  // straight from the share URL the user copied.
  1465→  // ─────────────────────────────────────────────────────────────
  1466→  try {
  1467→    const parsedUrl = new URL(originalUrl);
  1468→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1469→    if (hint) {
  1470→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1471→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1472→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1473→      // Strip non-digits
  1474→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1475→      if (cents > 0 && cents < 10000000) {
  1476→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1477→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1478→        const usd = cents / 100;
  1479→        if (usd >= 0.01 && usd < 100000) {
  1480→          price = usd;
  1481→          currency = "USD";
  1482→          priceSource = "url-hint";
  1483→        }
  1484→      }
  1485→    }
  1486→  } catch { /* not a parseable URL */ }
  1487→
  1488→  // 2a. JSON-LD structured data
  1489→  if (!price) {
  1490→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1491→    for (const match of jsonLdMatches) {
  1492→      try {
  1493→        const data = JSON.parse(match[1]);
  1494→        const schemas = Array.isArray(data) ? data : [data];
  1495→        for (const schema of schemas) {
  1496→          if (schema["@type"] === "Product" && schema.offers) {
  1497→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1498→            for (const offer of offers) {
  1499→              if (offer.price !== undefined) {
  1500→                price = parseFloat(offer.price);
  1501→                currency = offer.priceCurrency || "USD";
  1502→                priceSource = "json-ld";
  1503→                break;
  1504→              }
  1505→            }
  1506→          }
  1507→        }
  1508→      } catch { /* skip */ }
  1509→      if (price) break;
  1510→    }
  1511→  }
  1512→
  1513→  // 2b. OG product:price:amount meta
  1514→  if (!price) {
  1515→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1516→    if (ogPrice) {
  1517→      const p = parseFloat(ogPrice[1]);
  1518→      if (p > 0) {
  1519→        price = p;
  1520→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1521→        currency = ogCur?.[1] || "USD";
  1522→        priceSource = "og-meta";
  1523→      }
  1524→    }
  1525→  }
  1526→
  1527→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1528→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1529→  //     one per "similar product", one per cross-sell, etc.). The previous
  1530→  //     implementation picked the FIRST match, which often returned a
  1531→  //     cross-sell product's price (e.g. $30) instead of the actual
  1532→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1533→  //     pick the LOWEST plausible one, since the actual product the user
  1534→  //     is viewing is almost always the cheapest among the rendered
  1535→  //     priceInfo blocks.
  1536→  if (!price) {
  1537→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1538→    const candidates: Candidate[] = [];
  1539→
  1540→    // priceInfo with explicit priceStr (most reliable)
  1541→    const priceInfosWithStr = [...html.matchAll(
  1542→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1543→    )];
  1544→    for (const pi of priceInfosWithStr) {
  1545→      const cents = parseInt(pi[1]);
  1546→      const cur = pi[2];
  1547→      const priceStr = pi[3];
  1548→      // Skip i18n placeholder strings
  1549→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1550→      if (cents <= 0 || cents >= 100000000) continue;
  1551→      const usd = cents / 100;
  1552→      if (usd < 0.01 || usd > 100000) continue;
  1553→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1554→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1555→      candidates.push({ usd, currency: cur, marketUsd });
  1556→    }
  1557→
  1558→    // priceInfo without priceStr (broader fallback)
  1559→    if (candidates.length === 0) {
  1560→      const priceInfosAll = [...html.matchAll(
  1561→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1562→      )];
  1563→      for (const pi of priceInfosAll) {
  1564→        const cents = parseInt(pi[1]);
  1565→        const cur = pi[2];
  1566→        if (cents <= 50 || cents >= 100000000) continue;
  1567→        const usd = cents / 100;
  1568→        if (usd < 0.50 || usd > 100000) continue;
  1569→        candidates.push({ usd, currency: cur });
  1570→      }
  1571→    }
  1572→
  1573→    if (candidates.length > 0) {
  1574→      // Pick the LOWEST price — the user is always viewing the cheapest
  1575→      // variant of the actual product, not the cross-sells.
  1576→      candidates.sort((a, b) => a.usd - b.usd);
  1577→      const best = candidates[0];
  1578→      price = best.usd;
  1579→      currency = best.currency;
  1580→      if (best.marketUsd && best.marketUsd > best.usd) {
  1581→        originalPrice = best.marketUsd;
  1582→      }
  1583→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1584→    }
  1585→  }
  1586→
  1587→  // 2d. Embedded JSON price fields (broad regex sweep)
  1588→  if (!price) {
  1589→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1590→    const found: { value: number; field: string }[] = [];
  1591→    for (const f of fields) {
  1592→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1593→      for (const m of html.matchAll(re)) {
  1594→        const v = parseFloat(m[1]);
  1595→        if (v > 0 && v < 100000) {
  1596→          found.push({ value: v, field: f });
  1597→        }
  1598→      }
  1599→    }
  1600→    if (found.length > 0) {
  1601→      // Sort and pick lowest (likely sale price)
  1602→      found.sort((a, b) => a.value - b.value);
  1603→      const best = found[0];
  1604→      // Heuristic: if value > 100, probably in cents → divide by 100
  1605→      price = best.value > 100 ? best.value / 100 : best.value;
  1606→      priceSource = `embedded-${best.field}`;
  1607→    }
  1608→  }
  1609→
  1610→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1611→  if (!price) {
  1612→    const text = html.replace(/<[^>]*>/g, " ");
  1613→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1614→    const prices: { value: number; currency: string }[] = [];
  1615→    for (const m of matches) {
  1616→      const raw = m[1].replace(/,/g, "");
  1617→      const v = parseFloat(raw);
  1618→      if (v > 1.5 && v < 5000) {
  1619→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1620→        prices.push({ value: v, currency: cur });
  1621→      }
  1622→    }
  1623→    if (prices.length > 0) {
  1624→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1625→      const sorted = prices.sort((a, b) => a.value - b.value);
  1626→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1627→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1628→      if (filtered.length > 0) {
  1629→        price = filtered[0].value;
  1630→        currency = filtered[0].currency;
  1631→        priceSource = "text";
  1632→      }
  1633→    }
  1634→  }
  1635→
  1636→  return {
  1637→    price,
  1638→    currency,
  1639→    productName,
  1640→    productDescription,
  1641→    canonicalUrl: canonical,
  1642→    originalPrice,
  1643→    image: ogImage,
  1644→    source: priceSource || (productName ? "og-only" : "none"),
  1645→  };
  1646→}

  1648→function decodeHtmlEntities(s: string): string {
  1649→  return s
  1650→    .replace(/&amp;/g, "&")
  1651→    .replace(/&lt;/g, "<")
  1652→    .replace(/&gt;/g, ">")
  1653→    .replace(/&quot;/g, '"')
  1654→    .replace(/&#39;/g, "'")
  1655→    .replace(/&nbsp;/g, " ")
  1656→    .replace(/&#(\d+);/g, (_, n) => String.fromCharCode(parseInt(n)));
  1657→}

  1677→/* ───────────────────────────────────────────────────────────────────
  1678→ * Strategy -2: ZAI LLM Direct Price Query (MOST RELIABLE for share URLs)
  1679→ *
  1680→ * Temu aggressively blocks all server-side scraping with anti-bot
  1681→ * challenges. Even the page_reader gets the login page instead of
  1682→ * the product page. This strategy uses the ZAI LLM with web search
  1683→ * to directly find the product price.
  1684→ *
  1685→ * How it works:
  1686→ * 1. Search the web for the product (by goods_id or Item ID)
  1687→ * 2. Extract prices from search result snippets (in various currencies)
  1688→ * 3. Use the LLM to interpret and convert prices to USD
  1689→ *
  1690→ * This is the most reliable strategy because:
  1691→ * - It doesn't depend on scraping (which Temu blocks)
  1692→ * - Search engines have already indexed Temu pages with prices
  1693→ * - The LLM can handle different currencies and formats
  1694→ * ─────────────────────────────────────────────────────────────────── */
  1695→async function fetchPriceWithLLMDirect(
  1696→  goodsId: string | null,
  1697→  itemId: string | null,
  1698→  productNameHint: string | null,
  1699→  userLocale: string | null = null,
  1700→): Promise<TemuProductData | null> {
  1701→  try {
  1702→    const zai = await createZAI();
  1703→
  1704→    // Step 1: Web search to find the product with prices
  1705→    // Try multiple search queries to maximize the chance of finding a price
  1706→    const searchQueries: string[] = [];
  1707→
  1708→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
  1709→      // Item ID search
  1710→      searchQueries.push(`temu "${itemId}" price`);
  1711→      searchQueries.push(`site:temu.com ${itemId}`);
  1712→    }
  1713→    if (goodsId && /^\d{10,}$/.test(goodsId)) {
  1714→      // Goods ID search - include "OFF" to find pages with discount prices
  1715→      searchQueries.push(`site:temu.com ${goodsId}`);
  1716→      // Also search with product name hint if available
  1717→      if (productNameHint) {
  1718→        const shortName = productNameHint.replace(/\s+/g, " ").trim().slice(0, 60);
  1719→        searchQueries.push(`temu ${shortName} price`);
  1720→      }
  1721→    }
  1722→
  1723→    if (searchQueries.length === 0) return null;
  1724→
  1725→    // Collect search results from multiple queries
  1726→    const allResults: { name: string; url: string; snippet: string }[] = [];
  1727→    const seenUrls = new Set<string>();
  1728→
  1729→    for (const query of searchQueries) {
  1730→      try {
  1731→        console.log(`[LLM-Direct] Searching: ${query}`);
  1732→        const results = await (zai as any).invokeFunction("web_search", {
  1733→          query,
  1734→          num: 8,
  1735→        });
  1736→
  1737→        if (Array.isArray(results)) {
  1738→          for (const r of results) {
  1739→            if (r.url && !seenUrls.has(r.url)) {
  1740→              seenUrls.add(r.url);
  1741→              allResults.push({
  1742→                name: r.name || "",
  1743→                url: r.url,
  1744→                snippet: r.snippet || "",
  1745→              });
  1746→            }
  1747→          }
  1748→        }
  1749→      } catch (err) {
  1750→        console.log(`[LLM-Direct] Search error:`, String(err).slice(0, 80));
  1751→      }
  1752→    }
  1753→
  1754→    if (allResults.length === 0) {
  1755→      console.log("[LLM-Direct] No search results found");
  1756→      return null;
  1757→    }
  1758→
  1759→    console.log(`[LLM-Direct] Collected ${allResults.length} search results`);
  1760→
  1761→    // Step 2: Try to extract prices directly from snippets first (fast, no LLM cost)
  1762→    // But ONLY if the price comes from a locale that matches the user's locale,
  1763→    // or if we don't have a locale hint (in which case USD prices are preferred).
  1764→    const snippetPrices = extractPricesFromSnippets(allResults, goodsId);
  1765→    if (snippetPrices.length > 0) {
  1766→      // If we know the user's locale, strongly prefer prices from that locale
  1767→      // because Temu shows different prices in different regions.
  1768→      snippetPrices.sort((a, b) => {
  1769→        // If user locale is known, strongly prefer matching locale prices
  1770→        if (userLocale) {
  1771→          const aLocale = allResults.find(r => {
  1772→            const m = r.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
  1773→            return m && m[1].toLowerCase() === userLocale.toLowerCase();
  1774→          });
  1775→          // Prefer results from the user's locale region
  1776→          const userRegionPrefix = userLocale.split("-")[0].toLowerCase();
  1777→          const aUrlLocale = allResults.find(r => r.url.includes(`/${userLocale}/`) || r.url.includes(`/${userRegionPrefix}/`));
  1778→          
  1779→          // Strong preference for matching locale prices
  1780→          if (aUrlLocale) {
  1781→            const aIsUserLocale = a.currency === getCurrencyForLocale(userLocale);
  1782→            const bIsUserLocale = b.currency === getCurrencyForLocale(userLocale);
  1783→            if (aIsUserLocale && !bIsUserLocale) return -1;
  1784→            if (bIsUserLocale && !aIsUserLocale) return 1;
  1785→          }
  1786→        }
  1787→        // Prefer USD
  1788→        if (a.currency === "USD" && b.currency !== "USD") return -1;
  1789→        if (b.currency === "USD" && a.currency !== "USD") return 1;
  1790→        // Prefer results from temu.com
  1791→        if (a.fromTemu && !b.fromTemu) return -1;
  1792→        if (b.fromTemu && !a.fromTemu) return 1;
  1793→        // Prefer lower prices (likely sale price, not original)
  1794→        return a.usd - b.usd;
  1795→      });
  1796→
  1797→      const best = snippetPrices[0];
  1798→      // If we have a user locale and the best price is from a different locale,
  1799→      // skip snippet extraction and try the LLM instead — it might find a better price
  1800→      const userCurrency = userLocale ? getCurrencyForLocale(userLocale) : null;
  1801→      const shouldSkipSnippet = userCurrency && best.currency !== userCurrency && snippetPrices.every(s => s.currency !== userCurrency);
  1802→      
  1803→      if (shouldSkipSnippet) {
  1804→        console.log(`[LLM-Direct] Best snippet price is ${best.originalAmount} ${best.currency} ($${best.usd}) but user locale is ${userLocale} (${userCurrency}) — trying LLM for better price`);
  1805→      } else if (best.usd > 0.5 && best.usd < 500) {
  1806→        if (!isSuspiciousPrice(best.usd, `llm-direct-snippet(${best.currency})`)) {
  1807→          console.log(`[LLM-Direct] ✓ Found price from snippet: ${best.originalAmount} ${best.currency} = $${best.usd} USD`);
  1808→          return {
  1809→            price: best.usd,
  1810→            currency: "USD",
  1811→            productName: best.productName || productNameHint,
  1812→            productDescription: null,
  1813→            canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
  1814→            originalPrice: best.originalPriceUSD || null,
  1815→            image: null,
  1816→            source: `llm-direct-snippet(${best.currency})`,
  1817→            antiBotDetected: false,
  1818→          };
  1819→        }
  1820→      }
  1821→    }
  1822→
  1823→    // Step 3: Use LLM to extract price from search results
  1824→    const searchContext = allResults
  1825→      .slice(0, 8)
  1826→      .map((r, i) => `${i + 1}. ${r.name}\n   URL: ${r.url}\n   Snippet: ${r.snippet}`)
  1827→      .join("\n\n");
  1828→
  1829→    console.log(`[LLM-Direct] Using LLM on ${Math.min(allResults.length, 8)} search results...`);
  1830→
  1831→    // Build locale-specific instruction for the LLM
  1832→    const localeInstruction = userLocale
  1833→      ? `\n10. IMPORTANT: The user is viewing this product from the "${userLocale}" locale on Temu. The price may vary by region. Try to find the price that would be shown in this locale, or estimate it based on prices from other locales. For ${userLocale}, the currency is ${getCurrencyForLocale(userLocale)}.`
  1834→      : "";
  1835→
  1836→    const completion = await (zai as any).createChatCompletion({
  1837→      messages: [
  1838→        {
  1839→          role: "system",
  1840→          content:
  1841→            "You are a price extraction assistant for Temu products. " +
  1842→            "You will be given web search results for a Temu product. " +
  1843→            "Extract the SALE PRICE of the product from the search results.\n\n" +
  1844→            "IMPORTANT RULES:\n" +
  1845→            "1. The search results show the same product on different Temu locale pages (Oman, Bahrain, Mauritius, US, etc.)\n" +
  1846→            "2. Each locale shows the price in its local currency. Convert to USD using these rates:\n" +
  1847→            "   - OMR → USD: ×2.60 (e.g., OMR 3.56 = $9.26)\n" +
  1848→            "   - BHD → USD: ×2.65 (e.g., BHD 1.23 = $3.26)\n" +

   274→   274→/* ───────────────────────────────────────────────────────────────────
   275→   275→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   276→   276→ *
   277→   277→ * Temu share URLs include everything we need:
   278→   278→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   279→   279→ *   - top_gallery_url    = URL-encoded product image URL
   280→   280→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   281→   281→ *
   282→   282→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   283→   283→ * ─────────────────────────────────────────────────────────────────── */
   284→   284→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   285→   285→  let parsed: URL;
   286→   286→  try {
   287→   287→    parsed = new URL(originalUrl);
   288→   288→  } catch {
   289→   289→    return null;
   290→   290→  }
   291→   291→
   292→   292→  let price: number | null = null;
   293→   293→  let currency = "USD";
   294→   294→  let priceSource = "";
   295→   295→
   296→   296→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   297→   297→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   298→   298→  if (hint) {
   299→   299→    try {
   300→   300→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   301→   301→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   302→   302→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   303→   303→      if (cents > 0 && cents < 10000000) {
   304→   304→        const usd = cents / 100;
   305→   305→        if (usd >= 0.01 && usd < 100000) {
   306→   306→          price = usd;
   307→   307→          currency = "USD";
   308→   308→          priceSource = "url-hint";
   309→   309→        }
   310→   310→      }
   311→   311→    } catch { /* not valid base64 */ }
   312→   312→  }
   313→   313→
   314→   314→  // Product image from top_gallery_url
   315→   315→  let image: string | null = null;
   316→   316→  const topGallery = parsed.searchParams.get("top_gallery_url");
   317→   317→  if (topGallery) {
   318→   318→    try {
   319→   319→      // Validate it's a real URL
   320→   320→      const imgUrl = new URL(topGallery);
   321→   321→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   322→   322→        image = topGallery;
   323→   323→      }
   324→   324→    } catch { /* not a valid URL */ }
   325→   325→  }
   326→   326→
   327→   327→  // Product name from URL slug
   328→   328→  let productName: string | null = null;
   329→   329→  const segments = parsed.pathname.split("/").filter(Boolean);
   330→   330→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   331→   331→  const nameFromSlug = slug
   332→   332→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   333→   333→    .replace(/\.html?$/i, "")
   334→   334→    .replace(/-/g, " ")
   335→   335→    .replace(/\s+/g, " ")
   336→   336→    .trim();
   337→   337→  if (nameFromSlug && nameFromSlug.length > 3) {
   338→   338→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   339→   339→  }
   340→   340→
   341→   341→  // If we got at least the price (the most important piece), return success
   342→   342→  if (price) {
   343→   343→    return {
   344→   344→      price,
   345→   345→      currency,
   346→   346→      productName,
   347→   347→      productDescription: null,
   348→   348→      canonicalUrl: null,
   349→   349→      originalPrice: null,
   350→   350→      image,
   351→   351→      source: priceSource,
   352→   352→      antiBotDetected: false,
   353→   353→    };
   354→   354→  }
   355→   355→
   356→   356→  // No price from URL — return null so caller tries HTML-based strategies
   357→   357→  return null;
   358→   358→}

   713→   713→/**

   719→   719→  results: any[],
   720→   720→  goodsId: string | null,
   721→   721→): TemuProductData | null {

   724→   724→    if (!result.url?.includes("temu.com")) continue;

   747→   747→      const locale = localeMatch[1].toLowerCase();

   750→   750→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   751→   751→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman

   759→   759→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait

   763→   763→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka

   773→   773→    }
   774→   774→
   775→   775→    // Try to find a price in the snippet
   776→   776→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   777→   777→    // Common patterns:
   778→   778→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   779→   779→    //   "$6.99"                  → USD
   780→   780→    //   "OMR 1.200"             → Omani Rial

   782→   782→    //   "€5.99"                  → EUR
   783→   783→    //   "£4.99"                  → GBP
   784→   784→

   786→   786→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   787→   787→    const europeanCommaLocales = new Set([
   788→   788→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",

   792→   792→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   793→   793→

   795→   795→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   796→   796→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   797→   797→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   798→   798→      if (isEuropean) {
   799→   799→        // European: remove dots (thousands), replace comma with dot (decimal)
   800→   800→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   801→   801→      } else {
   802→   802→        // English: remove commas (thousands)
   803→   803→        return parseFloat(priceStr.replace(/,/g, ""));
   804→   804→      }
   805→   805→    };
   806→   806→

  1312→  1312→  if (!apiKey) return null;
  1313→  1313→

  1340→  1340→        signal: controller.signal,

  1342→  1342→      clearTimeout(timeout);

  1381→  1381→
  1382→  1382→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1383→  1383→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1384→  1384→  // 1. OG meta tags (work even on anti-bot page)
  1385→  1385→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1386→  1386→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1387→  1387→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1388→  1388→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1389→  1389→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1390→  1390→
  1391→  1391→  // Decode HTML entities in title
  1392→  1392→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1393→  1393→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1394→  1394→
  1395→  1395→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1396→  1396→  let price: number | null = null;
  1397→  1397→  let currency = "USD";
  1398→  1398→  let originalPrice: number | null = null;
  1399→  1399→  let priceSource = "";
  1400→  1400→
  1401→  1401→  // ─────────────────────────────────────────────────────────────
  1402→  1402→  // 2-preferred. URL-embedded price hint.
  1403→  1403→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1404→  1404→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1405→  1405→  // → 1.23 USD. This is the most reliable signal because it comes
  1406→  1406→  // straight from the share URL the user copied.
  1407→  1407→  // ─────────────────────────────────────────────────────────────
  1408→  1408→  try {
  1409→  1409→    const parsedUrl = new URL(originalUrl);
  1410→  1410→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1411→  1411→    if (hint) {
  1412→  1412→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1413→  1413→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1414→  1414→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1415→  1415→      // Strip non-digits
  1416→  1416→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1417→  1417→      if (cents > 0 && cents < 10000000) {
  1418→  1418→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1419→  1419→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1420→  1420→        const usd = cents / 100;
  1421→  1421→        if (usd >= 0.01 && usd < 100000) {
  1422→  1422→          price = usd;
  1423→  1423→          currency = "USD";
  1424→  1424→          priceSource = "url-hint";
  1425→  1425→        }
  1426→  1426→      }
  1427→  1427→    }
  1428→  1428→  } catch { /* not a parseable URL */ }
  1429→  1429→
  1430→  1430→  // 2a. JSON-LD structured data
  1431→  1431→  if (!price) {
  1432→  1432→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1433→  1433→    for (const match of jsonLdMatches) {
  1434→  1434→      try {
  1435→  1435→        const data = JSON.parse(match[1]);
  1436→  1436→        const schemas = Array.isArray(data) ? data : [data];
  1437→  1437→        for (const schema of schemas) {
  1438→  1438→          if (schema["@type"] === "Product" && schema.offers) {
  1439→  1439→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1440→  1440→            for (const offer of offers) {
  1441→  1441→              if (offer.price !== undefined) {
  1442→  1442→                price = parseFloat(offer.price);
  1443→  1443→                currency = offer.priceCurrency || "USD";
  1444→  1444→                priceSource = "json-ld";
  1445→  1445→                break;
  1446→  1446→              }
  1447→  1447→            }
  1448→  1448→          }
  1449→  1449→        }
  1450→  1450→      } catch { /* skip */ }
  1451→  1451→      if (price) break;
  1452→  1452→    }
  1453→  1453→  }
  1454→  1454→
  1455→  1455→  // 2b. OG product:price:amount meta
  1456→  1456→  if (!price) {
  1457→  1457→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1458→  1458→    if (ogPrice) {
  1459→  1459→      const p = parseFloat(ogPrice[1]);
  1460→  1460→      if (p > 0) {
  1461→  1461→        price = p;
  1462→  1462→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1463→  1463→        currency = ogCur?.[1] || "USD";
  1464→  1464→        priceSource = "og-meta";
  1465→  1465→      }
  1466→  1466→    }
  1467→  1467→  }
  1468→  1468→
  1469→  1469→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1470→  1470→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1471→  1471→  //     one per "similar product", one per cross-sell, etc.). The previous
  1472→  1472→  //     implementation picked the FIRST match, which often returned a
  1473→  1473→  //     cross-sell product's price (e.g. $30) instead of the actual
  1474→  1474→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1475→  1475→  //     pick the LOWEST plausible one, since the actual product the user
  1476→  1476→  //     is viewing is almost always the cheapest among the rendered
  1477→  1477→  //     priceInfo blocks.
  1478→  1478→  if (!price) {
  1479→  1479→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1480→  1480→    const candidates: Candidate[] = [];
  1481→  1481→
  1482→  1482→    // priceInfo with explicit priceStr (most reliable)
  1483→  1483→    const priceInfosWithStr = [...html.matchAll(
  1484→  1484→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1485→  1485→    )];
  1486→  1486→    for (const pi of priceInfosWithStr) {
  1487→  1487→      const cents = parseInt(pi[1]);
  1488→  1488→      const cur = pi[2];
  1489→  1489→      const priceStr = pi[3];
  1490→  1490→      // Skip i18n placeholder strings
  1491→  1491→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1492→  1492→      if (cents <= 0 || cents >= 100000000) continue;
  1493→  1493→      const usd = cents / 100;
  1494→  1494→      if (usd < 0.01 || usd > 100000) continue;
  1495→  1495→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1496→  1496→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1497→  1497→      candidates.push({ usd, currency: cur, marketUsd });
  1498→  1498→    }
  1499→  1499→
  1500→  1500→    // priceInfo without priceStr (broader fallback)
  1501→  1501→    if (candidates.length === 0) {
  1502→  1502→      const priceInfosAll = [...html.matchAll(
  1503→  1503→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1504→  1504→      )];
  1505→  1505→      for (const pi of priceInfosAll) {
  1506→  1506→        const cents = parseInt(pi[1]);
  1507→  1507→        const cur = pi[2];
  1508→  1508→        if (cents <= 50 || cents >= 100000000) continue;
  1509→  1509→        const usd = cents / 100;
  1510→  1510→        if (usd < 0.50 || usd > 100000) continue;
  1511→  1511→        candidates.push({ usd, currency: cur });
  1512→  1512→      }
  1513→  1513→    }
  1514→  1514→
  1515→  1515→    if (candidates.length > 0) {
  1516→  1516→      // Pick the LOWEST price — the user is always viewing the cheapest
  1517→  1517→      // variant of the actual product, not the cross-sells.
  1518→  1518→      candidates.sort((a, b) => a.usd - b.usd);
  1519→  1519→      const best = candidates[0];
  1520→  1520→      price = best.usd;
  1521→  1521→      currency = best.currency;
  1522→  1522→      if (best.marketUsd && best.marketUsd > best.usd) {
  1523→  1523→        originalPrice = best.marketUsd;
  1524→  1524→      }
  1525→  1525→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1526→  1526→    }
  1527→  1527→  }
  1528→  1528→
  1529→  1529→  // 2d. Embedded JSON price fields (broad regex sweep)
  1530→  1530→  if (!price) {
  1531→  1531→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1532→  1532→    const found: { value: number; field: string }[] = [];
  1533→  1533→    for (const f of fields) {
  1534→  1534→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1535→  1535→      for (const m of html.matchAll(re)) {
  1536→  1536→        const v = parseFloat(m[1]);
  1537→  1537→        if (v > 0 && v < 100000) {
  1538→  1538→          found.push({ value: v, field: f });
  1539→  1539→        }
  1540→  1540→      }
  1541→  1541→    }
  1542→  1542→    if (found.length > 0) {
  1543→  1543→      // Sort and pick lowest (likely sale price)
  1544→  1544→      found.sort((a, b) => a.value - b.value);
  1545→  1545→      const best = found[0];
  1546→  1546→      // Heuristic: if value > 100, probably in cents → divide by 100
  1547→  1547→      price = best.value > 100 ? best.value / 100 : best.value;
  1548→  1548→      priceSource = `embedded-${best.field}`;
  1549→  1549→    }
  1550→  1550→  }
  1551→  1551→
  1552→  1552→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1553→  1553→  if (!price) {
  1554→  1554→    const text = html.replace(/<[^>]*>/g, " ");
  1555→  1555→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1556→  1556→    const prices: { value: number; currency: string }[] = [];
  1557→  1557→    for (const m of matches) {
  1558→  1558→      const raw = m[1].replace(/,/g, "");
  1559→  1559→      const v = parseFloat(raw);
  1560→  1560→      if (v > 1.5 && v < 5000) {
  1561→  1561→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1562→  1562→        prices.push({ value: v, currency: cur });
  1563→  1563→      }
  1564→  1564→    }
  1565→  1565→    if (prices.length > 0) {

  1575→  1575→    }
  1576→  1576→  }
  1577→  1577→
  1578→  1578→  return {
  1579→  1579→    price,
  1580→  1580→    currency,
  1581→  1581→    productName,

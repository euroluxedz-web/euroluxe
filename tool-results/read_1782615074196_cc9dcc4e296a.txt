    20→/* ───────────────────────────────────────────────────────────────────
    21→ * Strategy 0-C: Page Reader + LLM — read the rendered product page.
    22→ *
    23→ * Uses the ZAI page_reader to read the fully rendered Temu product page,
    24→ * then extracts the price using LLM. This is more reliable than web
    25→ * search because it reads the actual page content.
    26→ *
    27→ * Key insight: When reading the share URL via page_reader, the browser
    28→ * follows the redirect and renders the page. The product data (including
    29→ * price) may be in the window.rawData JavaScript object.

    31→async function fetchPriceWithPageReader(
    32→  goodsId: string | null,
    33→  itemId: string | null,
    34→  shareUrl: string | null,
    35→  resolvedShareUrl?: string | null,
    36→  shareLocale?: string | null,
    37→): Promise<TemuProductData | null> {
    38→  try {
    39→    const zai = await ZAI.create();
    40→
    41→    // Determine the best URL(s) to read — try multiple URLs in order
    42→    const readUrls: { url: string; label: string }[] = [];
    43→    if (shareUrl) {
    44→      // Read the share URL directly — page_reader follows redirects
    45→      readUrls.push({ url: shareUrl, label: "share-url" });
    46→    }
    47→    // Try the RESOLVED share URL (with all original params like _bg_fs=1, goods_id, etc.)
    48→    // This is often more accessible than the share URL because it has the full product URL
    49→    if (resolvedShareUrl && resolvedShareUrl !== shareUrl) {
    50→      readUrls.push({ url: resolvedShareUrl, label: "resolved-share-url" });
    51→    }
    52→    if (goodsId && /^\d{10,}$/.test(goodsId)) {
    53→      // Try the US product page directly — more likely to have USD prices
    54→      readUrls.push({ url: `https://www.temu.com/-g-${goodsId}.html?_x_sessn=us&currency=USD`, label: "us-product-page" });
    55→      // Also try the goods.html page with US session
    56→      readUrls.push({ url: `https://www.temu.com/goods.html?goods_id=${goodsId}&_x_sessn=us&currency=USD`, label: "goods-api-us" });

    58→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {

    62→      readUrls.push({ url: `https://www.temu.com/search?q=${encodeURIComponent(itemId)}&_x_sessn=us&currency=USD`, label: "item-id-search" });
    63→    }
    64→    // Fallback: if no URLs were added, return null
    65→    if (readUrls.length === 0) return null;
    66→
    67→    // Try each URL until we get a price
    68→    for (const { url: readUrl, label } of readUrls) {
    69→      console.log(`[PageReader] Reading (${label}): ${readUrl}`);
    70→
    71→      try {
    72→        const pageResult = await (zai as any).invokeFunction("page_reader", {
    73→          url: readUrl,
    74→        });
    75→
    76→        const data = typeof pageResult === "string" ? JSON.parse(pageResult) : pageResult;
    77→        const content = data?.data?.content || data?.data?.text || data?.data?.html || data?.content || data?.text || data?.html;
    78→

    81→          continue;
    82→        }
    83→
    84→        console.log(`[PageReader] Got content from ${label}: ${content.length} chars`);
    85→
    86→        // Strategy A: Search for window.rawData in the HTML and extract price from it
    87→        const rawDataMatch = content.match(/window\.rawData\s*=\s*(\{[\s\S]*?\})\s*(?:;|window\.)/);
    88→        if (rawDataMatch) {
    89→          try {
    90→            const rawDataStr = rawDataMatch[1];
    91→
    92→            // Search for price patterns near the goods_id
    93→            if (goodsId && rawDataStr.includes(goodsId)) {
    94→              const gidIdx = rawDataStr.indexOf(goodsId);
    95→              const searchWindow = rawDataStr.slice(Math.max(0, gidIdx - 2000), Math.min(rawDataStr.length, gidIdx + 10000));
    96→
    97→              const priceFields: Record<string, number> = {};
    98→              const priceMatches = [...searchWindow.matchAll(/"(minPrice|salePrice|price|marketPrice|origPrice|appPrice|priceNum|displayPrice)"\s*:\s*"?(\d+\.?\d*)"?/gi)];
    99→              for (const m of priceMatches) {
   100→                priceFields[m[1]] = parseFloat(m[2]);
   101→              }
   102→
   103→              if (Object.keys(priceFields).length > 0) {
   104→                console.log(`[PageReader] Found price fields in rawData near goods_id:`, priceFields);
   105→
   106→                const salePrice = priceFields.minPrice || priceFields.salePrice || priceFields.price || priceFields.appPrice || priceFields.displayPrice;
   107→                const originalPrice = priceFields.marketPrice || priceFields.origPrice || null;
   108→
   109→                if (salePrice && salePrice > 0 && salePrice < 100000) {
   110→                  const actualPrice = salePrice > 100 ? salePrice / 100 : salePrice;
   111→                  const actualOrigPrice = originalPrice ? (originalPrice > 100 ? originalPrice / 100 : originalPrice) : null;
   112→
   113→                  const currencyMatch = rawDataStr.match(/"currency"\s*:\s*"([^"]+)"/);
   114→                  const rawCurrency = currencyMatch ? currencyMatch[1] : "USD";
   115→

   117→                  let priceUSD = actualPrice;
   118→                  let currency = "USD";

   121→                  } else {
   122→                    currency = rawCurrency;

   140→            }

   145→

   147→        // Sometimes the rawData structure is different and goods_id might not appear as text,

   156→              const cur = pi[2];

   158→              const value = cents / 100;
   159→              if (cur === "USD") {
   160→                candidates.push({ usd: value, currency: cur });
   161→              } else if (CURRENCY_TO_USD[cur]) {
   162→                candidates.push({ usd: Math.round(value * CURRENCY_TO_USD[cur] * 100) / 100, currency: cur });
   163→              }
   164→            }
   165→            if (candidates.length > 0) {
   166→              candidates.sort((a, b) => a.usd - b.usd);
   167→              const best = candidates[0];
   168→              if (best.usd >= 0.01 && best.usd < 100000) {
   169→                console.log(`[PageReader] ✓ Found priceInfo price (${label}): ${best.usd} USD (from ${best.currency})`);
   170→                return {
   171→                  price: best.usd,
   172→                  currency: "USD",
   173→                  productName: null,
   174→                  productDescription: null,
   175→                  canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   176→                  originalPrice: null,
   177→                  image: null,
   178→                  source: `page-reader-priceInfo(${label})`,
   179→                  antiBotDetected: false,
   180→                };
   181→              }
   182→            }
   183→          }
   184→        }
   185→
   186→        // Strategy B: Use LLM to extract price from the page content
   187→        console.log(`[PageReader] Using LLM to extract price from ${label}...`);
   188→
   189→        const contentForLLM = content.slice(0, Math.min(content.length, 60000));
   190→

   197→            `The REAL product price is typically shown near the top of the page, often with a "was" / "original" price struck through.`
   198→          : "";
   199→
   200→        const completion = await (zai as any).createChatCompletion({
   201→          messages: [
   202→            {
   203→              role: "system",
   204→              content:

   211→                "3. If you see 'window.rawData', look for price fields like minPrice, salePrice, or price near the goods_id.\n" +
   212→                "4. Look for priceInfo blocks with 'price' (in cents) and 'currency' fields.\n" +
   213→                "5. Prices in DZD (Algerian Dinar) are large numbers (e.g., 900-30000 DA). Prices in USD are small numbers (e.g., $3-$100).\n" +
   214→                "6. If the price is in DZD, convert it: USD = DZD / 300. If in EUR, USD = EUR * 1.08.\n" +
   215→                "7. Do NOT confuse 'delivery guarantee' prices, 'credit for delay' amounts, or '30% OFF' discount percentages with the product price.\n" +
   216→                "8. Do NOT confuse coupon amounts or shipping credits with the product price.\n" +
   217→                "9. ⚠️ ESPECIALLY ignore prices of exactly $30.00 or 9,000 DZD — these are Temu's 'delivery guarantee' amounts, NOT product prices.\n" +
   218→                "10. Return ONLY a JSON object: {\"price_usd\": <number_in_USD>, \"price_local\": \"<amount> <currency>\", \"product_name\": \"<name>\", \"confidence\": \"<high|medium|low>\"}\n" +
   219→                "11. If you cannot find a clear price for the main product, return {\"price_usd\": null, \"confidence\": \"low\"}\n" +

   275→      } catch (err) {
   276→        console.log(`[PageReader] Error reading ${label}:`, String(err).slice(0, 100));

   278→      }
   279→    }
   280→
   281→    return null;
   282→  } catch (err) {
   283→    console.error("[PageReader] Error:", String(err).slice(0, 200));

   288→/* ───────────────────────────────────────────────────────────────────
   289→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   290→ *
   291→ * Temu share URLs include everything we need:
   292→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   293→ *   - top_gallery_url    = URL-encoded product image URL
   294→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   295→ *
   296→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   297→ * ─────────────────────────────────────────────────────────────────── */
   298→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   299→  let parsed: URL;
   300→  try {
   301→    parsed = new URL(originalUrl);
   302→  } catch {
   303→    return null;
   304→  }
   305→
   306→  let price: number | null = null;
   307→  let currency = "USD";
   308→  let priceSource = "";
   309→
   310→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   311→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   312→  if (hint) {
   313→    try {
   314→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   315→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   316→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   317→      if (cents > 0 && cents < 10000000) {
   318→        const usd = cents / 100;
   319→        if (usd >= 0.01 && usd < 100000) {
   320→          price = usd;
   321→          currency = "USD";
   322→          priceSource = "url-hint";
   323→        }
   324→      }
   325→    } catch { /* not valid base64 */ }
   326→  }
   327→
   328→  // Product image from top_gallery_url
   329→  let image: string | null = null;
   330→  const topGallery = parsed.searchParams.get("top_gallery_url");
   331→  if (topGallery) {
   332→    try {
   333→      // Validate it's a real URL
   334→      const imgUrl = new URL(topGallery);
   335→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   336→        image = topGallery;
   337→      }
   338→    } catch { /* not a valid URL */ }
   339→  }
   340→
   341→  // Product name from URL slug
   342→  let productName: string | null = null;
   343→  const segments = parsed.pathname.split("/").filter(Boolean);
   344→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   345→  const nameFromSlug = slug
   346→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   347→    .replace(/\.html?$/i, "")
   348→    .replace(/-/g, " ")
   349→    .replace(/\s+/g, " ")
   350→    .trim();
   351→  if (nameFromSlug && nameFromSlug.length > 3) {
   352→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   353→  }
   354→
   355→  // If we got at least the price (the most important piece), return success
   356→  if (price) {
   357→    return {
   358→      price,
   359→      currency,
   360→      productName,
   361→      productDescription: null,
   362→      canonicalUrl: null,
   363→      originalPrice: null,
   364→      image,
   365→      source: priceSource,
   366→      antiBotDetected: false,
   367→    };
   368→  }
   369→
   370→  // No price from URL — return null so caller tries HTML-based strategies
   371→  return null;
   372→}

   430→async function fetchPriceWithWebSearch(
   431→  goodsId: string | null,
   432→  itemId: string | null,
   433→  originalUrl: string,
   434→): Promise<TemuProductData | null> {
   435→  try {
   436→    const zai = await ZAI.create();
   437→
   438→    // Build search query — try different strategies
   439→    let searchQuery = "";
   440→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   441→      // Item ID like TV10922608 — search broadly on Temu
   442→      searchQuery = `temu ${itemId}`;
   443→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   444→      // Numeric goods_id — search with the -g- pattern
   445→      searchQuery = `site:temu.com "g-${goodsId}"`;
   446→    } else {
   447→      // Fallback: search with the URL
   448→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   449→      searchQuery = `site:temu.com ${cleanUrl}`;
   450→    }
   451→
   452→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   453→
   454→    // Step 1: Web search to find the product
   455→    const searchResults = await (zai as any).invokeFunction("web_search", {
   456→      query: searchQuery,
   457→      num: 5,
   458→    });
   459→
   460→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   461→      console.log("[WebSearch] No results found, trying broader search...");
   462→
   463→      // Broader search without site: restriction
   464→      const broadQuery = itemId
   465→        ? `temu ${itemId} price`
   466→        : goodsId
   467→          ? `temu ${goodsId} price`
   468→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   469→
   470→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   471→      const broadResults = await (zai as any).invokeFunction("web_search", {
   472→        query: broadQuery,
   473→        num: 5,
   474→      });
   475→
   476→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   477→        console.log("[WebSearch] No results from broader search either");
   478→        return null;
   479→      }
   480→
   481→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   482→    }
   483→
   484→    // Step 2: Try to parse price from initial search results
   485→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   486→    if (initialPrice?.price && initialPrice.price > 0) {
   487→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   488→      return initialPrice;
   489→    }
   490→
   491→    // Step 3: If initial search found product names but no prices,
   492→    // try a more targeted search with the product name to find prices
   493→    const productNameFromResults = searchResults
   494→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   495→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   496→      .find((n: string) => n && n.length > 10);
   497→
   498→    if (productNameFromResults) {
   499→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   500→      try {
   501→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   502→        const nameResults = await (zai as any).invokeFunction("web_search", {
   503→          query: nameQuery,
   504→          num: 5,
   505→        });
   506→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   507→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   508→          if (namePrice?.price && namePrice.price > 0) {
   509→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   510→            return namePrice;
   511→          }
   512→          // Merge results and try LLM on combined results
   513→          const combinedResults = [...searchResults, ...nameResults];
   514→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   515→        }
   516→      } catch (err) {
   517→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   518→      }
   519→    }
   520→
   521→    // Step 4: Use LLM on whatever results we have
   522→    const llmResult = await extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   523→    if (llmResult?.price && llmResult.price > 0) {
   524→      return llmResult;
   525→    }
   526→
   527→    // Step 5: If LLM didn't find price, try reading Temu search result pages
   528→    // via AllOrigins proxy — sometimes the proxy can access the product page
   529→    // when direct access is blocked
   530→    const temuResults = searchResults
   531→      .filter((r: any) => r.url?.includes("temu.com") && r.url?.includes("-g-"))
   532→      .slice(0, 2);
   533→
   534→    for (const result of temuResults) {
   535→      try {
   536→        console.log(`[WebSearch] Trying AllOrigins for search result: ${result.url.slice(0, 80)}...`);
   537→        // Try both /raw and /get endpoints
   538→        const proxyUrls = [
   539→          `https://api.allorigins.win/raw?url=${encodeURIComponent(result.url)}`,
   540→          `https://api.allorigins.win/get?url=${encodeURIComponent(result.url)}`,
   541→        ];
   542→
   543→        let proxyHtml: string | null = null;
   544→
   545→        for (const pUrl of proxyUrls) {
   546→          try {
   547→            const controller = new AbortController();
   548→            const timeout = setTimeout(() => controller.abort(), 15000);
   549→            const proxyRes = await fetch(pUrl, { signal: controller.signal });
   550→            clearTimeout(timeout);
   551→            if (!proxyRes.ok) continue;
   552→
   553→            const isRaw = pUrl.includes("/raw?");
   554→            if (isRaw) {
   555→              const text = await proxyRes.text();
   556→              if (text && text.length > 5000 && text.includes("<")) {
   557→                proxyHtml = text;
   558→                break;
   559→              }
   560→            } else {
   561→              const data = await proxyRes.json();
   562→              const contents = typeof data === "string" ? data : data?.contents;
   563→              if (contents && typeof contents === "string" && contents.length > 5000) {
   564→                proxyHtml = contents;
   565→                break;
   566→              }
   567→            }
   568→          } catch { /* try next */ }
   569→        }
   570→
   571→        if (!proxyHtml || proxyHtml.length < 5000) continue;
   572→
   573→        // Check if we got the real product page (has OG title)
   574→        const hasOGTitle = /<meta[^>]*property=["']og:title["']/i.test(proxyHtml);
   575→        if (!hasOGTitle) continue;
   576→
   577→        // Extract price using extractProductInfo
   578→        const proxyResult = extractProductInfo(proxyHtml, result.url);
   579→        if (proxyResult.price && proxyResult.price > 0) {
   580→          console.log(`[WebSearch] ✓ Got price $${proxyResult.price} from AllOrigins search result`);
   581→          return proxyResult;
   582→        }
   583→
   584→        // Also try direct OG price extraction
   585→        const ogPriceMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   586→        const ogCurrencyMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   587→        if (ogPriceMatch) {
   588→          const priceVal = parseFloat(ogPriceMatch[1]);
   589→          const cur = ogCurrencyMatch?.[1] || "USD";
   590→          if (priceVal > 0 && priceVal < 100000) {
   591→            let priceUSD = priceVal;
   592→            if (cur !== "USD" && CURRENCY_TO_USD[cur]) {
   593→              priceUSD = Math.round(priceVal * CURRENCY_TO_USD[cur] * 100) / 100;
   594→            }
   595→            const ogTitle = proxyHtml.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1];
   596→            const ogImage = proxyHtml.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1];
   597→            const foundGoodsId = result.url.match(/-g-(\d{10,})/)?.[1] || goodsId;
   598→            console.log(`[WebSearch] ✓ Got OG price ${priceVal} ${cur} = $${priceUSD} from AllOrigins search result`);
   599→            return {
   600→              price: priceUSD,
   601→              currency: "USD",
   602→              productName: ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null,
   603→              productDescription: null,
   604→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   605→              originalPrice: null,
   606→              image: ogImage || null,
   607→              source: `websearch-allorigins-og(${cur})`,
   608→              antiBotDetected: false,
   609→            };
   610→          }
   611→        }
   612→      } catch (err) {
   613→        console.log(`[WebSearch] AllOrigins for search result failed:`, String(err).slice(0, 80));
   614→      }
   615→    }
   616→
   617→    return llmResult; // Return whatever the LLM found (may be null)
   618→  } catch (err) {
   619→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   620→    return null;
   621→  }
   622→}

   624→async function extractPriceFromSearchResults(

   626→  searchResults: any[],
   627→  goodsId: string | null,
   628→  itemId: string | null,

   630→  // Step 2: Try to parse prices directly from search snippets first (fast, no LLM cost)
   631→  const snippetPrice = parsePriceFromSnippets(searchResults, goodsId);
   632→  if (snippetPrice?.price && snippetPrice.price > 0) {
   633→    console.log(`[WebSearch] ✓ Found price from snippet: ${snippetPrice.price} USD (source: ${snippetPrice.source})`);
   634→    return snippetPrice;
   635→  }
   636→
   637→  // Step 3: If snippets don't have a clear price, use LLM to extract
   638→  console.log("[WebSearch] Snippets don't have clear price, using LLM...");

   641→    .slice(0, 5)
   642→    .map(
   643→      (r: any, i: number) =>

   656→            'If the price is in a non-USD currency, CONVERT it to USD using approximate exchange rates. ' +

   661→            'Do NOT confuse "delivery guarantee" prices, "credit for delay" amounts, or "30% OFF" discount percentages with the product price. ' +
   662→            'Do NOT use prices from unrelated products. Only extract the price that clearly corresponds to the product with the given goods_id or Item ID.',

   666→          content:
   667→            `Product goods_id: ${goodsId || "unknown"}\n` +
   668→            `Item ID: ${itemId || "unknown"}\n\n` +

   671→        },
   672→      ],
   673→    });

   675→    const aiResponse = completion.choices?.[0]?.message?.content || "";
   676→    console.log("[WebSearch] LLM response:", aiResponse.slice(0, 300));

   681→      console.log("[WebSearch] No JSON in LLM response");
   682→      return null;
   683→    }
   684→
   685→    const parsed = JSON.parse(jsonMatch[0]);
   686→    const priceUSD = typeof parsed.price_usd === "number"
   687→      ? parsed.price_usd
   688→      : parseFloat(String(parsed.price_usd));
   689→
   690→    if (priceUSD && priceUSD > 0 && priceUSD < 100000) {
   691→      console.log(`[WebSearch] ✓ LLM found price: $${priceUSD} (confidence: ${parsed.confidence})`);
   692→      return {
   693→        price: priceUSD,
   694→        currency: "USD",
   695→        productName: parsed.name || null,
   696→        productDescription: null,

   698→        originalPrice: null,
   699→        image: null, // Image will come from share URL params
   700→        source: `web-search-llm(${parsed.confidence})`,
   701→        antiBotDetected: false,
   702→      };
   703→    }
   704→
   705→    // LLM couldn't find price either — return what we can
   706→    if (parsed.name) {
   707→      return {
   708→        price: null,
   709→        currency: "USD",
   710→        productName: parsed.name,
   711→        productDescription: null,
   712→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   713→        originalPrice: null,
   714→        image: null,
   715→        source: "web-search-llm(no-price)",
   716→        antiBotDetected: false,
   717→      };
   718→    }
   719→
   720→    return null;
   721→  } catch (err) {
   722→    console.error("[WebSearch] LLM error:", String(err).slice(0, 150));
   723→    return null;
   724→  }

   727→/**
   728→ * Try to parse prices directly from search result snippets.
   729→ * This avoids the cost of an LLM call and is often sufficient
   730→ * because Google includes price info in Temu search snippets.
   731→ */
   732→function parsePriceFromSnippets(
   733→  results: any[],
   734→  goodsId: string | null,
   735→): TemuProductData | null {
   736→  // Look for Temu results that have price info in snippets
   737→  for (const result of results) {
   738→    if (!result.url?.includes("temu.com")) continue;
   739→    if (!result.snippet) continue;
   740→
   741→    const snippet = result.snippet;
   742→
   743→    // Extract the goods_id from the URL if we don't have it
   744→    let foundGoodsId = goodsId;
   745→    if (!foundGoodsId) {
   746→      const gMatch = result.url.match(/-g-(\d{10,})/);
   747→      if (gMatch) foundGoodsId = gMatch[1];
   748→    }
   749→
   750→    // Extract product name from search result
   751→    const productName = result.name
   752→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   753→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   754→      .trim() || null;
   755→
   756→    // Determine the local currency from the Temu URL locale prefix
   757→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   758→    let localCurrency = "USD";
   759→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   760→    if (localeMatch) {
   761→      const locale = localeMatch[1].toLowerCase();
   762→      const localeToCurrency: Record<string, string> = {
   763→        mu: "MUR",       // Mauritius
   764→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   765→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   766→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   767→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   768→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   769→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   770→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   771→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   772→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   773→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   774→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   775→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   776→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   777→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   778→        np: "NPR", "np-en": "NPR", // Nepal
   779→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   780→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   781→        ph: "PHP", "ph-en": "PHP", // Philippines
   782→        br: "BRL", "br-pt": "BRL", // Brazil
   783→        mx: "MXN", "mx-es": "MXN", // Mexico
   784→      };
   785→      const found = localeToCurrency[locale];
   786→      if (found) localCurrency = found;
   787→    }
   788→
   789→    // Try to find a price in the snippet
   790→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   791→    // Common patterns:
   792→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   793→    //   "$6.99"                  → USD
   794→    //   "OMR 1.200"             → Omani Rial
   795→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   796→    //   "€5.99"                  → EUR
   797→    //   "£4.99"                  → GBP
   798→
   799→    // Determine if this locale uses European comma format (comma = decimal separator)
   800→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   801→    const europeanCommaLocales = new Set([
   802→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",
   803→      "ma-fr", "tn-fr", // French-speaking locales
   804→    ]);
   805→    const usesEuropeanComma = europeanCommaLocales.has(localCurrency === "USD" && localeMatch ? localeMatch[1].toLowerCase() : "") ||
   806→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   807→
   808→    // Helper: parse price string respecting locale format
   809→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   810→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   811→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   812→      if (isEuropean) {
   813→        // European: remove dots (thousands), replace comma with dot (decimal)
   814→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   815→      } else {
   816→        // English: remove commas (thousands)
   817→        return parseFloat(priceStr.replace(/,/g, ""));
   818→      }
   819→    };
   820→
   821→    // First, try explicit currency patterns (most reliable)
   822→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   823→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   824→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   825→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   826→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   827→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   828→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   829→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   830→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   831→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   832→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   833→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   834→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   835→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   836→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   837→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   838→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   839→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   840→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   841→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   842→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   843→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   844→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   845→    ];
   846→
   847→    for (const { pattern, currency } of explicitPatterns) {
   848→      const match = snippet.match(pattern);
   849→      if (match) {
   850→        // For USD prices on European locale pages (like Ecuador), use comma-as-decimal parsing
   851→        const isEuroPrice = currency === "USD" && usesEuropeanComma;
   852→        const localPrice = parsePriceString(match[1], isEuroPrice);
   853→        if (localPrice > 0 && localPrice < 10000000) {
   854→          const rate = CURRENCY_TO_USD[currency];
   855→          if (rate) {
   856→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   857→            if (usdPrice >= 0.10 && usdPrice <= 5000) {
   858→              console.log(`[WebSearch] Found explicit price: ${localPrice} ${currency} (European format: ${isEuroPrice}) = $${usdPrice} USD`);
   859→              return {
   860→                price: usdPrice,
   861→                currency: "USD",
   862→                productName,
   863→                productDescription: null,
   864→                canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   865→                originalPrice: null,
   866→                image: null,
   867→                source: `web-search-snippet(${currency})`,
   868→                antiBotDetected: false,
   869→              };
   870→            }
   871→          }
   872→        }
   873→      }
   874→    }
   875→
   876→    // Also try European comma format pattern for $ prices (e.g., "$5,22" on Ecuador page)
   877→    // This catches prices where the comma is used as decimal separator
   878→    if (usesEuropeanComma) {
   879→      const euroPattern = /\$\s*(\d+,\d{1,2})\b/;
   880→      const euroMatch = snippet.match(euroPattern);
   881→      if (euroMatch) {
   882→        const localPrice = parseFloat(euroMatch[1].replace(",", "."));
   883→        if (localPrice > 0 && localPrice < 10000000) {
   884→          const usdPrice = Math.round(localPrice * 100) / 100; // Already USD
   885→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   886→            console.log(`[WebSearch] Found European format price: ${euroMatch[0]} = $${usdPrice} USD`);
   887→            return {
   888→              price: usdPrice,
   889→              currency: "USD",
   890→              productName,
   891→              productDescription: null,
   892→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   893→              originalPrice: null,
   894→              image: null,
   895→              source: "web-search-snippet(USD-european)",
   896→              antiBotDetected: false,
   897→            };
   898→          }
   899→        }
   900→      }
   901→    }
   902→
   903→    // Second, try "Rs" pattern — use the URL locale to determine the currency
   904→    // Rs could be MUR (Mauritius), PKR (Pakistan), LKR (Sri Lanka), NPR (Nepal)
   905→    const rsMatch = snippet.match(/Rs\.?\s*([\d,]+(?:\.\d{1,2})?)/);
   906→    if (rsMatch) {
   907→      const localPrice = parseFloat(rsMatch[1].replace(/,/g, ""));
   908→      const rsCurrency = localCurrency; // Determined from URL locale above
   909→      if (localPrice > 0 && localPrice < 10000000) {
   910→        const rate = CURRENCY_TO_USD[rsCurrency];
   911→        if (rate) {
   912→          const usdPrice = Math.round(localPrice * rate * 100) / 100;
   913→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   914→            console.log(`[WebSearch] Found Rs price: ${localPrice} ${rsCurrency} (from locale) = $${usdPrice} USD`);
   915→            return {

  1396→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1397→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1398→  // 1. OG meta tags (work even on anti-bot page)
  1399→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1400→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1401→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1402→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1403→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1404→
  1405→  // Decode HTML entities in title
  1406→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1407→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1408→
  1409→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1410→  let price: number | null = null;
  1411→  let currency = "USD";
  1412→  let originalPrice: number | null = null;
  1413→  let priceSource = "";
  1414→
  1415→  // ─────────────────────────────────────────────────────────────
  1416→  // 2-preferred. URL-embedded price hint.
  1417→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1418→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1419→  // → 1.23 USD. This is the most reliable signal because it comes
  1420→  // straight from the share URL the user copied.
  1421→  // ─────────────────────────────────────────────────────────────
  1422→  try {
  1423→    const parsedUrl = new URL(originalUrl);
  1424→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1425→    if (hint) {
  1426→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1427→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1428→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1429→      // Strip non-digits
  1430→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1431→      if (cents > 0 && cents < 10000000) {
  1432→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1433→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1434→        const usd = cents / 100;
  1435→        if (usd >= 0.01 && usd < 100000) {
  1436→          price = usd;
  1437→          currency = "USD";
  1438→          priceSource = "url-hint";
  1439→        }
  1440→      }
  1441→    }
  1442→  } catch { /* not a parseable URL */ }
  1443→
  1444→  // 2a. JSON-LD structured data
  1445→  if (!price) {
  1446→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1447→    for (const match of jsonLdMatches) {
  1448→      try {
  1449→        const data = JSON.parse(match[1]);
  1450→        const schemas = Array.isArray(data) ? data : [data];
  1451→        for (const schema of schemas) {
  1452→          if (schema["@type"] === "Product" && schema.offers) {
  1453→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1454→            for (const offer of offers) {
  1455→              if (offer.price !== undefined) {
  1456→                price = parseFloat(offer.price);
  1457→                currency = offer.priceCurrency || "USD";
  1458→                priceSource = "json-ld";
  1459→                break;
  1460→              }
  1461→            }
  1462→          }
  1463→        }
  1464→      } catch { /* skip */ }
  1465→      if (price) break;
  1466→    }
  1467→  }
  1468→
  1469→  // 2b. OG product:price:amount meta
  1470→  if (!price) {
  1471→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1472→    if (ogPrice) {
  1473→      const p = parseFloat(ogPrice[1]);
  1474→      if (p > 0) {
  1475→        price = p;
  1476→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1477→        currency = ogCur?.[1] || "USD";
  1478→        priceSource = "og-meta";
  1479→      }
  1480→    }
  1481→  }
  1482→
  1483→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1484→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1485→  //     one per "similar product", one per cross-sell, etc.). The previous
  1486→  //     implementation picked the FIRST match, which often returned a
  1487→  //     cross-sell product's price (e.g. $30) instead of the actual
  1488→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1489→  //     pick the LOWEST plausible one, since the actual product the user
  1490→  //     is viewing is almost always the cheapest among the rendered
  1491→  //     priceInfo blocks.
  1492→  if (!price) {
  1493→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1494→    const candidates: Candidate[] = [];
  1495→
  1496→    // priceInfo with explicit priceStr (most reliable)
  1497→    const priceInfosWithStr = [...html.matchAll(
  1498→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1499→    )];
  1500→    for (const pi of priceInfosWithStr) {
  1501→      const cents = parseInt(pi[1]);
  1502→      const cur = pi[2];
  1503→      const priceStr = pi[3];
  1504→      // Skip i18n placeholder strings
  1505→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1506→      if (cents <= 0 || cents >= 100000000) continue;
  1507→      const usd = cents / 100;
  1508→      if (usd < 0.01 || usd > 100000) continue;
  1509→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1510→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1511→      candidates.push({ usd, currency: cur, marketUsd });
  1512→    }
  1513→
  1514→    // priceInfo without priceStr (broader fallback)
  1515→    if (candidates.length === 0) {
  1516→      const priceInfosAll = [...html.matchAll(
  1517→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1518→      )];
  1519→      for (const pi of priceInfosAll) {
  1520→        const cents = parseInt(pi[1]);
  1521→        const cur = pi[2];
  1522→        if (cents <= 50 || cents >= 100000000) continue;
  1523→        const usd = cents / 100;
  1524→        if (usd < 0.50 || usd > 100000) continue;
  1525→        candidates.push({ usd, currency: cur });
  1526→      }
  1527→    }
  1528→
  1529→    if (candidates.length > 0) {
  1530→      // Pick the LOWEST price — the user is always viewing the cheapest
  1531→      // variant of the actual product, not the cross-sells.
  1532→      candidates.sort((a, b) => a.usd - b.usd);
  1533→      const best = candidates[0];
  1534→      price = best.usd;
  1535→      currency = best.currency;
  1536→      if (best.marketUsd && best.marketUsd > best.usd) {
  1537→        originalPrice = best.marketUsd;
  1538→      }
  1539→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1540→    }
  1541→  }
  1542→
  1543→  // 2d. Embedded JSON price fields (broad regex sweep)
  1544→  if (!price) {
  1545→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1546→    const found: { value: number; field: string }[] = [];
  1547→    for (const f of fields) {
  1548→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1549→      for (const m of html.matchAll(re)) {
  1550→        const v = parseFloat(m[1]);
  1551→        if (v > 0 && v < 100000) {
  1552→          found.push({ value: v, field: f });
  1553→        }
  1554→      }
  1555→    }
  1556→    if (found.length > 0) {
  1557→      // Sort and pick lowest (likely sale price)
  1558→      found.sort((a, b) => a.value - b.value);
  1559→      const best = found[0];
  1560→      // Heuristic: if value > 100, probably in cents → divide by 100
  1561→      price = best.value > 100 ? best.value / 100 : best.value;
  1562→      priceSource = `embedded-${best.field}`;
  1563→    }
  1564→  }
  1565→
  1566→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1567→  if (!price) {
  1568→    const text = html.replace(/<[^>]*>/g, " ");
  1569→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1570→    const prices: { value: number; currency: string }[] = [];
  1571→    for (const m of matches) {
  1572→      const raw = m[1].replace(/,/g, "");
  1573→      const v = parseFloat(raw);
  1574→      if (v > 1.5 && v < 5000) {
  1575→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1576→        prices.push({ value: v, currency: cur });
  1577→      }
  1578→    }
  1579→    if (prices.length > 0) {
  1580→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1581→      const sorted = prices.sort((a, b) => a.value - b.value);
  1582→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1583→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1584→      if (filtered.length > 0) {
  1585→        price = filtered[0].value;
  1586→        currency = filtered[0].currency;
  1587→        priceSource = "text";
  1588→      }
  1589→    }
  1590→  }
  1591→
  1592→  return {
  1593→    price,
  1594→    currency,
  1595→    productName,
  1596→    productDescription,
  1597→    canonicalUrl: canonical,
  1598→    originalPrice,
  1599→    image: ogImage,
  1600→    source: priceSource || (productName ? "og-only" : "none"),
  1601→  };
  1602→}

  1604→function decodeHtmlEntities(s: string): string {
  1605→  return s
  1606→    .replace(/&amp;/g, "&")
  1607→    .replace(/&lt;/g, "<")
  1608→    .replace(/&gt;/g, ">")
  1609→    .replace(/&quot;/g, '"')
  1610→    .replace(/&#39;/g, "'")
  1611→    .replace(/&nbsp;/g, " ")
  1612→    .replace(/&#(\d+);/g, (_, n) => String.fromCharCode(parseInt(n)));
  1613→}

  1615→/* ─── Helper: Extract product name from URL slug ─── */
  1616→function extractProductNameFromUrl(url: string): string | null {
  1617→  try {
  1618→    const parsed = new URL(url);
  1619→    const segments = parsed.pathname.split("/").filter(Boolean);
  1620→    const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
  1621→    const name = slug
  1622→      .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
  1623→      .replace(/\.html?$/i, "")
  1624→      .replace(/-/g, " ")
  1625→      .trim();
  1626→    if (name && name.length > 3) {
  1627→      return name.replace(/\b\w/g, (l) => l.toUpperCase());
  1628→    }
  1629→  } catch { /* skip */ }
  1630→  return null;
  1631→}

  1633→function cleanProductName(name: string | null): string | null {
  1634→  if (!name) return null;
  1635→  let cleaned = name
  1636→    .replace(/\s*[-|]\s*(Temu|AliExpress)\s*/gi, "")
  1637→    .replace(/\s*[-|]\s*(Login|Sign In|Register)\s*/gi, "")
  1638→    // Remove Temu locale suffixes like "Algeria", "Mauritius", "Oman", etc.
  1639→    .replace(/\s*(Algeria|Mauritius|Oman|Bahrain|Ecuador|Pakistan|Morocco|Tunisia|Kuwait|Qatar|Jordan|Egypt|Saudi Arabia|UAE|India|Philippines|Brazil|Mexico|Sri Lanka|Nepal|Bangladesh|China|Japan|Korea)$/gi, "")
  1640→    .replace(/\s+/g, " ")
  1641→    .trim();
  1642→  if (cleaned.length > 100) cleaned = cleaned.slice(0, 97) + "...";
  1643→  return cleaned || null;
  1644→}

  1855→        // We must extract and convert BEFORE reconstructing the URL (which removes

  1857→        {

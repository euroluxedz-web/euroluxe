    98→export default function CalculateurPage() {
    99→  const [productUrl, setProductUrl] = useState("");
   100→  const [manualPrice, setManualPrice] = useState("");
   101→  const [loading, setLoading] = useState(false);
   102→  const [result, setResult] = useState<PriceResult | null>(null);
   103→  const [error, setError] = useState("");
   104→  const [copied, setCopied] = useState(false);
   105→  const [addedToCart, setAddedToCart] = useState(false);
   106→  const [detectedCode, setDetectedCode] = useState<string | null>(null);
   107→  const [temuLink, setTemuLink] = useState<string | null>(null);
   108→  const [detectedProduct, setDetectedProduct] = useState<DetectedProduct | null>(null);
   109→  const priceInputRef = useRef<HTMLInputElement>(null);
   110→  const { t, isArabic } = useLanguage();
   111→  const { user, profile } = useAuth();
   112→  const router = useRouter();
   113→  const addItemToStore = useCartStore((s) => s.addItem);
   114→
   115→  // Checkout state
   116→  const [showCheckout, setShowCheckout] = useState(false);
   117→  const [orderSuccess, setOrderSuccess] = useState(false);
   118→  const [submitting, setSubmitting] = useState(false);
   119→  const [saveInfo, setSaveInfo] = useState(true);
   120→  const [useSaved, setUseSaved] = useState(false);
   121→  const [shippingError, setShippingError] = useState("");
   122→
   123→  const [shipping, setShipping] = useState<ShippingInfo>({
   124→    fullName: "",
   125→    phone: "",
   126→    wilaya: "",
   127→    commune: "",
   128→    codePostal: "",
   129→    address: "",
   130→    notes: "",
   131→  });
   132→
   133→  // Available communes based on selected wilaya
   134→  const [availableCommunes, setAvailableCommunes] = useState<Commune[]>([]);
   135→
   136→  const isAuthenticated = !!user;
   137→
   138→  // Pre-fill shipping info from user profile
   139→  useEffect(() => {
   140→    if (profile && isAuthenticated) {
   141→      const hasSavedInfo = profile.phone || profile.wilaya || profile.address;
   142→      if (hasSavedInfo) {
   143→        setUseSaved(true);
   144→        const communes = getCommunesForWilaya(profile.wilaya || "");
   145→        setAvailableCommunes(communes);
   146→        setShipping({
   147→          fullName: profile.name || "",
   148→          phone: profile.phone || "",
   149→          wilaya: profile.wilaya || "",
   150→          commune: profile.commune || "",
   151→          codePostal: profile.codePostal || "",
   152→          address: profile.address || "",
   153→          notes: "",
   154→        });
   155→      } else {
   156→        setShipping((prev) => ({
   157→          ...prev,
   158→          fullName: profile.name || "",
   159→          phone: profile.phone || "",
   160→        }));
   161→      }
   162→    }
   163→  }, [profile, isAuthenticated]);
   164→
   165→  // Detect Temu product code in real-time
   166→  useEffect(() => {
   167→    const input = productUrl.trim();
   168→    const isTemuCode = /^[a-zA-Z0-9]{6,20}$/.test(input);
   169→    if (isTemuCode) {
   170→      setDetectedCode(input);
   171→      setTemuLink(`https://www.temu.com/-g-${input}.html`);
   172→    } else if (input.includes("temu.com")) {
   173→      setDetectedCode(null);
   174→      setTemuLink(input.startsWith("http") ? input : `https://${input}`);
   175→    } else {
   176→      setDetectedCode(null);
   177→      setTemuLink(null);
   178→    }
   179→  }, [productUrl]);
   180→
   181→  // Extract product name from URL
   182→  const extractProductName = (url: string): string | null => {
   183→    try {
   184→      const parsed = new URL(url.startsWith("http") ? url : `https://${url}`);
   185→      const segments = parsed.pathname.split("/").filter(Boolean);
   186→      const slug =
   187→        segments.find((s) => s.includes("-g-") && s.length > 10) ||
   188→        segments.find((s) => s.includes("-") && s.length > 10) ||
   189→        segments[segments.length - 1] ||
   190→        "";
   191→      const name = slug
   192→        .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   193→        .replace(/\.html?$/i, "")
   194→        .replace(/-/g, " ")
   195→        .trim();
   196→      if (name && name.length > 3) {
   197→        return name.replace(/\b\w/g, (l) => l.toUpperCase());
   198→      }
   199→    } catch { /* skip */ }
   200→    return null;
   201→  };
   202→
   203→  // AUTO-EXTRACT
   204→  const handleAutoExtract = async () => {
   205→    setError("");
   206→    setResult(null);
   207→    setShowCheckout(false);
   208→    setOrderSuccess(false);
   209→    setDetectedProduct(null);
   210→
   211→    if (!productUrl.trim()) {
   212→      setError(t("calc.error.empty"));
   213→      return;
   214→    }
   215→
   216→    const isTemuProductId = /^[a-zA-Z0-9]{6,30}$/.test(productUrl.trim());
   217→    let finalUrl = productUrl.trim();
   218→
   219→    if (isTemuProductId) {
   220→      finalUrl = `https://www.temu.com/-g-${productUrl.trim()}.html`;
   221→    } else {
   222→      try {
   223→        new URL(finalUrl);
   224→      } catch {
   225→        setError(t("calc.error.invalidUrl"));
   226→        return;
   227→      }
   228→    }
   229→
   230→    setLoading(true);
   231→
   232→    try {
   233→      const response = await fetch("/api/scrape-price", {
   234→        method: "POST",
   235→        headers: { "Content-Type": "application/json" },
   236→        body: JSON.stringify({ url: finalUrl }),
   237→      });
   238→
   239→      const data = await response.json();
   240→
   241→      // Case 1: Auto-extracted price found
   242→      if (data.success && data.price && data.price > 0) {
   243→        setTemuLink(finalUrl);
   244→        setResult({
   245→          usd: data.price,
   246→          dzd: data.dzd || data.price * 270,
   247→          breakdown: data.breakdown,
   248→          productName: data.productName,
   249→          originalPrice: data.originalPrice || null,
   250→          image: data.productImage || data.image || null,
   251→          estimated: data.estimated || false,
   252→          manual: data.manual || false,
   253→          source: data.source || "auto",
   254→        });
   255→      }
   256→      // Case 2: Product detected but needs manual price entry
   257→      else if (data.success && data.requiresManualPrice && data.productName) {
   258→        setTemuLink(data.productUrl || finalUrl);
   259→        setDetectedProduct({
   260→          name: data.productName,
   261→          description: data.productDescription,
   262→          image: data.productImage || null,
   263→          url: data.productUrl || finalUrl,
   264→          antiBotDetected: data.antiBotDetected,
   265→          message: data.message,
   266→        });
   267→        setError(
   268→          data.message ||
   269→            (isArabic
   270→              ? "تم العثور على المنتج! يرجى إدخال السعر المعروض على Temu في الحقل أدناه."
   271→              : "Produit trouvé ! Veuillez saisir le prix affiché sur Temu dans le champ ci-dessous.")
   272→        );
   273→        setTimeout(() => priceInputRef.current?.focus(), 300);
   274→      }
   275→      // Case 3: Failure
   276→      else {
   277→        setTemuLink(finalUrl);
   278→        setError(
   279→          data.error ||
   280→            (isArabic
   281→              ? "لم نتمكن من استخراج السعر تلقائياً. يرجى إدخاله يدوياً."
   282→              : "Extraction automatique indisponible. Veuillez entrer le prix manuellement.")
   283→        );
   284→        setTimeout(() => priceInputRef.current?.focus(), 300);
   285→      }
   286→    } catch {
   287→      setError(
   288→        isArabic
   289→          ? "يرجى إدخال السعر يدوياً في الحقل أدناه."
   290→          : "Veuillez entrer le prix manuellement dans le champ ci-dessous."
   291→      );
   292→      setTimeout(() => priceInputRef.current?.focus(), 300);
   293→    } finally {
   294→      setLoading(false);
   295→    }
   296→  };
   297→
   298→  // Manual price calculation - uses API for proper breakdown
   299→  const handleManualCalculate = async () => {
   300→    setError("");
   301→    setResult(null);
   302→    setShowCheckout(false);
   303→    setOrderSuccess(false);
   304→
   305→    const normalized = manualPrice.trim().replace(/,/g, ".");
   306→    const priceStr = normalized.replace(/[^\d.]/g, "");
   307→    const parts = priceStr.split(".");
   308→    const cleaned = parts.length > 1
   309→      ? parts[0] + "." + parts.slice(1).join("")
   310→      : priceStr;
   311→    const price = parseFloat(cleaned);
   312→
   313→    if (!price || price <= 0) {
   314→      setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   315→      return;
   316→    }
   317→
   318→    setLoading(true);
   319→    try {
   320→      // Pass the detected product info + URL so the API can use it
   321→      const payload: Record<string, string> = { manualPrice: manualPrice.trim() };
   322→      if (detectedProduct?.name) payload.productName = detectedProduct.name;
   323→      if (detectedProduct?.image) payload.productImage = detectedProduct.image;
   324→      if (productUrl.trim()) payload.url = productUrl.trim();
   325→
   326→      const response = await fetch("/api/scrape-price", {
   327→        method: "POST",
   328→        headers: { "Content-Type": "application/json" },
   329→        body: JSON.stringify(payload),
   330→      });
   331→      const data = await response.json();
   332→
   333→      if (data.success && data.price) {
   334→        const productName = detectedProduct?.name
   335→          || (productUrl.trim() ? extractProductName(productUrl) : null)
   336→          || data.productName;
   337→        setResult({
   338→          usd: data.price,
   339→          dzd: data.dzd,
   340→          breakdown: data.breakdown,
   341→          productName,
   342→          image: detectedProduct?.image || null,
   343→          estimated: false,
   344→          manual: true,
   345→          source: "manual",
   346→        });
   347→      } else {
   348→        setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   349→      }
   350→    } catch {
   351→      setError(isArabic ? "حدث خطأ، حاول مرة أخرى" : "Une erreur est survenue, réessayez");
   352→    } finally {
   353→      setLoading(false);
   354→    }
   355→  };
   356→
   357→  const handleCopyResult = () => {
   358→    if (result) {
   359→      const text = result.productName
   360→        ? `${result.productName} — ${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`
   361→        : `${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`;
   362→      navigator.clipboard.writeText(text);
   363→      setCopied(true);
   364→      setTimeout(() => setCopied(false), 2000);
   365→    }
   366→  };
   367→
   368→  const handleAddToCart = () => {
   369→    if (!result) return;
   370→    const cartItem = {
   371→      id: `calc-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   372→      productId: detectedCode || undefined,
   373→      name: result.productName || (isArabic ? "منتج" : "Produit"),
   374→      image: result.image || undefined,
   375→      price: result.usd,
   376→      quantity: 1,
   377→      url: productUrl.trim() || undefined,
   378→    };
   379→    addItemToStore(cartItem);
   380→    syncAddToServer(cartItem);
   381→    setAddedToCart(true);
   382→    setTimeout(() => setAddedToCart(false), 2000);
   383→  };
   384→
   385→  // Validate phone
   386→  const isValidPhone = (phone: string) => /^(05|06|07)\d{8}$/.test(phone.trim());
   387→
   388→  // Handle checkout submission
   389→  const handleSubmitOrder = async () => {
   390→    setShippingError("");
   391→
   392→    // Validate
   393→    if (!shipping.fullName.trim()) {
   394→      setShippingError(isArabic ? "يرجى إدخال الاسم الكامل" : "Veuillez entrer votre nom complet");
   395→      return;
   396→    }
   397→    if (!isValidPhone(shipping.phone)) {
   398→      setShippingError(t("calc.checkout.errorPhone"));
   399→      return;
   400→    }
   401→    if (!shipping.wilaya) {
   402→      setShippingError(isArabic ? "يرجى اختيار الولاية" : "Veuillez sélectionner votre wilaya");
   403→      return;
   404→    }
   405→    if (!shipping.commune) {
   406→      setShippingError(isArabic ? "يرجى اختيار البلدية" : "Veuillez sélectionner votre commune");
   407→      return;
   408→    }
   409→    if (!shipping.address.trim()) {
   410→      setShippingError(isArabic ? "يرجى إدخال العنوان" : "Veuillez entrer votre adresse");
   411→      return;
   412→    }
   413→
   414→    setSubmitting(true);
   415→
   416→    try {
   417→      // Save shipping info to user profile if requested
   418→      if (isAuthenticated && saveInfo && user) {
   419→        try {
   420→          await updateUserData(user.uid, {
   421→            name: shipping.fullName,
   422→            phone: shipping.phone,
   423→            wilaya: shipping.wilaya,
   424→            commune: shipping.commune,
   425→            codePostal: shipping.codePostal,
   426→            address: shipping.address,
   427→          });
   428→        } catch (e) {
   429→          console.error("Failed to save shipping info:", e);
   430→        }
   431→      }
   432→
   433→      // Create the order
   434→      if (isAuthenticated && user) {
   435→        const orderItems = [{
   436→          name: result?.productName || (isArabic ? "منتج" : "Produit"),
   437→          price: result?.dzd || 0,
   438→          quantity: 1,
   439→          image: result?.image || undefined,
   440→          url: productUrl.trim() || undefined,
   441→          productId: detectedCode || undefined,
   442→        }];
   443→
   444→        const { auth } = await import("@/lib/firebase");
   445→        const token = auth.currentUser ? await auth.currentUser.getIdToken() : null;
   446→
   447→        const res = await fetch("/api/orders", {
   448→          method: "POST",
   449→          headers: {
   450→            "Content-Type": "application/json",
   451→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   452→          },
   453→          body: JSON.stringify({
   454→            items: orderItems,
   455→            total: result?.dzd || 0,
   456→            wilaya: shipping.wilaya,
   457→            commune: shipping.commune,
   458→            codePostal: shipping.codePostal,
   459→            address: shipping.address,
   460→            phone: shipping.phone,
   461→            notes: shipping.notes,
   462→          }),
   463→        });
   464→
   465→        if (res.ok) {
   466→          setOrderSuccess(true);
   467→        } else {
   468→          setShippingError(isArabic ? "حدث خطأ، يرجى المحاولة مرة أخرى" : "Une erreur est survenue, veuillez réessayer");
   469→        }
   470→      }
   471→    } catch (err) {
   472→      console.error("Order error:", err);
   473→      setShippingError(isArabic ? "حدث خطأ في الاتصال" : "Erreur de connexion");
   474→    } finally {
   475→      setSubmitting(false);
   476→    }
   477→  };
   478→
   479→  return (
   480→    <div className="relative min-h-screen flex flex-col bg-transparent text-foreground overflow-x-hidden">
   481→      <Navbar />
   482→      <main className="flex-1 pt-16 sm:pt-20">
   483→        <section className="relative py-20 sm:py-32 overflow-hidden min-h-[80vh]">
   484→          {/* Background Effects */}
   485→          <div className="absolute inset-0 bg-gradient-to-b from-brand-blue/20 via-brand-blue-light/15 to-white/60" />
   486→          <div className="absolute top-0 left-0 w-72 h-72 bg-brand-pink/8 rounded-full blur-3xl" />
   487→          <div className="absolute bottom-0 right-0 w-72 h-72 bg-brand-blue/8 rounded-full blur-3xl" />
   488→
   489→          <div className="relative z-10 max-w-4xl mx-auto px-4">
   490→            {/* Section Header */}
   491→            <motion.div
   492→              initial={{ opacity: 0, y: 30 }}
   493→              animate={{ opacity: 1, y: 0 }}
   494→              transition={{ duration: 0.6 }}
   495→              className="text-center mb-12 relative"
   496→            >
   497→              <div className="hidden lg:block">
   498→                <ImgPlaceholder number={32} className="absolute -left-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[-8deg]" />
   499→                <ImgPlaceholder number={33} className="absolute -right-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[8deg]" />
   500→              </div>
   501→
   502→              <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-brand-pink/10 border border-brand-pink/20 text-brand-pink text-sm font-medium mb-4 font-display">
   503→                <Zap className="w-4 h-4" />
   504→                {t("calc.badge")}
   505→              </div>
   506→              <h1 className="text-3xl sm:text-5xl lg:text-6xl font-black mb-4 font-heading">
   507→                <span className="text-brand-dark">{t("calc.titleCalc")}</span>{" "}
   508→                <span className="bg-brand-gold/30 px-2 py-1 rounded-md text-brand-dark">
   509→                  {t("calc.titleProduct")}
   510→                </span>
   511→              </h1>
   512→              <p className="text-brand-muted-text text-lg max-w-xl mx-auto font-sans">
   513→                {t("calc.subtitle")}
   514→              </p>
   515→            </motion.div>
   516→
   517→            {/* Calculator Card */}
   518→            <motion.div
   519→              initial={{ opacity: 0, y: 30 }}
   520→              animate={{ opacity: 1, y: 0 }}
   521→              transition={{ duration: 0.6, delay: 0.2 }}
   522→              className="bg-white/80 backdrop-blur-xl rounded-3xl p-6 sm:p-10 border border-brand-pink/15 shadow-lg"
   523→            >
   524→              {/* ──── Product URL / Code Input ──── */}
   525→              <div className="mb-6">
   526→                <label className="block text-brand-dark/80 text-sm font-medium mb-2 font-sans">
   527→                  <Link2 className={`w-4 h-4 inline ${isArabic ? "ml-1" : "mr-1"}`} />
   528→                  {t("calc.label")}
   529→                </label>
   530→                <div className="flex gap-3">
   531→                  <div className="relative flex-1">
   532→                    <Input
   533→                      type="text"
   534→                      placeholder={t("calc.placeholder")}
   535→                      value={productUrl}
   536→                      onChange={(e) => {
   537→                        setProductUrl(e.target.value);
   538→                        setResult(null);
   539→                        setError("");
   540→                        setShowCheckout(false);
   541→                        setDetectedProduct(null);
   542→                      }}
   543→                      onKeyDown={(e) => e.key === "Enter" && handleAutoExtract()}
   544→                      className="bg-brand-light/50 border-brand-muted-warm focus:border-brand-pink/50 focus:ring-brand-pink/20 text-brand-dark placeholder:text-brand-muted-text/50 rounded-xl h-14 text-base font-sans"
   545→                      disabled={loading}
   546→                    />
   547→                    <ShoppingBag
   548→                      className={`absolute top-1/2 -translate-y-1/2 w-5 h-5 text-brand-muted-text/40 ${
   549→                        isArabic ? "left-4" : "right-4"
   550→                      }`}
   551→                    />
   552→                  </div>
   553→                  <Button
   554→                    onClick={handleAutoExtract}
   555→                    disabled={loading || !productUrl.trim()}
   556→                    className="bg-brand-pink text-white hover:bg-brand-pink-light font-bold rounded-xl h-14 px-6 shadow-xl shadow-brand-pink/25 hover:shadow-brand-pink/40 hover:scale-[1.02] transition-all disabled:opacity-50 disabled:hover:scale-100 font-display"

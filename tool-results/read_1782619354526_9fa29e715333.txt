  2593→function isSuspiciousPrice(priceUSD: number, source: string): boolean {
  2594→  // Temu's "delivery guarantee" / "delay credit" is exactly $30.00 (or 9,000 DZD)
  2595→  // This is NOT the product price — it's a promotional amount that appears on many Temu pages
  2596→  if (priceUSD === 30.00 || priceUSD === 30) {
  2597→    console.log(`[PriceCheck] ⚠️ SUSPICIOUS: $30.00 from ${source} — likely "delivery guarantee" amount, not product price`);
  2598→    return true;
  2599→  }
  2600→  // Also check for $30.xx variations
  2601→  if (priceUSD >= 29.90 && priceUSD <= 30.10) {
  2602→    console.log(`[PriceCheck] ⚠️ SUSPICIOUS: $${priceUSD} from ${source} — close to $30.00 "delivery guarantee" amount`);
  2603→    return true;
  2604→  }

    36→/* ── Animated Counter Component ── */
    37→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
    38→  const [count, setCount] = useState(0);
    39→  const ref = useRef(null);
    40→  const isInView = useInView(ref, { once: true, margin: "-50px" });
    41→
    42→  useEffect(() => {
    43→    if (!isInView) return;
    44→    let start = 0;
    45→    const duration = 1500;
    46→    const startTime = Date.now();
    47→    const animate = () => {
    48→      const elapsed = Date.now() - startTime;
    49→      const progress = Math.min(elapsed / duration, 1);
    50→      // Ease out cubic
    51→      const eased = 1 - Math.pow(1 - progress, 3);
    52→      start = Math.floor(eased * target);
    53→      setCount(start);
    54→      if (progress < 1) {
    55→        requestAnimationFrame(animate);
    56→      } else {
    57→        setCount(target);
    58→      }
    59→    };
    60→    requestAnimationFrame(animate);
    61→  }, [isInView, target]);
    62→
    63→  return (
    64→    <span ref={ref}>
    65→      {count.toLocaleString()}{suffix}
    66→    </span>
    67→  );
    68→}

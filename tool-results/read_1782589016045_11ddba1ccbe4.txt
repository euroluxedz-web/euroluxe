    20→/* ───────────────────────────────────────────────────────────────────
    21→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
    22→ *
    23→ * Temu share URLs include everything we need:
    24→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
    25→ *   - top_gallery_url    = URL-encoded product image URL
    26→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
    27→ *
    28→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
    29→ * ─────────────────────────────────────────────────────────────────── */
    30→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
    31→  let parsed: URL;
    32→  try {
    33→    parsed = new URL(originalUrl);
    34→  } catch {
    35→    return null;
    36→  }
    37→
    38→  let price: number | null = null;
    39→  let currency = "USD";
    40→  let priceSource = "";
    41→
    42→  // Price from _oak_rec_ext_1 (base64-encoded cents)
    43→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
    44→  if (hint) {
    45→    try {
    46→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
    47→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
    48→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
    49→      if (cents > 0 && cents < 10000000) {
    50→        const usd = cents / 100;
    51→        if (usd >= 0.01 && usd < 100000) {
    52→          price = usd;
    53→          currency = "USD";
    54→          priceSource = "url-hint";
    55→        }
    56→      }
    57→    } catch { /* not valid base64 */ }
    58→  }
    59→
    60→  // Product image from top_gallery_url
    61→  let image: string | null = null;
    62→  const topGallery = parsed.searchParams.get("top_gallery_url");
    63→  if (topGallery) {
    64→    try {
    65→      // Validate it's a real URL
    66→      const imgUrl = new URL(topGallery);
    67→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
    68→        image = topGallery;
    69→      }
    70→    } catch { /* not a valid URL */ }
    71→  }
    72→
    73→  // Product name from URL slug
    74→  let productName: string | null = null;
    75→  const segments = parsed.pathname.split("/").filter(Boolean);
    76→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
    77→  const nameFromSlug = slug
    78→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
    79→    .replace(/\.html?$/i, "")
    80→    .replace(/-/g, " ")
    81→    .replace(/\s+/g, " ")
    82→    .trim();
    83→  if (nameFromSlug && nameFromSlug.length > 3) {
    84→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
    85→  }
    86→
    87→  // If we got at least the price (the most important piece), return success
    88→  if (price) {
    89→    return {
    90→      price,
    91→      currency,
    92→      productName,
    93→      productDescription: null,
    94→      canonicalUrl: null,
    95→      originalPrice: null,
    96→      image,
    97→      source: priceSource,
    98→      antiBotDetected: false,
    99→    };
   100→  }
   101→
   102→  // No price from URL — return null so caller tries HTML-based strategies
   103→  return null;
   104→}

   162→async function fetchPriceWithWebSearch(
   163→  goodsId: string | null,
   164→  itemId: string | null,
   165→  originalUrl: string,
   166→): Promise<TemuProductData | null> {
   167→  try {
   168→    const zai = await ZAI.create();
   169→
   170→    // Build search query — try different strategies
   171→    let searchQuery = "";
   172→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   173→      // Item ID like TV10922608 — search Temu specifically
   174→      searchQuery = `site:temu.com "${itemId}"`;
   175→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   176→      // Numeric goods_id — search with the -g- pattern
   177→      searchQuery = `site:temu.com "g-${goodsId}"`;
   178→    } else {
   179→      // Fallback: search with the URL
   180→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   181→      searchQuery = `site:temu.com ${cleanUrl}`;
   182→    }
   183→
   184→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   185→
   186→    // Step 1: Web search to find the product
   187→    const searchResults = await (zai as any).invokeFunction("web_search", {
   188→      query: searchQuery,
   189→      num: 5,
   190→    });
   191→
   192→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   193→      console.log("[WebSearch] No results found, trying broader search...");
   194→
   195→      // Broader search without site: restriction
   196→      const broadQuery = itemId
   197→        ? `temu ${itemId} price`
   198→        : goodsId
   199→          ? `temu ${goodsId} price`
   200→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   201→
   202→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   203→      const broadResults = await (zai as any).invokeFunction("web_search", {
   204→        query: broadQuery,
   205→        num: 5,
   206→      });
   207→
   208→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   209→        console.log("[WebSearch] No results from broader search either");
   210→        return null;
   211→      }
   212→
   213→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   214→    }
   215→
   216→    // Step 2: Try to parse price from initial search results
   217→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   218→    if (initialPrice?.price && initialPrice.price > 0) {
   219→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   220→      return initialPrice;
   221→    }
   222→
   223→    // Step 3: If initial search found product names but no prices,
   224→    // try a more targeted search with the product name to find prices
   225→    const productNameFromResults = searchResults
   226→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   227→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   228→      .find((n: string) => n && n.length > 10);
   229→
   230→    if (productNameFromResults) {
   231→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   232→      try {
   233→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   234→        const nameResults = await (zai as any).invokeFunction("web_search", {
   235→          query: nameQuery,
   236→          num: 5,
   237→        });
   238→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   239→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   240→          if (namePrice?.price && namePrice.price > 0) {
   241→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   242→            return namePrice;
   243→          }
   244→          // Merge results and try LLM on combined results
   245→          const combinedResults = [...searchResults, ...nameResults];
   246→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   247→        }
   248→      } catch (err) {
   249→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   250→      }
   251→    }
   252→
   253→    // Step 4: Use LLM on whatever results we have
   254→    return extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   255→  } catch (err) {
   256→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   257→    return null;
   258→  }
   259→}

   261→async function extractPriceFromSearchResults(
   262→  zai: any,
   263→  searchResults: any[],
   264→  goodsId: string | null,
   265→  itemId: string | null,
   266→): Promise<TemuProductData | null> {
   267→  // Step 2: Try to parse prices directly from search snippets first (fast, no LLM cost)
   268→  const snippetPrice = parsePriceFromSnippets(searchResults, goodsId);
   269→  if (snippetPrice?.price && snippetPrice.price > 0) {
   270→    console.log(`[WebSearch] ✓ Found price from snippet: ${snippetPrice.price} USD (source: ${snippetPrice.source})`);
   271→    return snippetPrice;
   272→  }
   273→
   274→  // Step 3: If snippets don't have a clear price, use LLM to extract
   275→  console.log("[WebSearch] Snippets don't have clear price, using LLM...");
   276→
   277→  const searchContext = searchResults
   278→    .slice(0, 5)
   279→    .map(
   280→      (r: any, i: number) =>
   281→        `${i + 1}. ${r.name || "No title"}\n   URL: ${r.url}\n   Snippet: ${r.snippet || "No snippet"}`,
   282→    )
   283→    .join("\n\n");
   284→
   285→  try {
   286→    const completion = await zai.createChatCompletion({
   287→      messages: [
   288→        {
   289→          role: "system",
   290→          content:
   291→            'You are a price extraction assistant for Temu products. Extract the product price from the search results. ' +
   292→            'Return ONLY a JSON object with: {"price_usd": <number_in_USD>, "name": "<product_name>", "currency": "<original_currency>", "original_price": <number_before_conversion>, "confidence": <high|medium|low>}. ' +
   293→            'If the price is in a non-USD currency, CONVERT it to USD using approximate exchange rates. ' +
   294→            'Common conversions: MUR→USD ÷47, OMR→USD ×2.60, BHD→USD ×2.65, PKR→USD ÷278, EUR→USD ×1.08, GBP→USD ×1.27, SAR→USD ×0.27, AED→USD ×0.27. ' +
   295→            'If you cannot find a clear price, return {"price_usd": null, "name": "<best_guess_name>", "confidence": "low"}. ' +
   296→            'ALWAYS return valid JSON. The price_usd must be a number, not a string.',
   297→        },
   298→        {
   299→          role: "user",
   300→          content:
   301→            `Product goods_id: ${goodsId || "unknown"}\n` +
   302→            `Item ID: ${itemId || "unknown"}\n\n` +
   303→            `Search Results:\n${searchContext}\n\n` +
   304→            `Extract the product price in USD from these results. Return JSON only.`,
   305→        },
   306→      ],
   307→    });
   308→
   309→    const aiResponse = completion.choices?.[0]?.message?.content || "";
   310→    console.log("[WebSearch] LLM response:", aiResponse.slice(0, 300));
   311→
   312→    // Parse AI response — extract JSON
   313→    const jsonMatch = aiResponse.match(/\{[\s\S]*?\}/);
   314→    if (!jsonMatch) {
   315→      console.log("[WebSearch] No JSON in LLM response");
   316→      return null;
   317→    }
   318→
   319→    const parsed = JSON.parse(jsonMatch[0]);
   320→    const priceUSD = typeof parsed.price_usd === "number"
   321→      ? parsed.price_usd
   322→      : parseFloat(String(parsed.price_usd));
   323→
   324→    if (priceUSD && priceUSD > 0 && priceUSD < 100000) {
   325→      console.log(`[WebSearch] ✓ LLM found price: $${priceUSD} (confidence: ${parsed.confidence})`);
   326→      return {
   327→        price: priceUSD,
   328→        currency: "USD",
   329→        productName: parsed.name || null,
   330→        productDescription: null,
   331→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   332→        originalPrice: null,
   333→        image: null, // Image will come from share URL params
   334→        source: `web-search-llm(${parsed.confidence})`,
   335→        antiBotDetected: false,
   336→      };
   337→    }
   338→
   339→    // LLM couldn't find price either — return what we can
   340→    if (parsed.name) {
   341→      return {
   342→        price: null,
   343→        currency: "USD",
   344→        productName: parsed.name,
   345→        productDescription: null,
   346→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   347→        originalPrice: null,
   348→        image: null,
   349→        source: "web-search-llm(no-price)",
   350→        antiBotDetected: false,
   351→      };
   352→    }
   353→
   354→    return null;
   355→  } catch (err) {
   356→    console.error("[WebSearch] LLM error:", String(err).slice(0, 150));
   357→    return null;
   358→  }
   359→}

   361→/**
   362→ * Try to parse prices directly from search result snippets.
   363→ * This avoids the cost of an LLM call and is often sufficient
   364→ * because Google includes price info in Temu search snippets.
   365→ */
   366→function parsePriceFromSnippets(
   367→  results: any[],
   368→  goodsId: string | null,
   369→): TemuProductData | null {
   370→  // Look for Temu results that have price info in snippets
   371→  for (const result of results) {
   372→    if (!result.url?.includes("temu.com")) continue;
   373→    if (!result.snippet) continue;
   374→
   375→    const snippet = result.snippet;
   376→
   377→    // Extract the goods_id from the URL if we don't have it
   378→    let foundGoodsId = goodsId;
   379→    if (!foundGoodsId) {
   380→      const gMatch = result.url.match(/-g-(\d{10,})/);
   381→      if (gMatch) foundGoodsId = gMatch[1];
   382→    }
   383→
   384→    // Extract product name from search result
   385→    const productName = result.name
   386→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   387→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   388→      .trim() || null;
   389→
   390→    // Determine the local currency from the Temu URL locale prefix
   391→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   392→    let localCurrency = "USD";
   393→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   394→    if (localeMatch) {
   395→      const locale = localeMatch[1].toLowerCase();
   396→      const localeToCurrency: Record<string, string> = {
   397→        mu: "MUR",       // Mauritius
   398→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   399→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   400→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   401→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   402→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   403→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   404→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   405→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   406→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   407→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   408→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   409→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   410→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   411→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   412→        np: "NPR", "np-en": "NPR", // Nepal
   413→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   414→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   415→        ph: "PHP", "ph-en": "PHP", // Philippines
   416→        br: "BRL", "br-pt": "BRL", // Brazil
   417→        mx: "MXN", "mx-es": "MXN", // Mexico
   418→      };
   419→      const found = localeToCurrency[locale];
   420→      if (found) localCurrency = found;
   421→    }
   422→
   423→    // Try to find a price in the snippet
   424→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   425→    // Common patterns:
   426→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   427→    //   "$6.99"                  → USD
   428→    //   "OMR 1.200"             → Omani Rial
   429→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   430→    //   "€5.99"                  → EUR
   431→    //   "£4.99"                  → GBP
   432→
   433→    // First, try explicit currency patterns (most reliable)
   434→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   435→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   436→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   437→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   438→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   439→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   440→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   441→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   442→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   443→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   444→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   445→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   446→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   447→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   448→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   449→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   450→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   451→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   452→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   453→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   454→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   455→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   456→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   457→    ];
   458→
   459→    for (const { pattern, currency } of explicitPatterns) {
   460→      const match = snippet.match(pattern);
   461→      if (match) {
   462→        const localPrice = parseFloat(match[1].replace(/,/g, ""));
   463→        if (localPrice > 0 && localPrice < 10000000) {
   464→          const rate = CURRENCY_TO_USD[currency];
   465→          if (rate) {
   466→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   467→            if (usdPrice >= 0.10 && usdPrice <= 5000) {
   468→              console.log(`[WebSearch] Found explicit price: ${localPrice} ${currency} = $${usdPrice} USD`);
   469→              return {
   470→                price: usdPrice,
   471→                currency: "USD",
   472→                productName,
   473→                productDescription: null,
   474→                canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   475→                originalPrice: null,
   476→                image: null,
   477→                source: `web-search-snippet(${currency})`,
   478→                antiBotDetected: false,
   479→              };
   480→            }
   481→          }
   482→        }
   483→      }
   484→    }
   485→
   486→    // Second, try "Rs" pattern — use the URL locale to determine the currency
   487→    // Rs could be MUR (Mauritius), PKR (Pakistan), LKR (Sri Lanka), NPR (Nepal)
   488→    const rsMatch = snippet.match(/Rs\.?\s*([\d,]+(?:\.\d{1,2})?)/);
   489→    if (rsMatch) {
   490→      const localPrice = parseFloat(rsMatch[1].replace(/,/g, ""));
   491→      const rsCurrency = localCurrency; // Determined from URL locale above
   492→      if (localPrice > 0 && localPrice < 10000000) {
   493→        const rate = CURRENCY_TO_USD[rsCurrency];
   494→        if (rate) {
   495→          const usdPrice = Math.round(localPrice * rate * 100) / 100;
   496→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   497→            console.log(`[WebSearch] Found Rs price: ${localPrice} ${rsCurrency} (from locale) = $${usdPrice} USD`);
   498→            return {
   499→              price: usdPrice,
   500→              currency: "USD",
   501→              productName,
   502→              productDescription: null,
   503→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   504→              originalPrice: null,
   505→              image: null,
   506→              source: `web-search-snippet(Rs→${rsCurrency})`,
   507→              antiBotDetected: false,
   508→            };
   509→          }
   510→        }
   511→      }
   512→    }
   513→
   514→    // Third, try number followed by Rs (reverse pattern)
   515→    const rsReverseMatch = snippet.match(/([\d,]+(?:\.\d{1,2})?)\s*Rs\.?/);
   516→    if (rsReverseMatch) {
   517→      const localPrice = parseFloat(rsReverseMatch[1].replace(/,/g, ""));
   518→      const rsCurrency = localCurrency;
   519→      if (localPrice > 0 && localPrice < 10000000) {
   520→        const rate = CURRENCY_TO_USD[rsCurrency];
   521→        if (rate) {
   522→          const usdPrice = Math.round(localPrice * rate * 100) / 100;
   523→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   524→            console.log(`[WebSearch] Found Rs price (reverse): ${localPrice} ${rsCurrency} = $${usdPrice} USD`);
   525→            return {
   526→              price: usdPrice,
   527→              currency: "USD",
   528→              productName,
   529→              productDescription: null,
   530→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   531→              originalPrice: null,
   532→              image: null,
   533→              source: `web-search-snippet(Rs→${rsCurrency})`,
   534→              antiBotDetected: false,
   535→            };
   536→          }
   537→        }
   538→      }
   539→    }
   540→  }
   541→
   542→  return null;
   543→}

   874→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
   875→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
   876→  // 1. OG meta tags (work even on anti-bot page)
   877→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   878→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   879→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   880→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   881→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
   882→
   883→  // Decode HTML entities in title
   884→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
   885→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
   886→
   887→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
   888→  let price: number | null = null;
   889→  let currency = "USD";
   890→  let originalPrice: number | null = null;
   891→  let priceSource = "";
   892→
   893→  // ─────────────────────────────────────────────────────────────
   894→  // 2-preferred. URL-embedded price hint.
   895→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
   896→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
   897→  // → 1.23 USD. This is the most reliable signal because it comes
   898→  // straight from the share URL the user copied.
   899→  // ─────────────────────────────────────────────────────────────
   900→  try {
   901→    const parsedUrl = new URL(originalUrl);
   902→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
   903→    if (hint) {
   904→      // base64 decode (URL-safe variant — replace - with + and _ with /)
   905→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   906→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   907→      // Strip non-digits
   908→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   909→      if (cents > 0 && cents < 10000000) {
   910→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
   911→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
   912→        const usd = cents / 100;
   913→        if (usd >= 0.01 && usd < 100000) {
   914→          price = usd;
   915→          currency = "USD";
   916→          priceSource = "url-hint";
   917→        }
   918→      }
   919→    }
   920→  } catch { /* not a parseable URL */ }
   921→
   922→  // 2a. JSON-LD structured data
   923→  if (!price) {
   924→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
   925→    for (const match of jsonLdMatches) {
   926→      try {
   927→        const data = JSON.parse(match[1]);
   928→        const schemas = Array.isArray(data) ? data : [data];
   929→        for (const schema of schemas) {
   930→          if (schema["@type"] === "Product" && schema.offers) {
   931→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
   932→            for (const offer of offers) {
   933→              if (offer.price !== undefined) {
   934→                price = parseFloat(offer.price);
   935→                currency = offer.priceCurrency || "USD";
   936→                priceSource = "json-ld";
   937→                break;
   938→              }
   939→            }
   940→          }
   941→        }
   942→      } catch { /* skip */ }
   943→      if (price) break;
   944→    }
   945→  }
   946→
   947→  // 2b. OG product:price:amount meta
   948→  if (!price) {
   949→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   950→    if (ogPrice) {
   951→      const p = parseFloat(ogPrice[1]);
   952→      if (p > 0) {
   953→        price = p;
   954→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   955→        currency = ogCur?.[1] || "USD";
   956→        priceSource = "og-meta";
   957→      }
   958→    }
   959→  }
   960→
   961→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
   962→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
   963→  //     one per "similar product", one per cross-sell, etc.). The previous
   964→  //     implementation picked the FIRST match, which often returned a
   965→  //     cross-sell product's price (e.g. $30) instead of the actual
   966→  //     product's price (e.g. $1.23). We now collect ALL matches and
   967→  //     pick the LOWEST plausible one, since the actual product the user
   968→  //     is viewing is almost always the cheapest among the rendered
   969→  //     priceInfo blocks.
   970→  if (!price) {
   971→    type Candidate = { usd: number; currency: string; marketUsd?: number };
   972→    const candidates: Candidate[] = [];
   973→
   974→    // priceInfo with explicit priceStr (most reliable)
   975→    const priceInfosWithStr = [...html.matchAll(
   976→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
   977→    )];
   978→    for (const pi of priceInfosWithStr) {
   979→      const cents = parseInt(pi[1]);
   980→      const cur = pi[2];
   981→      const priceStr = pi[3];
   982→      // Skip i18n placeholder strings
   983→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
   984→      if (cents <= 0 || cents >= 100000000) continue;
   985→      const usd = cents / 100;
   986→      if (usd < 0.01 || usd > 100000) continue;
   987→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
   988→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
   989→      candidates.push({ usd, currency: cur, marketUsd });
   990→    }
   991→
   992→    // priceInfo without priceStr (broader fallback)
   993→    if (candidates.length === 0) {
   994→      const priceInfosAll = [...html.matchAll(
   995→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
   996→      )];
   997→      for (const pi of priceInfosAll) {
   998→        const cents = parseInt(pi[1]);
   999→        const cur = pi[2];
  1000→        if (cents <= 50 || cents >= 100000000) continue;
  1001→        const usd = cents / 100;
  1002→        if (usd < 0.50 || usd > 100000) continue;
  1003→        candidates.push({ usd, currency: cur });
  1004→      }
  1005→    }
  1006→
  1007→    if (candidates.length > 0) {
  1008→      // Pick the LOWEST price — the user is always viewing the cheapest
  1009→      // variant of the actual product, not the cross-sells.
  1010→      candidates.sort((a, b) => a.usd - b.usd);
  1011→      const best = candidates[0];
  1012→      price = best.usd;
  1013→      currency = best.currency;
  1014→      if (best.marketUsd && best.marketUsd > best.usd) {
  1015→        originalPrice = best.marketUsd;
  1016→      }
  1017→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1018→    }
  1019→  }
  1020→
  1021→  // 2d. Embedded JSON price fields (broad regex sweep)
  1022→  if (!price) {
  1023→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1024→    const found: { value: number; field: string }[] = [];
  1025→    for (const f of fields) {
  1026→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1027→      for (const m of html.matchAll(re)) {
  1028→        const v = parseFloat(m[1]);
  1029→        if (v > 0 && v < 100000) {
  1030→          found.push({ value: v, field: f });
  1031→        }
  1032→      }
  1033→    }
  1034→    if (found.length > 0) {
  1035→      // Sort and pick lowest (likely sale price)
  1036→      found.sort((a, b) => a.value - b.value);
  1037→      const best = found[0];
  1038→      // Heuristic: if value > 100, probably in cents → divide by 100
  1039→      price = best.value > 100 ? best.value / 100 : best.value;
  1040→      priceSource = `embedded-${best.field}`;
  1041→    }
  1042→  }
  1043→
  1044→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1045→  if (!price) {
  1046→    const text = html.replace(/<[^>]*>/g, " ");
  1047→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1048→    const prices: { value: number; currency: string }[] = [];
  1049→    for (const m of matches) {
  1050→      const raw = m[1].replace(/,/g, "");
  1051→      const v = parseFloat(raw);
  1052→      if (v > 1.5 && v < 5000) {
  1053→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1054→        prices.push({ value: v, currency: cur });
  1055→      }
  1056→    }
  1057→    if (prices.length > 0) {
  1058→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1059→      const sorted = prices.sort((a, b) => a.value - b.value);
  1060→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1061→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1062→      if (filtered.length > 0) {
  1063→        price = filtered[0].value;
  1064→        currency = filtered[0].currency;
  1065→        priceSource = "text";
  1066→      }
  1067→    }
  1068→  }
  1069→
  1070→  return {
  1071→    price,
  1072→    currency,
  1073→    productName,
  1074→    productDescription,
  1075→    canonicalUrl: canonical,
  1076→    originalPrice,
  1077→    image: ogImage,
  1078→    source: priceSource || (productName ? "og-only" : "none"),
  1079→  };
  1080→}

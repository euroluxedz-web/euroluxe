   210→/* ───────────────────────────────────────────────────────────────────
   211→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   212→ *
   213→ * Temu share URLs include everything we need:
   214→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   215→ *   - top_gallery_url    = URL-encoded product image URL
   216→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   217→ *
   218→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   219→ * ─────────────────────────────────────────────────────────────────── */
   220→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   221→  let parsed: URL;
   222→  try {
   223→    parsed = new URL(originalUrl);
   224→  } catch {
   225→    return null;
   226→  }
   227→
   228→  let price: number | null = null;
   229→  let currency = "USD";
   230→  let priceSource = "";
   231→
   232→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   233→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   234→  if (hint) {
   235→    try {
   236→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   237→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   238→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   239→      if (cents > 0 && cents < 10000000) {
   240→        const usd = cents / 100;
   241→        if (usd >= 0.01 && usd < 100000) {
   242→          price = usd;
   243→          currency = "USD";
   244→          priceSource = "url-hint";
   245→        }
   246→      }
   247→    } catch { /* not valid base64 */ }
   248→  }
   249→
   250→  // Product image from top_gallery_url
   251→  let image: string | null = null;
   252→  const topGallery = parsed.searchParams.get("top_gallery_url");
   253→  if (topGallery) {
   254→    try {
   255→      // Validate it's a real URL
   256→      const imgUrl = new URL(topGallery);
   257→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   258→        image = topGallery;
   259→      }
   260→    } catch { /* not a valid URL */ }
   261→  }
   262→
   263→  // Product name from URL slug
   264→  let productName: string | null = null;
   265→  const segments = parsed.pathname.split("/").filter(Boolean);
   266→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   267→  const nameFromSlug = slug
   268→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   269→    .replace(/\.html?$/i, "")
   270→    .replace(/-/g, " ")
   271→    .replace(/\s+/g, " ")
   272→    .trim();
   273→  if (nameFromSlug && nameFromSlug.length > 3) {
   274→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   275→  }
   276→
   277→  // If we got at least the price (the most important piece), return success
   278→  if (price) {
   279→    return {
   280→      price,
   281→      currency,
   282→      productName,
   283→      productDescription: null,
   284→      canonicalUrl: null,
   285→      originalPrice: null,
   286→      image,
   287→      source: priceSource,
   288→      antiBotDetected: false,
   289→    };
   290→  }
   291→
   292→  // No price from URL — return null so caller tries HTML-based strategies
   293→  return null;
   294→}

   352→async function fetchPriceWithWebSearch(
   353→  goodsId: string | null,
   354→  itemId: string | null,
   355→  originalUrl: string,
   356→): Promise<TemuProductData | null> {
   357→  try {
   358→    const zai = await ZAI.create();
   359→
   360→    // Build search query — try different strategies
   361→    let searchQuery = "";
   362→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   363→      // Item ID like TV10922608 — search Temu specifically
   364→      searchQuery = `site:temu.com "${itemId}"`;
   365→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   366→      // Numeric goods_id — search with the -g- pattern
   367→      searchQuery = `site:temu.com "g-${goodsId}"`;
   368→    } else {
   369→      // Fallback: search with the URL
   370→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   371→      searchQuery = `site:temu.com ${cleanUrl}`;
   372→    }
   373→
   374→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   375→
   376→    // Step 1: Web search to find the product
   377→    const searchResults = await (zai as any).invokeFunction("web_search", {
   378→      query: searchQuery,
   379→      num: 5,
   380→    });
   381→
   382→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   383→      console.log("[WebSearch] No results found, trying broader search...");
   384→
   385→      // Broader search without site: restriction
   386→      const broadQuery = itemId
   387→        ? `temu ${itemId} price`
   388→        : goodsId
   389→          ? `temu ${goodsId} price`
   390→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   391→
   392→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   393→      const broadResults = await (zai as any).invokeFunction("web_search", {
   394→        query: broadQuery,
   395→        num: 5,
   396→      });
   397→
   398→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   399→        console.log("[WebSearch] No results from broader search either");
   400→        return null;
   401→      }
   402→
   403→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   404→    }
   405→
   406→    // Step 2: Try to parse price from initial search results
   407→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   408→    if (initialPrice?.price && initialPrice.price > 0) {
   409→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   410→      return initialPrice;
   411→    }
   412→
   413→    // Step 3: If initial search found product names but no prices,
   414→    // try a more targeted search with the product name to find prices
   415→    const productNameFromResults = searchResults
   416→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   417→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   418→      .find((n: string) => n && n.length > 10);
   419→
   420→    if (productNameFromResults) {
   421→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   422→      try {
   423→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   424→        const nameResults = await (zai as any).invokeFunction("web_search", {
   425→          query: nameQuery,
   426→          num: 5,
   427→        });
   428→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   429→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   430→          if (namePrice?.price && namePrice.price > 0) {
   431→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   432→            return namePrice;
   433→          }
   434→          // Merge results and try LLM on combined results
   435→          const combinedResults = [...searchResults, ...nameResults];
   436→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   437→        }
   438→      } catch (err) {
   439→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   440→      }
   441→    }
   442→
   443→    // Step 4: Use LLM on whatever results we have
   444→    return extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   445→  } catch (err) {
   446→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   447→    return null;
   448→  }
   449→}

   451→async function extractPriceFromSearchResults(
   452→  zai: any,
   453→  searchResults: any[],
   454→  goodsId: string | null,
   455→  itemId: string | null,
   456→): Promise<TemuProductData | null> {

   460→    console.log(`[WebSearch] ✓ Found price from snippet: ${snippetPrice.price} USD (source: ${snippetPrice.source})`);
   461→    return snippetPrice;
   462→  }
   463→
   464→  // Step 3: If snippets don't have a clear price, use LLM to extract
   465→  console.log("[WebSearch] Snippets don't have clear price, using LLM...");
   466→
   467→  const searchContext = searchResults
   468→    .slice(0, 5)
   469→    .map(
   470→      (r: any, i: number) =>
   471→        `${i + 1}. ${r.name || "No title"}\n   URL: ${r.url}\n   Snippet: ${r.snippet || "No snippet"}`,

   483→            'If the price is in a non-USD currency, CONVERT it to USD using approximate exchange rates. ' +
   484→            'Common conversions: MUR→USD ÷47, OMR→USD ×2.60, BHD→USD ×2.65, PKR→USD ÷278, EUR→USD ×1.08, GBP→USD ×1.27, SAR→USD ×0.27, AED→USD ×0.27. ' +

   487→            'IMPORTANT: Do NOT estimate or guess the price. If the search results do not contain a clear price for this specific product, return null. ' +
   488→            'Do NOT confuse "delivery guarantee" prices, "credit for delay" amounts, or "30% OFF" discount percentages with the product price. ' +
   489→            'Do NOT use prices from unrelated products. Only extract the price that clearly corresponds to the product with the given goods_id or Item ID.',
   490→        },
   491→        {
   492→          role: "user",
   493→          content:
   494→            `Product goods_id: ${goodsId || "unknown"}\n` +
   495→            `Item ID: ${itemId || "unknown"}\n\n` +

   500→    });
   501→
   502→    const aiResponse = completion.choices?.[0]?.message?.content || "";
   503→    console.log("[WebSearch] LLM response:", aiResponse.slice(0, 300));
   504→

   509→      return null;
   510→    }
   511→
   512→    const parsed = JSON.parse(jsonMatch[0]);
   513→    const priceUSD = typeof parsed.price_usd === "number"
   514→      ? parsed.price_usd
   515→      : parseFloat(String(parsed.price_usd));
   516→
   517→    if (priceUSD && priceUSD > 0 && priceUSD < 100000) {
   518→      console.log(`[WebSearch] ✓ LLM found price: $${priceUSD} (confidence: ${parsed.confidence})`);
   519→      return {
   520→        price: priceUSD,
   521→        currency: "USD",
   522→        productName: parsed.name || null,
   523→        productDescription: null,
   524→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   525→        originalPrice: null,
   526→        image: null, // Image will come from share URL params
   527→        source: `web-search-llm(${parsed.confidence})`,
   528→        antiBotDetected: false,
   529→      };
   530→    }
   531→
   532→    // LLM couldn't find price either — return what we can
   533→    if (parsed.name) {
   534→      return {
   535→        price: null,
   536→        currency: "USD",
   537→        productName: parsed.name,
   538→        productDescription: null,
   539→        canonicalUrl: goodsId ? `https://www.temu.com/-g-${goodsId}.html` : null,
   540→        originalPrice: null,
   541→        image: null,
   542→        source: "web-search-llm(no-price)",
   543→        antiBotDetected: false,
   544→      };
   545→    }
   546→
   547→    return null;
   548→  } catch (err) {
   549→    console.error("[WebSearch] LLM error:", String(err).slice(0, 150));
   550→    return null;
   551→  }

   554→/**
   555→ * Try to parse prices directly from search result snippets.
   556→ * This avoids the cost of an LLM call and is often sufficient
   557→ * because Google includes price info in Temu search snippets.
   558→ */
   559→function parsePriceFromSnippets(
   560→  results: any[],
   561→  goodsId: string | null,
   562→): TemuProductData | null {
   563→  // Look for Temu results that have price info in snippets
   564→  for (const result of results) {
   565→    if (!result.url?.includes("temu.com")) continue;
   566→    if (!result.snippet) continue;
   567→
   568→    const snippet = result.snippet;
   569→
   570→    // Extract the goods_id from the URL if we don't have it
   571→    let foundGoodsId = goodsId;
   572→    if (!foundGoodsId) {
   573→      const gMatch = result.url.match(/-g-(\d{10,})/);
   574→      if (gMatch) foundGoodsId = gMatch[1];
   575→    }
   576→
   577→    // Extract product name from search result
   578→    const productName = result.name
   579→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   580→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   581→      .trim() || null;
   582→
   583→    // Determine the local currency from the Temu URL locale prefix
   584→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   585→    let localCurrency = "USD";
   586→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   587→    if (localeMatch) {
   588→      const locale = localeMatch[1].toLowerCase();
   589→      const localeToCurrency: Record<string, string> = {
   590→        mu: "MUR",       // Mauritius
   591→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   592→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   593→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   594→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   595→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   596→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   597→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   598→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   599→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   600→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   601→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   602→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   603→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   604→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   605→        np: "NPR", "np-en": "NPR", // Nepal
   606→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   607→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   608→        ph: "PHP", "ph-en": "PHP", // Philippines
   609→        br: "BRL", "br-pt": "BRL", // Brazil
   610→        mx: "MXN", "mx-es": "MXN", // Mexico
   611→      };
   612→      const found = localeToCurrency[locale];
   613→      if (found) localCurrency = found;
   614→    }
   615→
   616→    // Try to find a price in the snippet
   617→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   618→    // Common patterns:
   619→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   620→    //   "$6.99"                  → USD
   621→    //   "OMR 1.200"             → Omani Rial
   622→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   623→    //   "€5.99"                  → EUR
   624→    //   "£4.99"                  → GBP
   625→
   626→    // Determine if this locale uses European comma format (comma = decimal separator)
   627→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   628→    const europeanCommaLocales = new Set([
   629→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",
   630→      "ma-fr", "tn-fr", // French-speaking locales
   631→    ]);
   632→    const usesEuropeanComma = europeanCommaLocales.has(localCurrency === "USD" && localeMatch ? localeMatch[1].toLowerCase() : "") ||
   633→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   634→
   635→    // Helper: parse price string respecting locale format
   636→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   637→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   638→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   639→      if (isEuropean) {
   640→        // European: remove dots (thousands), replace comma with dot (decimal)
   641→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   642→      } else {
   643→        // English: remove commas (thousands)
   644→        return parseFloat(priceStr.replace(/,/g, ""));
   645→      }
   646→    };
   647→
   648→    // First, try explicit currency patterns (most reliable)
   649→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   650→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   651→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   652→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   653→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   654→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   655→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   656→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   657→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   658→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   659→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   660→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   661→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   662→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   663→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   664→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   665→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   666→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   667→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   668→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   669→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   670→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   671→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   672→    ];
   673→
   674→    for (const { pattern, currency } of explicitPatterns) {
   675→      const match = snippet.match(pattern);
   676→      if (match) {
   677→        // For USD prices on European locale pages (like Ecuador), use comma-as-decimal parsing
   678→        const isEuroPrice = currency === "USD" && usesEuropeanComma;
   679→        const localPrice = parsePriceString(match[1], isEuroPrice);
   680→        if (localPrice > 0 && localPrice < 10000000) {
   681→          const rate = CURRENCY_TO_USD[currency];
   682→          if (rate) {
   683→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   684→            if (usdPrice >= 0.10 && usdPrice <= 5000) {
   685→              console.log(`[WebSearch] Found explicit price: ${localPrice} ${currency} (European format: ${isEuroPrice}) = $${usdPrice} USD`);
   686→              return {
   687→                price: usdPrice,
   688→                currency: "USD",
   689→                productName,
   690→                productDescription: null,
   691→                canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   692→                originalPrice: null,
   693→                image: null,
   694→                source: `web-search-snippet(${currency})`,
   695→                antiBotDetected: false,
   696→              };
   697→            }
   698→          }
   699→        }
   700→      }
   701→    }
   702→
   703→    // Also try European comma format pattern for $ prices (e.g., "$5,22" on Ecuador page)
   704→    // This catches prices where the comma is used as decimal separator
   705→    if (usesEuropeanComma) {
   706→      const euroPattern = /\$\s*(\d+,\d{1,2})\b/;
   707→      const euroMatch = snippet.match(euroPattern);
   708→      if (euroMatch) {
   709→        const localPrice = parseFloat(euroMatch[1].replace(",", "."));
   710→        if (localPrice > 0 && localPrice < 10000000) {
   711→          const usdPrice = Math.round(localPrice * 100) / 100; // Already USD
   712→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   713→            console.log(`[WebSearch] Found European format price: ${euroMatch[0]} = $${usdPrice} USD`);
   714→            return {
   715→              price: usdPrice,
   716→              currency: "USD",
   717→              productName,
   718→              productDescription: null,
   719→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   720→              originalPrice: null,
   721→              image: null,
   722→              source: "web-search-snippet(USD-european)",
   723→              antiBotDetected: false,
   724→            };
   725→          }
   726→        }
   727→      }
   728→    }
   729→
   730→    // Second, try "Rs" pattern — use the URL locale to determine the currency
   731→    // Rs could be MUR (Mauritius), PKR (Pakistan), LKR (Sri Lanka), NPR (Nepal)
   732→    const rsMatch = snippet.match(/Rs\.?\s*([\d,]+(?:\.\d{1,2})?)/);
   733→    if (rsMatch) {
   734→      const localPrice = parseFloat(rsMatch[1].replace(/,/g, ""));
   735→      const rsCurrency = localCurrency; // Determined from URL locale above
   736→      if (localPrice > 0 && localPrice < 10000000) {
   737→        const rate = CURRENCY_TO_USD[rsCurrency];
   738→        if (rate) {
   739→          const usdPrice = Math.round(localPrice * rate * 100) / 100;
   740→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   741→            console.log(`[WebSearch] Found Rs price: ${localPrice} ${rsCurrency} (from locale) = $${usdPrice} USD`);
   742→            return {
   743→              price: usdPrice,
   744→              currency: "USD",
   745→              productName,
   746→              productDescription: null,
   747→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   748→              originalPrice: null,
   749→              image: null,
   750→              source: `web-search-snippet(Rs→${rsCurrency})`,
   751→              antiBotDetected: false,
   752→            };
   753→          }
   754→        }
   755→      }
   756→    }
   757→
   758→    // Third, try number followed by Rs (reverse pattern)
   759→    const rsReverseMatch = snippet.match(/([\d,]+(?:\.\d{1,2})?)\s*Rs\.?/);
   760→    if (rsReverseMatch) {
   761→      const localPrice = parseFloat(rsReverseMatch[1].replace(/,/g, ""));
   762→      const rsCurrency = localCurrency;
   763→      if (localPrice > 0 && localPrice < 10000000) {
   764→        const rate = CURRENCY_TO_USD[rsCurrency];
   765→        if (rate) {
   766→          const usdPrice = Math.round(localPrice * rate * 100) / 100;
   767→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   768→            console.log(`[WebSearch] Found Rs price (reverse): ${localPrice} ${rsCurrency} = $${usdPrice} USD`);
   769→            return {
   770→              price: usdPrice,
   771→              currency: "USD",
   772→              productName,
   773→              productDescription: null,
   774→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   775→              originalPrice: null,
   776→              image: null,
   777→              source: `web-search-snippet(Rs→${rsCurrency})`,
   778→              antiBotDetected: false,
   779→            };
   780→          }
   781→        }
   782→      }
   783→    }
   784→  }
   785→
   786→  return null;
   787→}

   945→

   947→    try {

   950→
   951→      const response = await fetch(endpoint.url, {

  1046→async function fetchWithScrapingBee(url: string): Promise<TemuProductData | null> {

  1062→  let lastError: string | null = null;

  1100→          source: isAntiBot ? "scrapingbee-og" : "scrapingbee-full",

  1108→      lastError = String(err).slice(0, 150);
  1109→      console.error(`[ScrapingBee attempt ${attempt}] Error:`, lastError);
  1110→      await new Promise(r => setTimeout(r, 1000));

  1118→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1119→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1120→  // 1. OG meta tags (work even on anti-bot page)
  1121→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1122→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1123→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1124→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1125→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1126→
  1127→  // Decode HTML entities in title
  1128→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1129→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1130→
  1131→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1132→  let price: number | null = null;
  1133→  let currency = "USD";
  1134→  let originalPrice: number | null = null;
  1135→  let priceSource = "";
  1136→
  1137→  // ─────────────────────────────────────────────────────────────
  1138→  // 2-preferred. URL-embedded price hint.
  1139→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1140→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1141→  // → 1.23 USD. This is the most reliable signal because it comes
  1142→  // straight from the share URL the user copied.
  1143→  // ─────────────────────────────────────────────────────────────
  1144→  try {
  1145→    const parsedUrl = new URL(originalUrl);
  1146→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1147→    if (hint) {
  1148→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1149→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1150→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1151→      // Strip non-digits
  1152→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1153→      if (cents > 0 && cents < 10000000) {
  1154→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1155→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1156→        const usd = cents / 100;
  1157→        if (usd >= 0.01 && usd < 100000) {
  1158→          price = usd;
  1159→          currency = "USD";
  1160→          priceSource = "url-hint";
  1161→        }
  1162→      }
  1163→    }
  1164→  } catch { /* not a parseable URL */ }
  1165→
  1166→  // 2a. JSON-LD structured data
  1167→  if (!price) {
  1168→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1169→    for (const match of jsonLdMatches) {
  1170→      try {
  1171→        const data = JSON.parse(match[1]);
  1172→        const schemas = Array.isArray(data) ? data : [data];
  1173→        for (const schema of schemas) {
  1174→          if (schema["@type"] === "Product" && schema.offers) {
  1175→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1176→            for (const offer of offers) {
  1177→              if (offer.price !== undefined) {
  1178→                price = parseFloat(offer.price);
  1179→                currency = offer.priceCurrency || "USD";
  1180→                priceSource = "json-ld";
  1181→                break;
  1182→              }
  1183→            }
  1184→          }
  1185→        }
  1186→      } catch { /* skip */ }
  1187→      if (price) break;
  1188→    }
  1189→  }
  1190→
  1191→  // 2b. OG product:price:amount meta
  1192→  if (!price) {
  1193→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1194→    if (ogPrice) {
  1195→      const p = parseFloat(ogPrice[1]);
  1196→      if (p > 0) {
  1197→        price = p;
  1198→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1199→        currency = ogCur?.[1] || "USD";
  1200→        priceSource = "og-meta";
  1201→      }
  1202→    }
  1203→  }
  1204→
  1205→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1206→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1207→  //     one per "similar product", one per cross-sell, etc.). The previous
  1208→  //     implementation picked the FIRST match, which often returned a
  1209→  //     cross-sell product's price (e.g. $30) instead of the actual
  1210→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1211→  //     pick the LOWEST plausible one, since the actual product the user
  1212→  //     is viewing is almost always the cheapest among the rendered
  1213→  //     priceInfo blocks.
  1214→  if (!price) {
  1215→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1216→    const candidates: Candidate[] = [];
  1217→
  1218→    // priceInfo with explicit priceStr (most reliable)
  1219→    const priceInfosWithStr = [...html.matchAll(
  1220→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1221→    )];
  1222→    for (const pi of priceInfosWithStr) {
  1223→      const cents = parseInt(pi[1]);
  1224→      const cur = pi[2];
  1225→      const priceStr = pi[3];
  1226→      // Skip i18n placeholder strings
  1227→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1228→      if (cents <= 0 || cents >= 100000000) continue;
  1229→      const usd = cents / 100;
  1230→      if (usd < 0.01 || usd > 100000) continue;
  1231→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1232→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1233→      candidates.push({ usd, currency: cur, marketUsd });
  1234→    }
  1235→
  1236→    // priceInfo without priceStr (broader fallback)
  1237→    if (candidates.length === 0) {
  1238→      const priceInfosAll = [...html.matchAll(
  1239→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1240→      )];
  1241→      for (const pi of priceInfosAll) {
  1242→        const cents = parseInt(pi[1]);
  1243→        const cur = pi[2];
  1244→        if (cents <= 50 || cents >= 100000000) continue;
  1245→        const usd = cents / 100;
  1246→        if (usd < 0.50 || usd > 100000) continue;
  1247→        candidates.push({ usd, currency: cur });
  1248→      }
  1249→    }
  1250→
  1251→    if (candidates.length > 0) {
  1252→      // Pick the LOWEST price — the user is always viewing the cheapest
  1253→      // variant of the actual product, not the cross-sells.
  1254→      candidates.sort((a, b) => a.usd - b.usd);
  1255→      const best = candidates[0];
  1256→      price = best.usd;
  1257→      currency = best.currency;
  1258→      if (best.marketUsd && best.marketUsd > best.usd) {
  1259→        originalPrice = best.marketUsd;
  1260→      }
  1261→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1262→    }
  1263→  }
  1264→
  1265→  // 2d. Embedded JSON price fields (broad regex sweep)
  1266→  if (!price) {
  1267→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1268→    const found: { value: number; field: string }[] = [];
  1269→    for (const f of fields) {
  1270→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1271→      for (const m of html.matchAll(re)) {
  1272→        const v = parseFloat(m[1]);
  1273→        if (v > 0 && v < 100000) {
  1274→          found.push({ value: v, field: f });
  1275→        }
  1276→      }
  1277→    }
  1278→    if (found.length > 0) {
  1279→      // Sort and pick lowest (likely sale price)
  1280→      found.sort((a, b) => a.value - b.value);
  1281→      const best = found[0];
  1282→      // Heuristic: if value > 100, probably in cents → divide by 100
  1283→      price = best.value > 100 ? best.value / 100 : best.value;
  1284→      priceSource = `embedded-${best.field}`;
  1285→    }
  1286→  }
  1287→
  1288→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1289→  if (!price) {
  1290→    const text = html.replace(/<[^>]*>/g, " ");
  1291→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1292→    const prices: { value: number; currency: string }[] = [];
  1293→    for (const m of matches) {
  1294→      const raw = m[1].replace(/,/g, "");
  1295→      const v = parseFloat(raw);
  1296→      if (v > 1.5 && v < 5000) {
  1297→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1298→        prices.push({ value: v, currency: cur });
  1299→      }
  1300→    }
  1301→    if (prices.length > 0) {
  1302→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1303→      const sorted = prices.sort((a, b) => a.value - b.value);
  1304→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1305→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1306→      if (filtered.length > 0) {
  1307→        price = filtered[0].value;
  1308→        currency = filtered[0].currency;
  1309→        priceSource = "text";
  1310→      }
  1311→    }
  1312→  }
  1313→
  1314→  return {
  1315→    price,
  1316→    currency,
  1317→    productName,
  1318→    productDescription,
  1319→    canonicalUrl: canonical,
  1320→    originalPrice,
  1321→    image: ogImage,
  1322→    source: priceSource || (productName ? "og-only" : "none"),
  1323→  };
  1324→}

  1432→        // The web search strategy will handle finding the product
  1433→        console.log(`[Item ID] Detected Item ID format: ${trimmedUrl}`);
  1434→        goodsId = ""; // No goods_id yet — will be found by web search
  1435→        finalUrl = trimmedUrl; // Pass the raw Item ID, web search will handle it
  1436→      } else {
  1437→        // Numeric goods_id — construct the URL
  1438→        goodsId = trimmedUrl;
  1439→        finalUrl = `https://www.temu.com/-g-${trimmedUrl}.html?_x_sessn=us&currency=USD`;
  1440→      }
  1441→    } else {
  1442→      try {

  1444→

  1446→        // When a user taps "Partager" in the Temu app, they get a short

  1455→            const ctrl = new AbortController();
  1456→            const timer = setTimeout(() => ctrl.abort(), 10000);
  1457→            const shareRes = await fetch(finalUrl, {
  1458→              signal: ctrl.signal,
  1459→              redirect: "follow",
  1460→              headers: {
  1461→                "User-Agent":
  1462→                  "Mozilla/5.0 (Linux; Android 14; Pixel 8) AppleWebKit/537.36 " +

  1464→                Accept: "text/html,application/xhtml+xml",
  1465→              },
  1466→            });
  1467→            clearTimeout(timer);
  1468→
  1469→            // The redirected URL is in the response's url property
  1470→            const resolvedUrl = shareRes.url || shareRes.headers.get("location") || "";

  1472→              console.log(`[Share URL] Resolved to: ${resolvedUrl}`);
  1473→              finalUrl = resolvedUrl;
  1474→            } else {
  1475→              // Sometimes fetch with redirect:"follow" merges all redirects
  1476→              // and shareRes.url gives the final URL. If not, try manual
  1477→              // redirect by making a HEAD request with redirect:"manual".
  1478→              try {
  1479→                const headCtrl = new AbortController();
  1480→                const headTimer = setTimeout(() => headCtrl.abort(), 8000);
  1481→                const headRes = await fetch(finalUrl, {
  1482→                  method: "HEAD",
  1483→                  signal: headCtrl.signal,
  1484→                  redirect: "manual",
  1485→                  headers: {
  1486→                    "User-Agent":
  1487→                      "Mozilla/5.0 (Linux; Android 14; Pixel 8) AppleWebKit/537.36 " +
  1488→                      "(KHTML, like Gecko) Chrome/131.0.0.0 Mobile Safari/537.36",
  1489→                  },
  1490→                });
  1491→                clearTimeout(headTimer);
  1492→                const location = headRes.headers.get("location");
  1493→                if (location) {
  1494→                  console.log(`[Share URL] Manual redirect to: ${location}`);
  1495→                  finalUrl = location;
  1496→                }
  1497→              } catch { /* manual redirect failed */ }
  1498→            }
  1499→          } catch (err) {
  1500→            console.log("[Share URL] Resolution failed:", String(err).slice(0, 100));
  1501→          }
  1502→        }
  1503→
  1504→        // Re-parse with potentially-resolved URL
  1505→        const resolved = new URL(finalUrl);
  1506→
  1507→        // Extract product image from share URL params (available even without price)
  1508→        const topGallery = resolved.searchParams.get("top_gallery_url");
  1509→        const shareImg = resolved.searchParams.get("share_img");
  1510→        if (topGallery) shareImage = topGallery;
  1511→        else if (shareImg) shareImage = shareImg;
  1512→

  1514→        const gMatch = resolved.pathname.match(/-g-([a-zA-Z0-9]+)/);
  1515→        if (gMatch) {
  1516→          goodsId = gMatch[1];
  1517→        } else {
  1518→          const gidParam = resolved.searchParams.get("goods_id");
  1519→          if (gidParam) {
  1520→            goodsId = gidParam;
  1521→          } else {
  1522→            const numMatch = resolved.pathname.match(/(\d{10,})/);
  1523→            if (numMatch) goodsId = numMatch[1];
  1524→          }
  1525→        }
  1526→
  1527→        // ── Reconstruct clean URL for reliable price extraction ──
  1528→        // When a share.temu.com link resolves to a localized URL (e.g. /dz-en/),
  1529→        // the page shows prices in the LOCAL currency (DZD), not USD.
  1530→        // Also, the _oak_rec_ext_1 param (if present) encodes the price in
  1531→        // the local currency, which Strategy 0 would wrongly treat as USD.
  1532→        //
  1533→        // FIX: Reconstruct the URL with ONLY the goods_id and US session params.
  1534→        // This removes _oak_rec_ext_1 (wrong currency) and the locale prefix
  1535→        // (which would show prices in local currency), ensuring all strategies
  1536→        // get the US version of the page with USD prices.
  1537→        if (goodsId) {
  1538→          finalUrl = `https://www.temu.com/-g-${goodsId}.html?_x_sessn=us&currency=USD`;
  1539→          console.log(`[URL] Reconstructed clean URL (no locale, no _oak_rec_ext_1): ${finalUrl}`);

  1542→          resolved.searchParams.delete("_oak_rec_ext_1");
  1543→          if (!resolved.searchParams.has("_x_sessn")) {
  1544→            resolved.searchParams.set("_x_sessn", "us");
  1545→            resolved.searchParams.set("currency", "USD");
  1546→          }
  1547→          // Remove locale prefix like /dz-en/ from pathname
  1548→          resolved.pathname = resolved.pathname.replace(/^\/[a-z]{2}-[a-z]{2}\//i, "/");
  1549→          finalUrl = resolved.toString();
  1550→          console.log(`[URL] Cleaned URL (no _oak_rec_ext_1, no locale): ${finalUrl}`);
  1551→        }
  1552→      } catch {
  1553→        return NextResponse.json(
  1554→          {
  1555→            success: false,
  1556→            error: "Format de lien invalide / صيغة الرابط غير صالحة",
  1557→            allowManual: true,
  1558→          },
  1559→          { status: 400 }
  1560→        );
  1561→      }
  1562→    }
  1563→
  1564→    const urlProductName = extractProductNameFromUrl(finalUrl);
  1565→
  1566→    // ────────────────────────────────────────────────────────────────
  1567→    // STRATEGY CHAIN (most reliable first):
  1568→    //   0. URL params       (FREE, 0 network) — _oak_rec_ext_1 + top_gallery_url + slug
  1569→    //   0-C. Page Reader+LLM (ZAI SDK)        — reads the rendered product page
  1570→    //   0-AI. Web Search+LLM (ZAI SDK)        — searches Google-indexed Temu pages
  1571→    //   0b. Temu BG API     (FREE)            — /bg/goods/api endpoint with goods_id
  1572→    //   1. Direct fetch     (FREE)            — realistic browser headers
  1573→    //   2. AllOrigins       (FREE)            — public CORS proxy
  1574→    //   3. CorsProxy.io     (FREE)            — another public CORS proxy
  1575→    //   4. ScrapingBee      (PAID, last)      — only if SCRAPINGBEE_API_KEY set
  1576→    // ────────────────────────────────────────────────────────────────
  1577→
  1578→    // Determine if this is an Item ID (like TV10922608) vs a goods_id (like 601101613236742)
  1579→    const isItemId = isTemuProductId && /^[A-Z]{2}\d+/i.test(trimmedUrl);
  1580→
  1581→    // Strategy 0: URL params (the cheapest & most reliable for share URLs with _oak_rec_ext_1)
  1582→    console.log("[Strategy 0] Trying URL params extraction (FREE)...");
  1583→    const urlResult = extractFromUrlParams(finalUrl);
  1584→    if (urlResult?.price && urlResult.price > 0) {
  1585→      console.log(`[Strategy 0] ✓ Got price $${urlResult.price} from URL hint`);
  1586→      return await buildSuccessResponse(urlResult, urlProductName, shareImage, goodsId);
  1587→    }
  1588→
  1589→    // Strategy 0-C: Page Reader + LLM (reads the rendered Temu product page)
  1590→    // This is the most reliable strategy for share URLs because:
  1591→    // - page_reader renders JavaScript and gets the full page content
  1592→    // - The LLM can extract the product name and price from the rendered HTML
  1593→    // - For share URLs, it follows the redirect and reads the product page
  1594→    // - For Item IDs, it reads the Temu search page
  1595→    {
  1596→      const prGoodsId = isItemId ? null : (goodsId || null);
  1597→      const prItemId = isItemId ? trimmedUrl : null;
  1598→      const prShareUrl = originalShareUrl; // Only set for share.temu.com URLs
  1599→      console.log(`[Strategy 0-C] Trying Page Reader (goodsId=${prGoodsId}, itemId=${prItemId}, shareUrl=${prShareUrl ? "yes" : "no"})...`);
  1600→      const pageReaderResult = await fetchPriceWithPageReader(prGoodsId, prItemId, prShareUrl);
  1601→      if (pageReaderResult) {
  1602→        if (pageReaderResult.price && pageReaderResult.price > 0) {
  1603→          console.log(`[Strategy 0-C] ✓ Got price $${pageReaderResult.price} from page reader`);
  1604→          const bestImage = shareImage || pageReaderResult.image;
  1605→          const bestGoodsId = goodsId || pageReaderResult.canonicalUrl?.match(/-g-(\d{10,})/)?.[1] || "";
  1606→          return await buildSuccessResponse(pageReaderResult, urlProductName, bestImage, bestGoodsId);
  1607→        }
  1608→        // Page reader found product name but no price — still useful
  1609→        if (pageReaderResult.productName && !urlProductName) {
  1610→          // Continue to next strategies — we have the product name
  1611→        }
  1612→        // If page reader found a goods_id from Item ID search, save it
  1613→        const foundGoodsId = pageReaderResult.canonicalUrl?.match(/-g-(\d{10,})/)?.[1];
  1614→        if (foundGoodsId && !goodsId) {
  1615→          goodsId = foundGoodsId;
  1616→          console.log(`[Strategy 0-C] Found goods_id from page reader: ${goodsId}`);
  1617→        }
  1618→      }
  1619→    }
  1620→
  1621→    // Strategy 0-AI: ZAI Web Search + LLM (most reliable for Temu since anti-bot blocks all scraping)
  1622→    // This searches Google-indexed Temu pages and extracts prices from search snippets or uses LLM.
  1623→    // It's especially effective for:
  1624→    //   - Share URLs: we have the goods_id, search finds the product on various Temu locales
  1625→    //   - Item IDs (TV10922608): search finds the product page with the goods_id

  1627→      const searchGoodsId = isItemId ? null : (goodsId || null);
  1628→      const searchItemId = isItemId ? trimmedUrl : null;
  1629→      console.log(`[Strategy 0-AI] Trying ZAI Web Search (goodsId=${searchGoodsId}, itemId=${searchItemId})...`);
  1630→      const webSearchResult = await fetchPriceWithWebSearch(searchGoodsId, searchItemId, finalUrl);
  1631→      if (webSearchResult) {
  1632→        if (webSearchResult.price && webSearchResult.price > 0) {
  1633→          console.log(`[Strategy 0-AI] ✓ Got price $${webSearchResult.price} from web search`);
  1634→          // Use the image from share URL params (more reliable) if available
  1635→          const bestImage = shareImage || webSearchResult.image;
  1636→          // Use the goods_id from the web search result if we didn't have one (e.g. from Item ID search)
  1637→          const bestGoodsId = goodsId || webSearchResult.canonicalUrl?.match(/-g-(\d{10,})/)?.[1] || "";
  1638→          return await buildSuccessResponse(webSearchResult, urlProductName, bestImage, bestGoodsId);
  1639→        }
  1640→        // Web search found the product but no price — still useful for product name and image
  1641→        if (webSearchResult.productName) {
  1642→          // If we also found a goods_id from the search, try the BG API with it
  1643→          const foundGoodsId = webSearchResult.canonicalUrl?.match(/-g-(\d{10,})/)?.[1];
  1644→          if (foundGoodsId && !goodsId) {
  1645→            goodsId = foundGoodsId;
  1646→            console.log(`[Strategy 0-AI] Found goods_id from web search: ${goodsId}`);
  1647→          }
  1648→          // Continue to next strategies — we have the product name
  1649→        }
  1650→      }
  1651→    }
  1652→
  1653→    // Strategy 0b: Temu BG API (try fetching product JSON directly)
  1654→    // When we have a goods_id, we can call Temu's internal API endpoint
  1655→    // which returns product details as JSON, including price, name, and image.
  1656→    // This is more reliable than scraping HTML pages.
  1657→    if (goodsId && /^\d{10,}$/.test(goodsId)) {
  1658→      console.log(`[Strategy 0b] Trying Temu BG API with goods_id=${goodsId}...`);
  1659→      const apiResult = await fetchTemuBgApi(goodsId);
  1660→      if (apiResult) {
  1661→        if (apiResult.price && apiResult.price > 0) {
  1662→          console.log(`[Strategy 0b] ✓ Got price $${apiResult.price} from Temu BG API`);
  1663→          return await buildSuccessResponse(apiResult, urlProductName, shareImage, goodsId);
  1664→        }
  1665→        if (apiResult.productName) {
  1666→          return NextResponse.json({
  1667→            success: true,
  1668→            price: null,
  1669→            requiresManualPrice: true,
  1670→            productName: cleanProductName(apiResult.productName) || urlProductName,
  1671→            productDescription: apiResult.productDescription,
  1672→            productImage: apiResult.image || shareImage,
  1673→            productUrl: apiResult.canonicalUrl || finalUrl,
  1674→            itemId: goodsId,
  1675→            source: apiResult.source,
  1676→            message: "Produit trouvé. Veuillez saisir le prix affiché على Temu. / تم العثور على المنتج. يرجى إدخال السعر المعروض على Temu.",
  1677→          });
  1678→        }
  1679→      }
  1680→    }
  1681→
  1682→    // Strategies 1-3: free HTTP-based extraction
  1683→    const freeStrategies = [
  1684→      { name: "Direct", fn: fetchDirect },
  1685→      { name: "AllOrigins", fn: fetchViaAllOrigins },
  1686→      { name: "CorsProxy", fn: fetchViaCorsProxy },
  1687→    ];
  1688→
  1689→    for (const strat of freeStrategies) {
  1690→      console.log(`[Strategy] Trying ${strat.name} (FREE)...`);
  1691→      const result = await strat.fn(finalUrl);
  1692→      if (result) {
  1693→        // If we got a price, return full success
  1694→        if (result.price && result.price > 0) {
  1695→          console.log(`[Strategy ${strat.name}] ✓ Got price $${result.price}`);
  1696→          return await buildSuccessResponse(result, urlProductName, shareImage, goodsId);
  1697→        }
  1698→        // If we got a product name (from OG), prompt for manual price entry
  1699→        if (result.productName) {
  1700→          return NextResponse.json({
  1701→            success: true,
  1702→            price: null,
  1703→            requiresManualPrice: true,
  1704→            productName: cleanProductName(result.productName) || urlProductName,
  1705→            productDescription: result.productDescription,
  1706→            productImage: result.image || shareImage,
  1707→            productUrl: result.canonicalUrl || finalUrl,
  1708→            itemId: goodsId || undefined,

  1714→          });

  1716→      }

  1718→
  1719→    // Strategy 4: ScrapingBee (paid, last resort — only if env var set)
  1720→    if (process.env.SCRAPINGBEE_API_KEY) {

  1722→      const result = await fetchWithScrapingBee(finalUrl);

  1735→            productUrl: result.canonicalUrl || finalUrl,

  1741→              : "Produit trouvé. Veuillez saisir le prix affiché sur Temu. / تم العثور على المنتج. يرجى إدخال السعر المعروض على Temu.",
  1742→          });

  1747→    // ─── All strategies failed ───

  1751→      allowManual: true,
  1752→      productName: urlProductName,

  1756→    });
  1757→  } catch (error) {

  1760→      {
  1761→        success: false,
  1762→        error:
  1763→          "Une erreur est survenue. Veuillez entrer le prix manuellement. / حدث خطأ. يرجى إدخال السعر يدويًا.",
  1764→        allowManual: true,

  1771→/* ─── Build success response with Algeria pricing ─── */
  1772→async function buildSuccessResponse(
  1773→  result: TemuProductData,
  1774→  urlProductName: string | null,
  1775→  shareImage: string | null = null,
  1776→  itemId: string | null = null
  1777→) {
  1778→  let priceUSD = result.price!;
  1779→
  1780→  // Convert to USD if in different currency
  1781→  if (result.currency?.toUpperCase() !== "USD") {
  1782→    const rates: Record<string, number> = {
  1783→      EUR: 1.08,
  1784→      GBP: 1.27,
  1785→      CNY: 0.14,
  1786→      DZD: 0.0075,
  1787→    };
  1788→    const rate = rates[result.currency.toUpperCase()];
  1789→    if (rate) priceUSD = result.price! * rate;
  1790→  }
  1791→
  1792→  // Ensure we have a fresh exchange rate
  1793→  await getUsdToDzdRate();
  1794→
  1795→  // Calculate Algeria price with all fees
  1796→  const breakdown = calculateAlgeriaPrice(priceUSD);
  1797→
  1798→  return NextResponse.json({
  1799→    success: true,
  1800→    price: breakdown.basePriceUSD,
  1801→    dzd: breakdown.totalDZD,
  1802→    breakdown,
  1803→    productName: cleanProductName(result.productName) || urlProductName,
  1804→    productDescription: result.productDescription,
  1805→    productImage: result.image || shareImage,
  1806→    productUrl: result.canonicalUrl,
  1807→    originalPrice: result.originalPrice ? Math.round(result.originalPrice * 100) / 100 : null,
  1808→    estimated: false,
  1809→    manual: false,
  1810→    source: result.source,
  1811→    itemId: itemId || undefined,
  1812→  });
  1813→}

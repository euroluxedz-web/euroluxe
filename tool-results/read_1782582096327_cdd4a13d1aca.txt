    19→    19→/* ───────────────────────────────────────────────────────────────────
    20→    20→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
    21→    21→ *
    22→    22→ * Temu share URLs include everything we need:
    23→    23→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
    24→    24→ *   - top_gallery_url    = URL-encoded product image URL
    25→    25→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
    26→    26→ *
    27→    27→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
    28→    28→ * ─────────────────────────────────────────────────────────────────── */
    29→    29→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
    30→    30→  let parsed: URL;
    31→    31→  try {
    32→    32→    parsed = new URL(originalUrl);
    33→    33→  } catch {
    34→    34→    return null;
    35→    35→  }
    36→    36→
    37→    37→  let price: number | null = null;
    38→    38→  let currency = "USD";
    39→    39→  let priceSource = "";
    40→    40→
    41→    41→  // Price from _oak_rec_ext_1 (base64-encoded cents)
    42→    42→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
    43→    43→  if (hint) {
    44→    44→    try {
    45→    45→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
    46→    46→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
    47→    47→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
    48→    48→      if (cents > 0 && cents < 10000000) {
    49→    49→        const usd = cents / 100;
    50→    50→        if (usd >= 0.01 && usd < 100000) {
    51→    51→          price = usd;
    52→    52→          currency = "USD";
    53→    53→          priceSource = "url-hint";
    54→    54→        }
    55→    55→      }
    56→    56→    } catch { /* not valid base64 */ }
    57→    57→  }
    58→    58→
    59→    59→  // Product image from top_gallery_url
    60→    60→  let image: string | null = null;
    61→    61→  const topGallery = parsed.searchParams.get("top_gallery_url");
    62→    62→  if (topGallery) {
    63→    63→    try {
    64→    64→      // Validate it's a real URL
    65→    65→      const imgUrl = new URL(topGallery);
    66→    66→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
    67→    67→        image = topGallery;
    68→    68→      }
    69→    69→    } catch { /* not a valid URL */ }
    70→    70→  }
    71→    71→
    72→    72→  // Product name from URL slug
    73→    73→  let productName: string | null = null;
    74→    74→  const segments = parsed.pathname.split("/").filter(Boolean);
    75→    75→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
    76→    76→  const nameFromSlug = slug
    77→    77→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
    78→    78→    .replace(/\.html?$/i, "")
    79→    79→    .replace(/-/g, " ")
    80→    80→    .replace(/\s+/g, " ")
    81→    81→    .trim();
    82→    82→  if (nameFromSlug && nameFromSlug.length > 3) {
    83→    83→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
    84→    84→  }
    85→    85→
    86→    86→  // If we got at least the price (the most important piece), return success
    87→    87→  if (price) {
    88→    88→    return {
    89→    89→      price,
    90→    90→      currency,
    91→    91→      productName,
    92→    92→      productDescription: null,
    93→    93→      canonicalUrl: null,
    94→    94→      originalPrice: null,
    95→    95→      image,
    96→    96→      source: priceSource,
    97→    97→      antiBotDetected: false,
    98→    98→    };
    99→    99→  }
   100→   100→
   101→   101→  // No price from URL — return null so caller tries HTML-based strategies
   102→   102→  return null;
   103→   103→}

   245→   245→async function fetchWithScrapingBee(url: string): Promise<TemuProductData | null> {
   246→   246→  const apiKey = process.env.SCRAPINGBEE_API_KEY;
   247→   247→  if (!apiKey) return null;
   248→   248→
   249→   249→  // Always add ?_x_sessn=us&currency=USD to force US site (gives cleaner data)
   250→   250→  let fetchUrl = url;
   251→   251→  try {
   252→   252→    const parsed = new URL(fetchUrl);
   253→   253→    if (!parsed.searchParams.has("_x_sessn")) {
   254→   254→      parsed.searchParams.set("_x_sessn", "us");
   255→   255→      parsed.searchParams.set("currency", "USD");
   256→   256→      fetchUrl = parsed.toString();
   257→   257→    }
   258→   258→  } catch { /* not a parseable URL, use as-is */ }
   259→   259→
   260→   260→  // Retry up to 3 times - Temu's anti-bot is probabilistic, sometimes we get the real page
   261→   261→  let lastError: string | null = null;
   262→   262→  for (let attempt = 1; attempt <= 3; attempt++) {
   263→   263→    try {
   264→   264→      const apiUrl = new URL("https://app.scrapingbee.com/api/v1/");
   265→   265→      apiUrl.searchParams.set("api_key", apiKey);
   266→   266→      apiUrl.searchParams.set("url", fetchUrl);
   267→   267→      apiUrl.searchParams.set("render_js", "true");
   268→   268→      apiUrl.searchParams.set("premium_proxy", "true");
   269→   269→      apiUrl.searchParams.set("country_code", "us");
   270→   270→
   271→   271→      const controller = new AbortController();
   272→   272→      const timeout = setTimeout(() => controller.abort(), 30000);
   273→   273→
   274→   274→      const response = await fetch(apiUrl.toString(), {
   275→   275→        signal: controller.signal,
   276→   276→      });
   277→   277→      clearTimeout(timeout);
   278→   278→
   279→   279→      if (!response.ok) {
   280→   280→        lastError = `ScrapingBee HTTP ${response.status}`;
   281→   281→        console.error(`[ScrapingBee attempt ${attempt}] ${lastError}`);
   282→   282→        await new Promise(r => setTimeout(r, 1000));
   283→   283→        continue;
   284→   284→      }
   285→   285→
   286→   286→      const html = await response.text();
   287→   287→      console.log(`[ScrapingBee attempt ${attempt}] HTML length: ${html.length}, cost: ${response.headers.get("Spb-cost")} credits`);
   288→   288→
   289→   289→      const result = extractProductInfo(html, url);
   290→   290→
   291→   291→      // If we got the real product page (not anti-bot), use it
   292→   292→      // Heuristic: anti-bot page is ~310-325KB and has many "verify" mentions
   293→   293→      const isAntiBot = html.length < 450000 && (html.match(/verify/gi) || []).length > 100;
   294→   294→
   295→   295→      if (result.productName || result.price) {
   296→   296→        return {
   297→   297→          ...result,
   298→   298→          antiBotDetected: isAntiBot,
   299→   299→          source: isAntiBot ? "scrapingbee-og" : "scrapingbee-full",
   300→   300→        };
   301→   301→      }
   302→   302→
   303→   303→      // Got a page but no useful data — retry
   304→   304→      lastError = "Page returned but no product info extracted";
   305→   305→      await new Promise(r => setTimeout(r, 1000));
   306→   306→    } catch (err) {
   307→   307→      lastError = String(err).slice(0, 150);
   308→   308→      console.error(`[ScrapingBee attempt ${attempt}] Error:`, lastError);
   309→   309→      await new Promise(r => setTimeout(r, 1000));
   310→   310→    }
   311→   311→  }
   312→   312→
   313→   313→  console.error("[ScrapingBee] All attempts failed. Last error:", lastError);
   314→   314→  return null;
   315→   315→}

   317→   317→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
   318→   318→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
   319→   319→  // 1. OG meta tags (work even on anti-bot page)
   320→   320→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   321→   321→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   322→   322→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   323→   323→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
   324→   324→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
   325→   325→
   326→   326→  // Decode HTML entities in title
   327→   327→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
   328→   328→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
   329→   329→
   330→   330→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
   331→   331→  let price: number | null = null;
   332→   332→  let currency = "USD";
   333→   333→  let originalPrice: number | null = null;
   334→   334→  let priceSource = "";
   335→   335→
   336→   336→  // ─────────────────────────────────────────────────────────────
   337→   337→  // 2-preferred. URL-embedded price hint.
   338→   338→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
   339→   339→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
   340→   340→  // → 1.23 USD. This is the most reliable signal because it comes
   341→   341→  // straight from the share URL the user copied.
   342→   342→  // ─────────────────────────────────────────────────────────────
   343→   343→  try {
   344→   344→    const parsedUrl = new URL(originalUrl);
   345→   345→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
   346→   346→    if (hint) {
   347→   347→      // base64 decode (URL-safe variant — replace - with + and _ with /)
   348→   348→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   349→   349→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   350→   350→      // Strip non-digits
   351→   351→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   352→   352→      if (cents > 0 && cents < 10000000) {
   353→   353→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
   354→   354→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
   355→   355→        const usd = cents / 100;
   356→   356→        if (usd >= 0.01 && usd < 100000) {
   357→   357→          price = usd;
   358→   358→          currency = "USD";
   359→   359→          priceSource = "url-hint";
   360→   360→        }
   361→   361→      }
   362→   362→    }
   363→   363→  } catch { /* not a parseable URL */ }
   364→   364→
   365→   365→  // 2a. JSON-LD structured data
   366→   366→  if (!price) {
   367→   367→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
   368→   368→    for (const match of jsonLdMatches) {
   369→   369→      try {
   370→   370→        const data = JSON.parse(match[1]);
   371→   371→        const schemas = Array.isArray(data) ? data : [data];
   372→   372→        for (const schema of schemas) {
   373→   373→          if (schema["@type"] === "Product" && schema.offers) {
   374→   374→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
   375→   375→            for (const offer of offers) {
   376→   376→              if (offer.price !== undefined) {
   377→   377→                price = parseFloat(offer.price);
   378→   378→                currency = offer.priceCurrency || "USD";
   379→   379→                priceSource = "json-ld";
   380→   380→                break;
   381→   381→              }
   382→   382→            }
   383→   383→          }
   384→   384→        }
   385→   385→      } catch { /* skip */ }
   386→   386→      if (price) break;
   387→   387→    }
   388→   388→  }
   389→   389→
   390→   390→  // 2b. OG product:price:amount meta
   391→   391→  if (!price) {
   392→   392→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   393→   393→    if (ogPrice) {
   394→   394→      const p = parseFloat(ogPrice[1]);
   395→   395→      if (p > 0) {
   396→   396→        price = p;
   397→   397→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   398→   398→        currency = ogCur?.[1] || "USD";
   399→   399→        priceSource = "og-meta";
   400→   400→      }
   401→   401→    }
   402→   402→  }
   403→   403→
   404→   404→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
   405→   405→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
   406→   406→  //     one per "similar product", one per cross-sell, etc.). The previous
   407→   407→  //     implementation picked the FIRST match, which often returned a
   408→   408→  //     cross-sell product's price (e.g. $30) instead of the actual
   409→   409→  //     product's price (e.g. $1.23). We now collect ALL matches and
   410→   410→  //     pick the LOWEST plausible one, since the actual product the user
   411→   411→  //     is viewing is almost always the cheapest among the rendered
   412→   412→  //     priceInfo blocks.
   413→   413→  if (!price) {
   414→   414→    type Candidate = { usd: number; currency: string; marketUsd?: number };
   415→   415→    const candidates: Candidate[] = [];
   416→   416→
   417→   417→    // priceInfo with explicit priceStr (most reliable)
   418→   418→    const priceInfosWithStr = [...html.matchAll(
   419→   419→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
   420→   420→    )];
   421→   421→    for (const pi of priceInfosWithStr) {
   422→   422→      const cents = parseInt(pi[1]);
   423→   423→      const cur = pi[2];
   424→   424→      const priceStr = pi[3];
   425→   425→      // Skip i18n placeholder strings
   426→   426→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
   427→   427→      if (cents <= 0 || cents >= 100000000) continue;
   428→   428→      const usd = cents / 100;
   429→   429→      if (usd < 0.01 || usd > 100000) continue;
   430→   430→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
   431→   431→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
   432→   432→      candidates.push({ usd, currency: cur, marketUsd });
   433→   433→    }
   434→   434→
   435→   435→    // priceInfo without priceStr (broader fallback)
   436→   436→    if (candidates.length === 0) {
   437→   437→      const priceInfosAll = [...html.matchAll(
   438→   438→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
   439→   439→      )];
   440→   440→      for (const pi of priceInfosAll) {
   441→   441→        const cents = parseInt(pi[1]);
   442→   442→        const cur = pi[2];
   443→   443→        if (cents <= 50 || cents >= 100000000) continue;
   444→   444→        const usd = cents / 100;
   445→   445→        if (usd < 0.50 || usd > 100000) continue;
   446→   446→        candidates.push({ usd, currency: cur });
   447→   447→      }
   448→   448→    }
   449→   449→
   450→   450→    if (candidates.length > 0) {
   451→   451→      // Pick the LOWEST price — the user is always viewing the cheapest
   452→   452→      // variant of the actual product, not the cross-sells.
   453→   453→      candidates.sort((a, b) => a.usd - b.usd);
   454→   454→      const best = candidates[0];
   455→   455→      price = best.usd;
   456→   456→      currency = best.currency;
   457→   457→      if (best.marketUsd && best.marketUsd > best.usd) {
   458→   458→        originalPrice = best.marketUsd;
   459→   459→      }
   460→   460→      priceSource = `priceInfo(lowest of ${candidates.length})`;
   461→   461→    }
   462→   462→  }
   463→   463→
   464→   464→  // 2d. Embedded JSON price fields (broad regex sweep)
   465→   465→  if (!price) {
   466→   466→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
   467→   467→    const found: { value: number; field: string }[] = [];
   468→   468→    for (const f of fields) {
   469→   469→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
   470→   470→      for (const m of html.matchAll(re)) {
   471→   471→        const v = parseFloat(m[1]);
   472→   472→        if (v > 0 && v < 100000) {
   473→   473→          found.push({ value: v, field: f });
   474→   474→        }
   475→   475→      }
   476→   476→    }
   477→   477→    if (found.length > 0) {
   478→   478→      // Sort and pick lowest (likely sale price)
   479→   479→      found.sort((a, b) => a.value - b.value);
   480→   480→      const best = found[0];
   481→   481→      // Heuristic: if value > 100, probably in cents → divide by 100
   482→   482→      price = best.value > 100 ? best.value / 100 : best.value;
   483→   483→      priceSource = `embedded-${best.field}`;
   484→   484→    }
   485→   485→  }
   486→   486→
   487→   487→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
   488→   488→  if (!price) {
   489→   489→    const text = html.replace(/<[^>]*>/g, " ");
   490→   490→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
   491→   491→    const prices: { value: number; currency: string }[] = [];
   492→   492→    for (const m of matches) {
   493→   493→      const raw = m[1].replace(/,/g, "");
   494→   494→      const v = parseFloat(raw);
   495→   495→      if (v > 1.5 && v < 5000) {
   496→   496→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
   497→   497→        prices.push({ value: v, currency: cur });
   498→   498→      }
   499→   499→    }
   500→   500→    if (prices.length > 0) {
   501→   501→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
   502→   502→      const sorted = prices.sort((a, b) => a.value - b.value);
   503→   503→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
   504→   504→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
   505→   505→      if (filtered.length > 0) {
   506→   506→        price = filtered[0].value;
   507→   507→        currency = filtered[0].currency;
   508→   508→        priceSource = "text";
   509→   509→      }
   510→   510→    }
   511→   511→  }
   512→   512→
   513→   513→  return {
   514→   514→    price,
   515→   515→    currency,
   516→   516→    productName,
   517→   517→    productDescription,
   518→   518→    canonicalUrl: canonical,
   519→   519→    originalPrice,
   520→   520→    image: ogImage,
   521→   521→    source: priceSource || (productName ? "og-only" : "none"),
   522→   522→  };
   523→   523→}

   526→   526→  return s
   527→   527→    .replace(/&amp;/g, "&")
   528→   528→    .replace(/&lt;/g, "<")

   536→   536→/* ─── Helper: Extract product name from URL slug ─── */
   537→   537→function extractProductNameFromUrl(url: string): string | null {
   538→   538→  try {
   539→   539→    const parsed = new URL(url);
   540→   540→    const segments = parsed.pathname.split("/").filter(Boolean);
   541→   541→    const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   542→   542→    const name = slug
   543→   543→      .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   544→   544→      .replace(/\.html?$/i, "")
   545→   545→      .replace(/-/g, " ")
   546→   546→      .trim();
   547→   547→    if (name && name.length > 3) {
   548→   548→      return name.replace(/\b\w/g, (l) => l.toUpperCase());
   549→   549→    }
   550→   550→  } catch { /* skip */ }
   551→   551→  return null;
   552→   552→}

   554→   554→function cleanProductName(name: string | null): string | null {
   555→   555→  if (!name) return null;
   556→   556→  let cleaned = name
   557→   557→    .replace(/\s*[-|]\s*(Temu|AliExpress)\s*/gi, "")
   558→   558→    .replace(/\s*[-|]\s*(Login|Sign In|Register)\s*/gi, "")
   559→   559→    .replace(/\s+/g, " ")
   560→   560→    .trim();
   561→   561→  if (cleaned.length > 100) cleaned = cleaned.slice(0, 97) + "...";
   562→   562→  return cleaned || null;
   563→   563→}

   566→   566→export async function POST(request: NextRequest) {
   567→   567→  try {
   568→   568→    const body = await request.json();
   569→   569→    const { url, manualPrice } = body;
   570→   570→
   571→   571→    // ─── Manual price entry ───
   572→   572→    if (manualPrice !== undefined && manualPrice !== null && manualPrice !== "") {
   573→   573→      const price = parseFloat(String(manualPrice).replace(/[^\d.]/g, ""));
   574→   574→      if (price > 0 && price < 100000) {
   575→   575→        // Detect currency from the input string
   576→   576→        const priceStr = String(manualPrice);
   577→   577→        const isDZD = /DA|dzd|DZD|دينار/i.test(priceStr);
   578→   578→        const isGBP = /£|GBP/i.test(priceStr);
   579→   579→        const isEUR = /€|EUR/i.test(priceStr);
   580→   580→
   581→   581→        let priceUSD = price;
   582→   582→        if (isDZD) {
   583→   583→          const rate = await getUsdToDzdRate();
   584→   584→          priceUSD = price / rate.rate;
   585→   585→        } else if (isGBP) {
   586→   586→          priceUSD = price * 1.27;
   587→   587→        } else if (isEUR) {
   588→   588→          priceUSD = price * 1.08;
   589→   589→        }
   590→   590→
   591→   591→        const breakdown = calculateAlgeriaPrice(priceUSD);
   592→   592→        return NextResponse.json({

   598→   598→          productImage: body.productImage || null,
   599→   599→          productUrl: url || null,
   600→   600→          estimated: false,
   601→   601→          manual: true,
   602→   602→        });
   603→   603→      }
   604→   604→    }
   605→   605→
   606→   606→    if (!url || typeof url !== "string") {
   607→   607→      return NextResponse.json(

   609→   609→          success: false,

   707→   707→      if (result) {

   711→   711→        if (result.productName) {
   712→   712→          return NextResponse.json({
   713→   713→            success: true,
   714→   714→            price: null,

   716→   716→            productName: cleanProductName(result.productName) || urlProductName,
   717→   717→            productDescription: result.productDescription,

   752→   752→/* ─── Build success response with Algeria pricing ─── */
   753→   753→async function buildSuccessResponse(
   754→   754→  result: TemuProductData,
   755→   755→  urlProductName: string | null
   756→   756→) {
   757→   757→  let priceUSD = result.price!;
   758→   758→
   759→   759→  // Convert to USD if in different currency
   760→   760→  if (result.currency?.toUpperCase() !== "USD") {
   761→   761→    const rates: Record<string, number> = {
   762→   762→      EUR: 1.08,
   763→   763→      GBP: 1.27,
   764→   764→      CNY: 0.14,
   765→   765→      DZD: 0.0075,
   766→   766→    };
   767→   767→    const rate = rates[result.currency.toUpperCase()];
   768→   768→    if (rate) priceUSD = result.price! * rate;
   769→   769→  }
   770→   770→
   771→   771→  // Ensure we have a fresh exchange rate
   772→   772→  await getUsdToDzdRate();
   773→   773→
   774→   774→  // Calculate Algeria price with all fees
   775→   775→  const breakdown = calculateAlgeriaPrice(priceUSD);
   776→   776→
   777→   777→  return NextResponse.json({
   778→   778→    success: true,
   779→   779→    price: breakdown.basePriceUSD,
   780→   780→    dzd: breakdown.totalDZD,
   781→   781→    breakdown,
   782→   782→    productName: cleanProductName(result.productName) || urlProductName,
   783→   783→    productDescription: result.productDescription,
   784→   784→    productImage: result.image,
   785→   785→    productUrl: result.canonicalUrl,
   786→   786→    originalPrice: result.originalPrice ? Math.round(result.originalPrice * 100) / 100 : null,
   787→   787→    estimated: false,
   788→   788→    manual: false,
   789→   789→    source: result.source,
   790→   790→  });
   791→   791→}

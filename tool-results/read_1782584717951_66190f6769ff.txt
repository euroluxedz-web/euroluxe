    99→export default function CalculateurPage() {
   100→  const [productUrl, setProductUrl] = useState("");
   101→  const [manualPrice, setManualPrice] = useState("");
   102→  const [loading, setLoading] = useState(false);
   103→  const [result, setResult] = useState<PriceResult | null>(null);
   104→  const [error, setError] = useState("");
   105→  const [copied, setCopied] = useState(false);
   106→  const [addedToCart, setAddedToCart] = useState(false);
   107→  const [detectedCode, setDetectedCode] = useState<string | null>(null);
   108→  const [temuLink, setTemuLink] = useState<string | null>(null);
   109→  const [detectedProduct, setDetectedProduct] = useState<DetectedProduct | null>(null);
   110→  const priceInputRef = useRef<HTMLInputElement>(null);
   111→  const { t, isArabic } = useLanguage();
   112→  const { user, profile } = useAuth();
   113→  const router = useRouter();
   114→  const addItemToStore = useCartStore((s) => s.addItem);
   115→
   116→  // Checkout state
   117→  const [showCheckout, setShowCheckout] = useState(false);
   118→  const [orderSuccess, setOrderSuccess] = useState(false);
   119→  const [submitting, setSubmitting] = useState(false);
   120→  const [saveInfo, setSaveInfo] = useState(true);
   121→  const [useSaved, setUseSaved] = useState(false);
   122→  const [shippingError, setShippingError] = useState("");
   123→
   124→  const [shipping, setShipping] = useState<ShippingInfo>({
   125→    fullName: "",
   126→    phone: "",
   127→    wilaya: "",
   128→    commune: "",
   129→    codePostal: "",
   130→    address: "",
   131→    notes: "",
   132→  });
   133→
   134→  // Available communes based on selected wilaya
   135→  const [availableCommunes, setAvailableCommunes] = useState<Commune[]>([]);
   136→
   137→  const isAuthenticated = !!user;
   138→
   139→  // Pre-fill shipping info from user profile
   140→  useEffect(() => {
   141→    if (profile && isAuthenticated) {
   142→      const hasSavedInfo = profile.phone || profile.wilaya || profile.address;
   143→      if (hasSavedInfo) {
   144→        setUseSaved(true);
   145→        const communes = getCommunesForWilaya(profile.wilaya || "");
   146→        setAvailableCommunes(communes);
   147→        setShipping({
   148→          fullName: profile.name || "",
   149→          phone: profile.phone || "",
   150→          wilaya: profile.wilaya || "",
   151→          commune: profile.commune || "",
   152→          codePostal: profile.codePostal || "",
   153→          address: profile.address || "",
   154→          notes: "",
   155→        });
   156→      } else {
   157→        setShipping((prev) => ({
   158→          ...prev,
   159→          fullName: profile.name || "",
   160→          phone: profile.phone || "",
   161→        }));
   162→      }
   163→    }
   164→  }, [profile, isAuthenticated]);
   165→
   166→  // Detect Temu product code / Item ID in real-time
   167→  useEffect(() => {
   168→    const input = productUrl.trim();
   169→    const isTemuCode = /^[a-zA-Z0-9]{6,30}$/.test(input);
   170→    if (isTemuCode) {
   171→      setDetectedCode(input);
   172→      setTemuLink(`https://www.temu.com/-g-${input}.html`);
   173→    } else if (input.includes("temu.com")) {
   174→      setDetectedCode(null);
   175→      setTemuLink(input.startsWith("http") ? input : `https://${input}`);
   176→    } else {
   177→      setDetectedCode(null);
   178→      setTemuLink(null);
   179→    }
   180→  }, [productUrl]);
   181→
   182→  // Extract product name from URL
   183→  const extractProductName = (url: string): string | null => {
   184→    try {
   185→      const parsed = new URL(url.startsWith("http") ? url : `https://${url}`);
   186→      const segments = parsed.pathname.split("/").filter(Boolean);
   187→      const slug =
   188→        segments.find((s) => s.includes("-g-") && s.length > 10) ||
   189→        segments.find((s) => s.includes("-") && s.length > 10) ||
   190→        segments[segments.length - 1] ||
   191→        "";
   192→      const name = slug
   193→        .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   194→        .replace(/\.html?$/i, "")
   195→        .replace(/-/g, " ")
   196→        .trim();
   197→      if (name && name.length > 3) {
   198→        return name.replace(/\b\w/g, (l) => l.toUpperCase());
   199→      }
   200→    } catch { /* skip */ }
   201→    return null;
   202→  };
   203→
   204→  // AUTO-EXTRACT
   205→  const handleAutoExtract = async () => {
   206→    setError("");
   207→    setResult(null);
   208→    setShowCheckout(false);
   209→    setOrderSuccess(false);
   210→    setDetectedProduct(null);
   211→
   212→    if (!productUrl.trim()) {
   213→      setError(t("calc.error.empty"));
   214→      return;
   215→    }
   216→
   217→    const isTemuProductId = /^[a-zA-Z0-9]{6,30}$/.test(productUrl.trim());
   218→    let finalUrl = productUrl.trim();
   219→
   220→    if (isTemuProductId) {
   221→      finalUrl = `https://www.temu.com/-g-${productUrl.trim()}.html`;
   222→    } else {
   223→      try {
   224→        new URL(finalUrl);
   225→      } catch {
   226→        setError(t("calc.error.invalidUrl"));
   227→        return;
   228→      }
   229→    }
   230→
   231→    setLoading(true);
   232→
   233→    try {
   234→      const response = await fetch("/api/scrape-price", {
   235→        method: "POST",
   236→        headers: { "Content-Type": "application/json" },
   237→        body: JSON.stringify({ url: finalUrl }),
   238→      });
   239→
   240→      const data = await response.json();
   241→
   242→      // Case 1: Auto-extracted price found
   243→      if (data.success && data.price && data.price > 0) {
   244→        setTemuLink(finalUrl);
   245→        setResult({
   246→          usd: data.price,
   247→          dzd: data.dzd || data.price * 270,
   248→          breakdown: data.breakdown,
   249→          productName: data.productName,
   250→          originalPrice: data.originalPrice || null,
   251→          image: data.productImage || data.image || null,
   252→          estimated: data.estimated || false,
   253→          manual: data.manual || false,
   254→          source: data.source || "auto",
   255→          itemId: data.itemId || undefined,
   256→        });
   257→      }
   258→      // Case 2: Product detected but needs manual price entry
   259→      else if (data.success && data.requiresManualPrice && data.productName) {
   260→        setTemuLink(data.productUrl || finalUrl);
   261→        setDetectedProduct({
   262→          name: data.productName,
   263→          description: data.productDescription,
   264→          image: data.productImage || null,
   265→          url: data.productUrl || finalUrl,
   266→          antiBotDetected: data.antiBotDetected,
   267→          message: data.message,
   268→        });
   269→        setError(
   270→          data.message ||
   271→            (isArabic
   272→              ? "تم العثور على المنتج! يرجى إدخال السعر المعروض على Temu في الحقل أدناه."
   273→              : "Produit trouvé ! Veuillez saisir le prix affiché sur Temu dans le champ ci-dessous.")
   274→        );
   275→        setTimeout(() => priceInputRef.current?.focus(), 300);
   276→      }
   277→      // Case 3: Failure
   278→      else {
   279→        setTemuLink(finalUrl);
   280→        setError(
   281→          data.error ||
   282→            (isArabic
   283→              ? "لم نتمكن من استخراج السعر تلقائياً. يرجى إدخاله يدوياً."
   284→              : "Extraction automatique indisponible. Veuillez entrer le prix manuellement.")
   285→        );
   286→        setTimeout(() => priceInputRef.current?.focus(), 300);
   287→      }
   288→    } catch {
   289→      setError(
   290→        isArabic
   291→          ? "يرجى إدخال السعر يدوياً في الحقل أدناه."
   292→          : "Veuillez entrer le prix manuellement dans le champ ci-dessous."
   293→      );
   294→      setTimeout(() => priceInputRef.current?.focus(), 300);
   295→    } finally {
   296→      setLoading(false);
   297→    }
   298→  };
   299→
   300→  // Manual price calculation - uses API for proper breakdown
   301→  const handleManualCalculate = async () => {
   302→    setError("");
   303→    setResult(null);
   304→    setShowCheckout(false);
   305→    setOrderSuccess(false);
   306→
   307→    const normalized = manualPrice.trim().replace(/,/g, ".");
   308→    const priceStr = normalized.replace(/[^\d.]/g, "");
   309→    const parts = priceStr.split(".");
   310→    const cleaned = parts.length > 1
   311→      ? parts[0] + "." + parts.slice(1).join("")
   312→      : priceStr;
   313→    const price = parseFloat(cleaned);
   314→
   315→    if (!price || price <= 0) {
   316→      setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   317→      return;
   318→    }
   319→
   320→    setLoading(true);
   321→    try {
   322→      // Pass the detected product info + URL so the API can use it
   323→      const payload: Record<string, string> = { manualPrice: manualPrice.trim() };
   324→      if (detectedProduct?.name) payload.productName = detectedProduct.name;
   325→      if (detectedProduct?.image) payload.productImage = detectedProduct.image;
   326→      if (productUrl.trim()) payload.url = productUrl.trim();
   327→
   328→      const response = await fetch("/api/scrape-price", {
   329→        method: "POST",
   330→        headers: { "Content-Type": "application/json" },
   331→        body: JSON.stringify(payload),
   332→      });
   333→      const data = await response.json();
   334→
   335→      if (data.success && data.price) {
   336→        const productName = detectedProduct?.name
   337→          || (productUrl.trim() ? extractProductName(productUrl) : null)
   338→          || data.productName;
   339→        setResult({
   340→          usd: data.price,
   341→          dzd: data.dzd,
   342→          breakdown: data.breakdown,
   343→          productName,
   344→          image: detectedProduct?.image || null,
   345→          estimated: false,
   346→          manual: true,
   347→          source: "manual",
   348→        });
   349→      } else {
   350→        setError(isArabic ? "يرجى إدخال سعر صالح" : "Veuillez entrer un prix valide");
   351→      }
   352→    } catch {
   353→      setError(isArabic ? "حدث خطأ، حاول مرة أخرى" : "Une erreur est survenue, réessayez");
   354→    } finally {
   355→      setLoading(false);
   356→    }
   357→  };
   358→
   359→  const handleCopyResult = () => {
   360→    if (result) {
   361→      const text = result.productName
   362→        ? `${result.productName} — ${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`
   363→        : `${t("calc.priceDzd")}: ${result.dzd.toLocaleString()} DA`;
   364→      navigator.clipboard.writeText(text);
   365→      setCopied(true);
   366→      setTimeout(() => setCopied(false), 2000);
   367→    }
   368→  };
   369→
   370→  const handleAddToCart = () => {
   371→    if (!result) return;
   372→    const cartItem = {
   373→      id: `calc-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   374→      productId: detectedCode || undefined,
   375→      name: result.productName || (isArabic ? "منتج" : "Produit"),
   376→      image: result.image || undefined,
   377→      price: result.usd,
   378→      quantity: 1,
   379→      url: productUrl.trim() || undefined,
   380→    };
   381→    addItemToStore(cartItem);
   382→    syncAddToServer(cartItem);
   383→    setAddedToCart(true);
   384→    setTimeout(() => setAddedToCart(false), 2000);
   385→  };
   386→
   387→  // Validate phone
   388→  const isValidPhone = (phone: string) => /^(05|06|07)\d{8}$/.test(phone.trim());
   389→
   390→  // Handle checkout submission
   391→  const handleSubmitOrder = async () => {
   392→    setShippingError("");
   393→
   394→    // Validate
   395→    if (!shipping.fullName.trim()) {
   396→      setShippingError(isArabic ? "يرجى إدخال الاسم الكامل" : "Veuillez entrer votre nom complet");
   397→      return;
   398→    }
   399→    if (!isValidPhone(shipping.phone)) {
   400→      setShippingError(t("calc.checkout.errorPhone"));
   401→      return;
   402→    }
   403→    if (!shipping.wilaya) {
   404→      setShippingError(isArabic ? "يرجى اختيار الولاية" : "Veuillez sélectionner votre wilaya");
   405→      return;
   406→    }
   407→    if (!shipping.commune) {
   408→      setShippingError(isArabic ? "يرجى اختيار البلدية" : "Veuillez sélectionner votre commune");
   409→      return;
   410→    }
   411→    if (!shipping.address.trim()) {
   412→      setShippingError(isArabic ? "يرجى إدخال العنوان" : "Veuillez entrer votre adresse");
   413→      return;
   414→    }
   415→
   416→    setSubmitting(true);
   417→
   418→    try {
   419→      // Save shipping info to user profile if requested
   420→      if (isAuthenticated && saveInfo && user) {
   421→        try {
   422→          await updateUserData(user.uid, {
   423→            name: shipping.fullName,
   424→            phone: shipping.phone,
   425→            wilaya: shipping.wilaya,
   426→            commune: shipping.commune,
   427→            codePostal: shipping.codePostal,
   428→            address: shipping.address,
   429→          });
   430→        } catch (e) {
   431→          console.error("Failed to save shipping info:", e);
   432→        }
   433→      }
   434→
   435→      // Create the order
   436→      if (isAuthenticated && user) {
   437→        const orderItems = [{
   438→          name: result?.productName || (isArabic ? "منتج" : "Produit"),
   439→          price: result?.dzd || 0,
   440→          quantity: 1,
   441→          image: result?.image || undefined,
   442→          url: productUrl.trim() || undefined,
   443→          productId: detectedCode || undefined,
   444→        }];
   445→
   446→        const { auth } = await import("@/lib/firebase");
   447→        const token = auth.currentUser ? await auth.currentUser.getIdToken() : null;
   448→
   449→        const res = await fetch("/api/orders", {
   450→          method: "POST",
   451→          headers: {
   452→            "Content-Type": "application/json",
   453→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   454→          },
   455→          body: JSON.stringify({
   456→            items: orderItems,
   457→            total: result?.dzd || 0,
   458→            wilaya: shipping.wilaya,
   459→            commune: shipping.commune,
   460→            codePostal: shipping.codePostal,
   461→            address: shipping.address,
   462→            phone: shipping.phone,
   463→            notes: shipping.notes,
   464→          }),
   465→        });
   466→
   467→        if (res.ok) {
   468→          setOrderSuccess(true);
   469→        } else {
   470→          setShippingError(isArabic ? "حدث خطأ، يرجى المحاولة مرة أخرى" : "Une erreur est survenue, veuillez réessayer");
   471→        }
   472→      }
   473→    } catch (err) {
   474→      console.error("Order error:", err);
   475→      setShippingError(isArabic ? "حدث خطأ في الاتصال" : "Erreur de connexion");
   476→    } finally {
   477→      setSubmitting(false);
   478→    }
   479→  };
   480→
   481→  return (
   482→    <div className="relative min-h-screen flex flex-col bg-transparent text-foreground overflow-x-hidden">
   483→      <Navbar />
   484→      <main className="flex-1 pt-16 sm:pt-20">
   485→        <section className="relative py-20 sm:py-32 overflow-hidden min-h-[80vh]">
   486→          {/* Background Effects */}
   487→          <div className="absolute inset-0 bg-gradient-to-b from-brand-blue/20 via-brand-blue-light/15 to-white/60" />
   488→          <div className="absolute top-0 left-0 w-72 h-72 bg-brand-pink/8 rounded-full blur-3xl" />
   489→          <div className="absolute bottom-0 right-0 w-72 h-72 bg-brand-blue/8 rounded-full blur-3xl" />
   490→
   491→          <div className="relative z-10 max-w-4xl mx-auto px-4">
   492→            {/* Section Header */}
   493→            <motion.div
   494→              initial={{ opacity: 0, y: 30 }}
   495→              animate={{ opacity: 1, y: 0 }}
   496→              transition={{ duration: 0.6 }}
   497→              className="text-center mb-12 relative"
   498→            >
   499→              <div className="hidden lg:block">
   500→                <ImgPlaceholder number={32} className="absolute -left-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[-8deg]" />
   501→                <ImgPlaceholder number={33} className="absolute -right-16 top-4 w-[120px] h-[150px] rounded-xl rotate-[8deg]" />
   502→              </div>
   503→
   504→              <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full bg-brand-pink/10 border border-brand-pink/20 text-brand-pink text-sm font-medium mb-4 font-display">
   505→                <Zap className="w-4 h-4" />
   506→                {t("calc.badge")}
   507→              </div>
   508→              <h1 className="text-3xl sm:text-5xl lg:text-6xl font-black mb-4 font-heading">
   509→                <span className="text-brand-dark">{t("calc.titleCalc")}</span>{" "}
   510→                <span className="bg-brand-gold/30 px-2 py-1 rounded-md text-brand-dark">
   511→                  {t("calc.titleProduct")}
   512→                </span>
   513→              </h1>
   514→              <p className="text-brand-muted-text text-lg max-w-xl mx-auto font-sans">
   515→                {t("calc.subtitle")}
   516→              </p>
   517→            </motion.div>
   518→
   519→            {/* Calculator Card */}
   520→            <motion.div
   521→              initial={{ opacity: 0, y: 30 }}
   522→              animate={{ opacity: 1, y: 0 }}
   523→              transition={{ duration: 0.6, delay: 0.2 }}
   524→              className="bg-white/80 backdrop-blur-xl rounded-3xl p-6 sm:p-10 border border-brand-pink/15 shadow-lg"
   525→            >
   526→              {/* ──── Product URL / Code Input ──── */}
   527→              <div className="mb-6">
   528→                <label className="block text-brand-dark/80 text-sm font-medium mb-2 font-sans">
   529→                  <Link2 className={`w-4 h-4 inline ${isArabic ? "ml-1" : "mr-1"}`} />
   530→                  {t("calc.label")}
   531→                </label>
   532→                <div className="flex gap-3">
   533→                  <div className="relative flex-1">
   534→                    <Input
   535→                      type="text"
   536→                      placeholder={t("calc.placeholder")}
   537→                      value={productUrl}
   538→                      onChange={(e) => {
   539→                        setProductUrl(e.target.value);
   540→                        setResult(null);
   541→                        setError("");
   542→                        setShowCheckout(false);
   543→                        setDetectedProduct(null);
   544→                      }}
   545→                      onKeyDown={(e) => e.key === "Enter" && handleAutoExtract()}
   546→                      className="bg-brand-light/50 border-brand-muted-warm focus:border-brand-pink/50 focus:ring-brand-pink/20 text-brand-dark placeholder:text-brand-muted-text/50 rounded-xl h-14 text-base font-sans"
   547→                      disabled={loading}
   548→                    />
   549→                    <ShoppingBag
   550→                      className={`absolute top-1/2 -translate-y-1/2 w-5 h-5 text-brand-muted-text/40 ${
   551→                        isArabic ? "left-4" : "right-4"
   552→                      }`}
   553→                    />
   554→                  </div>
   555→                  <Button

   274→   274→/* ───────────────────────────────────────────────────────────────────
   275→   275→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   276→   276→ *
   277→   277→ * Temu share URLs include everything we need:
   278→   278→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   279→   279→ *   - top_gallery_url    = URL-encoded product image URL
   280→   280→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   281→   281→ *
   282→   282→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   283→   283→ * ─────────────────────────────────────────────────────────────────── */
   284→   284→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   285→   285→  let parsed: URL;
   286→   286→  try {
   287→   287→    parsed = new URL(originalUrl);
   288→   288→  } catch {
   289→   289→    return null;
   290→   290→  }
   291→   291→
   292→   292→  let price: number | null = null;
   293→   293→  let currency = "USD";
   294→   294→  let priceSource = "";
   295→   295→
   296→   296→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   297→   297→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   298→   298→  if (hint) {
   299→   299→    try {
   300→   300→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   301→   301→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   302→   302→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   303→   303→      if (cents > 0 && cents < 10000000) {
   304→   304→        const usd = cents / 100;
   305→   305→        if (usd >= 0.01 && usd < 100000) {
   306→   306→          price = usd;
   307→   307→          currency = "USD";
   308→   308→          priceSource = "url-hint";
   309→   309→        }
   310→   310→      }
   311→   311→    } catch { /* not valid base64 */ }
   312→   312→  }
   313→   313→
   314→   314→  // Product image from top_gallery_url
   315→   315→  let image: string | null = null;
   316→   316→  const topGallery = parsed.searchParams.get("top_gallery_url");
   317→   317→  if (topGallery) {
   318→   318→    try {
   319→   319→      // Validate it's a real URL
   320→   320→      const imgUrl = new URL(topGallery);
   321→   321→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   322→   322→        image = topGallery;
   323→   323→      }
   324→   324→    } catch { /* not a valid URL */ }
   325→   325→  }
   326→   326→
   327→   327→  // Product name from URL slug
   328→   328→  let productName: string | null = null;
   329→   329→  const segments = parsed.pathname.split("/").filter(Boolean);
   330→   330→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   331→   331→  const nameFromSlug = slug
   332→   332→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   333→   333→    .replace(/\.html?$/i, "")
   334→   334→    .replace(/-/g, " ")
   335→   335→    .replace(/\s+/g, " ")
   336→   336→    .trim();
   337→   337→  if (nameFromSlug && nameFromSlug.length > 3) {
   338→   338→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   339→   339→  }
   340→   340→
   341→   341→  // If we got at least the price (the most important piece), return success
   342→   342→  if (price) {
   343→   343→    return {
   344→   344→      price,
   345→   345→      currency,
   346→   346→      productName,
   347→   347→      productDescription: null,
   348→   348→      canonicalUrl: null,
   349→   349→      originalPrice: null,
   350→   350→      image,
   351→   351→      source: priceSource,
   352→   352→      antiBotDetected: false,
   353→   353→    };
   354→   354→  }
   355→   355→
   356→   356→  // No price from URL — return null so caller tries HTML-based strategies
   357→   357→  return null;
   358→   358→}

   624→   624→  results: any[],

   630→   630→    if (!result.snippet) continue;

   697→   697→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);

   700→   700→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56

   730→   730→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },

   744→   744→        if (localPrice > 0 && localPrice < 10000000) {
   745→   745→          const rate = CURRENCY_TO_USD[currency];
   746→   746→          if (rate) {
   747→   747→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   748→   748→            if (usdPrice >= 0.10 && usdPrice <= 5000) {

  1207→  1207→

  1218→  1218→
  1219→  1219→      const controller = new AbortController();
  1220→  1220→      const timeout = setTimeout(() => controller.abort(), 30000);
  1221→  1221→
  1222→  1222→      const response = await fetch(apiUrl.toString(), {
  1223→  1223→        signal: controller.signal,
  1224→  1224→      });
  1225→  1225→      clearTimeout(timeout);
  1226→  1226→
  1227→  1227→      if (!response.ok) {

  1230→  1230→        await new Promise(r => setTimeout(r, 1000));
  1231→  1231→        continue;
  1232→  1232→      }
  1233→  1233→
  1234→  1234→      const html = await response.text();

  1236→  1236→
  1237→  1237→      const result = extractProductInfo(html, url);
  1238→  1238→
  1239→  1239→      // If we got the real product page (not anti-bot), use it
  1240→  1240→      // Heuristic: anti-bot page is ~310-325KB and has many "verify" mentions
  1241→  1241→      const isAntiBot = html.length < 450000 && (html.match(/verify/gi) || []).length > 100;
  1242→  1242→
  1243→  1243→      if (result.productName || result.price) {
  1244→  1244→        return {
  1245→  1245→          ...result,
  1246→  1246→          antiBotDetected: isAntiBot,
  1247→  1247→          source: isAntiBot ? "scrapingbee-og" : "scrapingbee-full",
  1248→  1248→        };
  1249→  1249→      }
  1250→  1250→
  1251→  1251→      // Got a page but no useful data — retry
  1252→  1252→      lastError = "Page returned but no product info extracted";
  1253→  1253→      await new Promise(r => setTimeout(r, 1000));
  1254→  1254→    } catch (err) {
  1255→  1255→      lastError = String(err).slice(0, 150);
  1256→  1256→      console.error(`[ScrapingBee attempt ${attempt}] Error:`, lastError);
  1257→  1257→      await new Promise(r => setTimeout(r, 1000));
  1258→  1258→    }
  1259→  1259→  }
  1260→  1260→
  1261→  1261→  console.error("[ScrapingBee] All attempts failed. Last error:", lastError);
  1262→  1262→  return null;
  1263→  1263→}

  1265→  1265→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1266→  1266→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1267→  1267→  // 1. OG meta tags (work even on anti-bot page)
  1268→  1268→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1269→  1269→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1270→  1270→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1271→  1271→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1272→  1272→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1273→  1273→
  1274→  1274→  // Decode HTML entities in title
  1275→  1275→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1276→  1276→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1277→  1277→
  1278→  1278→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1279→  1279→  let price: number | null = null;
  1280→  1280→  let currency = "USD";
  1281→  1281→  let originalPrice: number | null = null;
  1282→  1282→  let priceSource = "";
  1283→  1283→
  1284→  1284→  // ─────────────────────────────────────────────────────────────
  1285→  1285→  // 2-preferred. URL-embedded price hint.
  1286→  1286→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1287→  1287→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1288→  1288→  // → 1.23 USD. This is the most reliable signal because it comes
  1289→  1289→  // straight from the share URL the user copied.
  1290→  1290→  // ─────────────────────────────────────────────────────────────
  1291→  1291→  try {
  1292→  1292→    const parsedUrl = new URL(originalUrl);
  1293→  1293→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1294→  1294→    if (hint) {
  1295→  1295→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1296→  1296→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1297→  1297→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1298→  1298→      // Strip non-digits
  1299→  1299→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1300→  1300→      if (cents > 0 && cents < 10000000) {
  1301→  1301→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1302→  1302→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1303→  1303→        const usd = cents / 100;
  1304→  1304→        if (usd >= 0.01 && usd < 100000) {
  1305→  1305→          price = usd;
  1306→  1306→          currency = "USD";
  1307→  1307→          priceSource = "url-hint";
  1308→  1308→        }
  1309→  1309→      }
  1310→  1310→    }
  1311→  1311→  } catch { /* not a parseable URL */ }
  1312→  1312→
  1313→  1313→  // 2a. JSON-LD structured data
  1314→  1314→  if (!price) {
  1315→  1315→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1316→  1316→    for (const match of jsonLdMatches) {
  1317→  1317→      try {
  1318→  1318→        const data = JSON.parse(match[1]);
  1319→  1319→        const schemas = Array.isArray(data) ? data : [data];
  1320→  1320→        for (const schema of schemas) {
  1321→  1321→          if (schema["@type"] === "Product" && schema.offers) {
  1322→  1322→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1323→  1323→            for (const offer of offers) {
  1324→  1324→              if (offer.price !== undefined) {
  1325→  1325→                price = parseFloat(offer.price);
  1326→  1326→                currency = offer.priceCurrency || "USD";
  1327→  1327→                priceSource = "json-ld";
  1328→  1328→                break;
  1329→  1329→              }
  1330→  1330→            }
  1331→  1331→          }
  1332→  1332→        }
  1333→  1333→      } catch { /* skip */ }
  1334→  1334→      if (price) break;
  1335→  1335→    }
  1336→  1336→  }
  1337→  1337→
  1338→  1338→  // 2b. OG product:price:amount meta
  1339→  1339→  if (!price) {
  1340→  1340→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1341→  1341→    if (ogPrice) {
  1342→  1342→      const p = parseFloat(ogPrice[1]);
  1343→  1343→      if (p > 0) {
  1344→  1344→        price = p;
  1345→  1345→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1346→  1346→        currency = ogCur?.[1] || "USD";
  1347→  1347→        priceSource = "og-meta";
  1348→  1348→      }
  1349→  1349→    }
  1350→  1350→  }
  1351→  1351→
  1352→  1352→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1353→  1353→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1354→  1354→  //     one per "similar product", one per cross-sell, etc.). The previous
  1355→  1355→  //     implementation picked the FIRST match, which often returned a
  1356→  1356→  //     cross-sell product's price (e.g. $30) instead of the actual
  1357→  1357→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1358→  1358→  //     pick the LOWEST plausible one, since the actual product the user
  1359→  1359→  //     is viewing is almost always the cheapest among the rendered
  1360→  1360→  //     priceInfo blocks.
  1361→  1361→  if (!price) {
  1362→  1362→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1363→  1363→    const candidates: Candidate[] = [];
  1364→  1364→
  1365→  1365→    // priceInfo with explicit priceStr (most reliable)
  1366→  1366→    const priceInfosWithStr = [...html.matchAll(
  1367→  1367→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1368→  1368→    )];
  1369→  1369→    for (const pi of priceInfosWithStr) {
  1370→  1370→      const cents = parseInt(pi[1]);
  1371→  1371→      const cur = pi[2];
  1372→  1372→      const priceStr = pi[3];
  1373→  1373→      // Skip i18n placeholder strings
  1374→  1374→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1375→  1375→      if (cents <= 0 || cents >= 100000000) continue;
  1376→  1376→      const usd = cents / 100;
  1377→  1377→      if (usd < 0.01 || usd > 100000) continue;
  1378→  1378→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1379→  1379→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1380→  1380→      candidates.push({ usd, currency: cur, marketUsd });
  1381→  1381→    }
  1382→  1382→
  1383→  1383→    // priceInfo without priceStr (broader fallback)
  1384→  1384→    if (candidates.length === 0) {
  1385→  1385→      const priceInfosAll = [...html.matchAll(
  1386→  1386→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1387→  1387→      )];
  1388→  1388→      for (const pi of priceInfosAll) {
  1389→  1389→        const cents = parseInt(pi[1]);
  1390→  1390→        const cur = pi[2];
  1391→  1391→        if (cents <= 50 || cents >= 100000000) continue;
  1392→  1392→        const usd = cents / 100;
  1393→  1393→        if (usd < 0.50 || usd > 100000) continue;
  1394→  1394→        candidates.push({ usd, currency: cur });
  1395→  1395→      }
  1396→  1396→    }
  1397→  1397→
  1398→  1398→    if (candidates.length > 0) {
  1399→  1399→      // Pick the LOWEST price — the user is always viewing the cheapest
  1400→  1400→      // variant of the actual product, not the cross-sells.
  1401→  1401→      candidates.sort((a, b) => a.usd - b.usd);
  1402→  1402→      const best = candidates[0];
  1403→  1403→      price = best.usd;
  1404→  1404→      currency = best.currency;
  1405→  1405→      if (best.marketUsd && best.marketUsd > best.usd) {
  1406→  1406→        originalPrice = best.marketUsd;
  1407→  1407→      }
  1408→  1408→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1409→  1409→    }
  1410→  1410→  }
  1411→  1411→
  1412→  1412→  // 2d. Embedded JSON price fields (broad regex sweep)
  1413→  1413→  if (!price) {
  1414→  1414→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1415→  1415→    const found: { value: number; field: string }[] = [];
  1416→  1416→    for (const f of fields) {
  1417→  1417→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1418→  1418→      for (const m of html.matchAll(re)) {
  1419→  1419→        const v = parseFloat(m[1]);
  1420→  1420→        if (v > 0 && v < 100000) {
  1421→  1421→          found.push({ value: v, field: f });
  1422→  1422→        }
  1423→  1423→      }
  1424→  1424→    }
  1425→  1425→    if (found.length > 0) {
  1426→  1426→      // Sort and pick lowest (likely sale price)
  1427→  1427→      found.sort((a, b) => a.value - b.value);
  1428→  1428→      const best = found[0];
  1429→  1429→      // Heuristic: if value > 100, probably in cents → divide by 100
  1430→  1430→      price = best.value > 100 ? best.value / 100 : best.value;
  1431→  1431→      priceSource = `embedded-${best.field}`;
  1432→  1432→    }
  1433→  1433→  }
  1434→  1434→
  1435→  1435→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1436→  1436→  if (!price) {
  1437→  1437→    const text = html.replace(/<[^>]*>/g, " ");
  1438→  1438→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1439→  1439→    const prices: { value: number; currency: string }[] = [];
  1440→  1440→    for (const m of matches) {
  1441→  1441→      const raw = m[1].replace(/,/g, "");
  1442→  1442→      const v = parseFloat(raw);
  1443→  1443→      if (v > 1.5 && v < 5000) {
  1444→  1444→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1445→  1445→        prices.push({ value: v, currency: cur });
  1446→  1446→      }
  1447→  1447→    }
  1448→  1448→    if (prices.length > 0) {
  1449→  1449→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1450→  1450→      const sorted = prices.sort((a, b) => a.value - b.value);
  1451→  1451→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1452→  1452→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1453→  1453→      if (filtered.length > 0) {
  1454→  1454→        price = filtered[0].value;
  1455→  1455→        currency = filtered[0].currency;
  1456→  1456→        priceSource = "text";
  1457→  1457→      }
  1458→  1458→    }
  1459→  1459→  }
  1460→  1460→
  1461→  1461→  return {
  1462→  1462→    price,
  1463→  1463→    currency,
  1464→  1464→    productName,
  1465→  1465→    productDescription,
  1466→  1466→    canonicalUrl: canonical,
  1467→  1467→    originalPrice,
  1468→  1468→    image: ogImage,
  1469→  1469→    source: priceSource || (productName ? "og-only" : "none"),
  1470→  1470→  };
  1471→  1471→}

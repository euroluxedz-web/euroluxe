   305→/* ───────────────────────────────────────────────────────────────────
   306→ * Strategy 0 (FREE, 0 network): extract everything from URL params.
   307→ *
   308→ * Temu share URLs include everything we need:
   309→ *   - _oak_rec_ext_1     = base64-encoded price in MINOR units (cents)
   310→ *   - top_gallery_url    = URL-encoded product image URL
   311→ *   - pathname slug      = human-readable product name (e.g. "tongue-scrapers--...")
   312→ *
   313→ * This strategy costs ZERO API credits and is INSTANT. We try it first.
   314→ * ─────────────────────────────────────────────────────────────────── */
   315→function extractFromUrlParams(originalUrl: string): TemuProductData | null {
   316→  let parsed: URL;
   317→  try {
   318→    parsed = new URL(originalUrl);
   319→  } catch {
   320→    return null;
   321→  }
   322→
   323→  let price: number | null = null;
   324→  let currency = "USD";
   325→  let priceSource = "";
   326→
   327→  // Price from _oak_rec_ext_1 (base64-encoded cents)
   328→  const hint = parsed.searchParams.get("_oak_rec_ext_1");
   329→  if (hint) {
   330→    try {
   331→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
   332→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
   333→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
   334→      if (cents > 0 && cents < 10000000) {
   335→        const usd = cents / 100;
   336→        if (usd >= 0.01 && usd < 100000) {
   337→          price = usd;
   338→          currency = "USD";
   339→          priceSource = "url-hint";
   340→        }
   341→      }
   342→    } catch { /* not valid base64 */ }
   343→  }
   344→
   345→  // Product image from top_gallery_url
   346→  let image: string | null = null;
   347→  const topGallery = parsed.searchParams.get("top_gallery_url");
   348→  if (topGallery) {
   349→    try {
   350→      // Validate it's a real URL
   351→      const imgUrl = new URL(topGallery);
   352→      if (imgUrl.protocol === "http:" || imgUrl.protocol === "https:") {
   353→        image = topGallery;
   354→      }
   355→    } catch { /* not a valid URL */ }
   356→  }
   357→
   358→  // Product name from URL slug
   359→  let productName: string | null = null;
   360→  const segments = parsed.pathname.split("/").filter(Boolean);
   361→  const slug = segments.find((s) => s.includes("-g-")) || segments[segments.length - 1] || "";
   362→  const nameFromSlug = slug
   363→    .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
   364→    .replace(/\.html?$/i, "")
   365→    .replace(/-/g, " ")
   366→    .replace(/\s+/g, " ")
   367→    .trim();
   368→  if (nameFromSlug && nameFromSlug.length > 3) {
   369→    productName = nameFromSlug.replace(/\b\w/g, (l) => l.toUpperCase());
   370→  }
   371→
   372→  // If we got at least the price (the most important piece), return success
   373→  if (price) {
   374→    return {
   375→      price,
   376→      currency,
   377→      productName,
   378→      productDescription: null,
   379→      canonicalUrl: null,
   380→      originalPrice: null,
   381→      image,
   382→      source: priceSource,
   383→      antiBotDetected: false,
   384→    };
   385→  }
   386→
   387→  // No price from URL — return null so caller tries HTML-based strategies
   388→  return null;
   389→}

   447→async function fetchPriceWithWebSearch(
   448→  goodsId: string | null,
   449→  itemId: string | null,
   450→  originalUrl: string,
   451→): Promise<TemuProductData | null> {
   452→  try {
   453→    const zai = await ZAI.create();
   454→
   455→    // Build search query — try different strategies
   456→    let searchQuery = "";
   457→    if (itemId && /^[A-Z]{2}\d+/i.test(itemId)) {
   458→      // Item ID like TV10922608 — search broadly on Temu
   459→      searchQuery = `temu ${itemId}`;
   460→    } else if (goodsId && /^\d{10,}$/.test(goodsId)) {
   461→      // Numeric goods_id — search with the -g- pattern
   462→      searchQuery = `site:temu.com "g-${goodsId}"`;
   463→    } else {
   464→      // Fallback: search with the URL
   465→      const cleanUrl = originalUrl.replace(/https?:\/\//, "").split("?")[0];
   466→      searchQuery = `site:temu.com ${cleanUrl}`;
   467→    }
   468→
   469→    console.log(`[WebSearch] Searching: ${searchQuery}`);
   470→
   471→    // Step 1: Web search to find the product
   472→    const searchResults = await (zai as any).invokeFunction("web_search", {
   473→      query: searchQuery,
   474→      num: 5,
   475→    });
   476→
   477→    if (!Array.isArray(searchResults) || searchResults.length === 0) {
   478→      console.log("[WebSearch] No results found, trying broader search...");
   479→
   480→      // Broader search without site: restriction
   481→      const broadQuery = itemId
   482→        ? `temu ${itemId} price`
   483→        : goodsId
   484→          ? `temu ${goodsId} price`
   485→          : `temu ${originalUrl.replace(/https?:\/\//, "").split("?")[0]} price`;
   486→
   487→      console.log(`[WebSearch] Broader search: ${broadQuery}`);
   488→      const broadResults = await (zai as any).invokeFunction("web_search", {
   489→        query: broadQuery,
   490→        num: 5,
   491→      });
   492→
   493→      if (!Array.isArray(broadResults) || broadResults.length === 0) {
   494→        console.log("[WebSearch] No results from broader search either");
   495→        return null;
   496→      }
   497→
   498→      return extractPriceFromSearchResults(zai, broadResults, goodsId, itemId);
   499→    }
   500→
   501→    // Step 2: Try to parse price from initial search results
   502→    const initialPrice = parsePriceFromSnippets(searchResults, goodsId);
   503→    if (initialPrice?.price && initialPrice.price > 0) {
   504→      console.log(`[WebSearch] ✓ Found price from initial search: $${initialPrice.price}`);
   505→      return initialPrice;
   506→    }
   507→
   508→    // Step 3: If initial search found product names but no prices,
   509→    // try a more targeted search with the product name to find prices
   510→    const productNameFromResults = searchResults
   511→      .filter((r: any) => r.url?.includes("temu.com") && r.name)
   512→      .map((r: any) => r.name?.replace(/\s*[-|]\s*Temu\s*$/i, "").replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "").trim())
   513→      .find((n: string) => n && n.length > 10);
   514→
   515→    if (productNameFromResults) {
   516→      console.log(`[WebSearch] No price in initial results. Trying name-based search: "${productNameFromResults.slice(0, 60)} price temu"`);
   517→      try {
   518→        const nameQuery = `${productNameFromResults.slice(0, 80)} price temu`;
   519→        const nameResults = await (zai as any).invokeFunction("web_search", {
   520→          query: nameQuery,
   521→          num: 5,
   522→        });
   523→        if (Array.isArray(nameResults) && nameResults.length > 0) {
   524→          const namePrice = parsePriceFromSnippets(nameResults, goodsId);
   525→          if (namePrice?.price && namePrice.price > 0) {
   526→            console.log(`[WebSearch] ✓ Found price from name search: $${namePrice.price}`);
   527→            return namePrice;
   528→          }
   529→          // Merge results and try LLM on combined results
   530→          const combinedResults = [...searchResults, ...nameResults];
   531→          return extractPriceFromSearchResults(zai, combinedResults, goodsId, itemId);
   532→        }
   533→      } catch (err) {
   534→        console.log("[WebSearch] Name search failed:", String(err).slice(0, 80));
   535→      }
   536→    }
   537→
   538→    // Step 4: Use LLM on whatever results we have
   539→    const llmResult = await extractPriceFromSearchResults(zai, searchResults, goodsId, itemId);
   540→    if (llmResult?.price && llmResult.price > 0) {
   541→      return llmResult;
   542→    }
   543→
   544→    // Step 5: If LLM didn't find price, try reading Temu search result pages
   545→    // via AllOrigins proxy — sometimes the proxy can access the product page
   546→    // when direct access is blocked
   547→    const temuResults = searchResults
   548→      .filter((r: any) => r.url?.includes("temu.com") && r.url?.includes("-g-"))
   549→      .slice(0, 2);
   550→
   551→    for (const result of temuResults) {
   552→      try {
   553→        console.log(`[WebSearch] Trying AllOrigins for search result: ${result.url.slice(0, 80)}...`);
   554→        // Try both /raw and /get endpoints
   555→        const proxyUrls = [
   556→          `https://api.allorigins.win/raw?url=${encodeURIComponent(result.url)}`,
   557→          `https://api.allorigins.win/get?url=${encodeURIComponent(result.url)}`,
   558→        ];
   559→
   560→        let proxyHtml: string | null = null;
   561→
   562→        for (const pUrl of proxyUrls) {
   563→          try {
   564→            const controller = new AbortController();
   565→            const timeout = setTimeout(() => controller.abort(), 15000);
   566→            const proxyRes = await fetch(pUrl, { signal: controller.signal });
   567→            clearTimeout(timeout);
   568→            if (!proxyRes.ok) continue;
   569→
   570→            const isRaw = pUrl.includes("/raw?");
   571→            if (isRaw) {
   572→              const text = await proxyRes.text();
   573→              if (text && text.length > 5000 && text.includes("<")) {
   574→                proxyHtml = text;
   575→                break;
   576→              }
   577→            } else {
   578→              const data = await proxyRes.json();
   579→              const contents = typeof data === "string" ? data : data?.contents;
   580→              if (contents && typeof contents === "string" && contents.length > 5000) {
   581→                proxyHtml = contents;
   582→                break;
   583→              }
   584→            }
   585→          } catch { /* try next */ }
   586→        }
   587→
   588→        if (!proxyHtml || proxyHtml.length < 5000) continue;
   589→
   590→        // Check if we got the real product page (has OG title)
   591→        const hasOGTitle = /<meta[^>]*property=["']og:title["']/i.test(proxyHtml);
   592→        if (!hasOGTitle) continue;
   593→
   594→        // Extract price using extractProductInfo
   595→        const proxyResult = extractProductInfo(proxyHtml, result.url);
   596→        if (proxyResult.price && proxyResult.price > 0) {
   597→          console.log(`[WebSearch] ✓ Got price $${proxyResult.price} from AllOrigins search result`);
   598→          return proxyResult;
   599→        }
   600→
   601→        // Also try direct OG price extraction
   602→        const ogPriceMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
   603→        const ogCurrencyMatch = proxyHtml.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
   604→        if (ogPriceMatch) {
   605→          const priceVal = parseFloat(ogPriceMatch[1]);
   606→          const cur = ogCurrencyMatch?.[1] || "USD";
   607→          if (priceVal > 0 && priceVal < 100000) {
   608→            let priceUSD = priceVal;
   609→            if (cur !== "USD" && CURRENCY_TO_USD[cur]) {
   610→              priceUSD = Math.round(priceVal * CURRENCY_TO_USD[cur] * 100) / 100;
   611→            }
   612→            const ogTitle = proxyHtml.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1];
   613→            const ogImage = proxyHtml.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1];
   614→            const foundGoodsId = result.url.match(/-g-(\d{10,})/)?.[1] || goodsId;
   615→            console.log(`[WebSearch] ✓ Got OG price ${priceVal} ${cur} = $${priceUSD} from AllOrigins search result`);
   616→            return {
   617→              price: priceUSD,
   618→              currency: "USD",
   619→              productName: ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null,
   620→              productDescription: null,
   621→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   622→              originalPrice: null,
   623→              image: ogImage || null,
   624→              source: `websearch-allorigins-og(${cur})`,
   625→              antiBotDetected: false,
   626→            };
   627→          }
   628→        }
   629→      } catch (err) {
   630→        console.log(`[WebSearch] AllOrigins for search result failed:`, String(err).slice(0, 80));
   631→      }
   632→    }
   633→
   634→    return llmResult; // Return whatever the LLM found (may be null)
   635→  } catch (err) {
   636→    console.error("[WebSearch] Error:", String(err).slice(0, 200));
   637→    return null;
   638→  }
   639→}

   744→/**
   745→ * Try to parse prices directly from search result snippets.
   746→ * This avoids the cost of an LLM call and is often sufficient
   747→ * because Google includes price info in Temu search snippets.
   748→ */
   749→function parsePriceFromSnippets(
   750→  results: any[],
   751→  goodsId: string | null,
   752→): TemuProductData | null {
   753→  // Look for Temu results that have price info in snippets
   754→  for (const result of results) {
   755→    if (!result.url?.includes("temu.com")) continue;
   756→    if (!result.snippet) continue;
   757→
   758→    const snippet = result.snippet;
   759→
   760→    // Extract the goods_id from the URL if we don't have it
   761→    let foundGoodsId = goodsId;
   762→    if (!foundGoodsId) {
   763→      const gMatch = result.url.match(/-g-(\d{10,})/);
   764→      if (gMatch) foundGoodsId = gMatch[1];
   765→    }
   766→
   767→    // Extract product name from search result
   768→    const productName = result.name
   769→      ?.replace(/\s*[-|]\s*Temu\s*$/i, "")
   770→      .replace(/\s*[-|]\s*(Mauritius|Oman|Bahrain|Ecuador|Pakistan|Algeria|Morocco|Tunisia).*$/i, "")
   771→      .trim() || null;
   772→
   773→    // Determine the local currency from the Temu URL locale prefix
   774→    // e.g. /mu/ → MUR, /pk-en/ → PKR, /om-en/ → OMR, /bh/ → BHD, etc.
   775→    let localCurrency = "USD";
   776→    const localeMatch = result.url.match(/temu\.com\/([a-z]{2}(?:-[a-z]{2})?)\//i);
   777→    if (localeMatch) {
   778→      const locale = localeMatch[1].toLowerCase();
   779→      const localeToCurrency: Record<string, string> = {
   780→        mu: "MUR",       // Mauritius
   781→        pk: "PKR", "pk-en": "PKR", "pk-ur": "PKR", // Pakistan
   782→        om: "OMR", "om-en": "OMR", "om-ar": "OMR", // Oman
   783→        bh: "BHD", "bh-en": "BHD", "bh-ar": "BHD", // Bahrain
   784→        sa: "SAR", "sa-en": "SAR", "sa-ar": "SAR", // Saudi Arabia
   785→        ae: "AED", "ae-en": "AED", "ae-ar": "AED", // UAE
   786→        eg: "EGP", "eg-en": "EGP", "eg-ar": "EGP", // Egypt
   787→        ma: "MAD", "ma-en": "MAD", "ma-ar": "MAD", "ma-fr": "MAD", // Morocco
   788→        tn: "TND", "tn-en": "TND", "tn-ar": "TND", "tn-fr": "TND", // Tunisia
   789→        dz: "DZD", "dz-en": "DZD", "dz-ar": "DZD", "dz-fr": "DZD", // Algeria
   790→        kw: "KWD", "kw-en": "KWD", "kw-ar": "KWD", // Kuwait
   791→        qa: "QAR", "qa-en": "QAR", "qa-ar": "QAR", // Qatar
   792→        jo: "JOD", "jo-en": "JOD", "jo-ar": "JOD", // Jordan
   793→        ec: "USD", "ec-es": "USD", // Ecuador (uses USD)
   794→        lk: "LKR", "lk-en": "LKR", "lk-si": "LKR", "lk-ta": "LKR", // Sri Lanka
   795→        np: "NPR", "np-en": "NPR", // Nepal
   796→        bd: "BDT", "bd-en": "BDT", // Bangladesh
   797→        in: "INR", "in-en": "INR", "in-hi": "INR", // India
   798→        ph: "PHP", "ph-en": "PHP", // Philippines
   799→        br: "BRL", "br-pt": "BRL", // Brazil
   800→        mx: "MXN", "mx-es": "MXN", // Mexico
   801→      };
   802→      const found = localeToCurrency[locale];
   803→      if (found) localCurrency = found;
   804→    }
   805→
   806→    // Try to find a price in the snippet
   807→    // We'll use the URL locale to determine the correct currency for "Rs" amounts
   808→    // Common patterns:
   809→    //   "Rs 157.77 35% OFF"     → Rs means local rupee based on locale
   810→    //   "$6.99"                  → USD
   811→    //   "OMR 1.200"             → Omani Rial
   812→    //   "6,399 Rs.2,612 59% OFF" → number Rs. = sale price in local rupee
   813→    //   "€5.99"                  → EUR
   814→    //   "£4.99"                  → GBP
   815→
   816→    // Determine if this locale uses European comma format (comma = decimal separator)
   817→    // e.g., Ecuador (ec), Spain (es), Mexico (mx), Brazil (br), France (ma-fr, tn-fr)
   818→    const europeanCommaLocales = new Set([
   819→      "ec", "ec-es", "mx", "mx-es", "br", "br-pt",
   820→      "ma-fr", "tn-fr", // French-speaking locales
   821→    ]);
   822→    const usesEuropeanComma = europeanCommaLocales.has(localCurrency === "USD" && localeMatch ? localeMatch[1].toLowerCase() : "") ||
   823→      (result.url.match(/temu\.com\/(ec|mx|br)\b/i) != null);
   824→
   825→    // Helper: parse price string respecting locale format
   826→    // European format: "5,22" = 5.22, "1.234,56" = 1234.56
   827→    // English format:  "5.22" = 5.22, "1,234.56" = 1234.56
   828→    const parsePriceString = (priceStr: string, isEuropean: boolean): number => {
   829→      if (isEuropean) {
   830→        // European: remove dots (thousands), replace comma with dot (decimal)
   831→        return parseFloat(priceStr.replace(/\./g, "").replace(",", "."));
   832→      } else {
   833→        // English: remove commas (thousands)
   834→        return parseFloat(priceStr.replace(/,/g, ""));
   835→      }
   836→    };
   837→
   838→    // First, try explicit currency patterns (most reliable)
   839→    const explicitPatterns: { pattern: RegExp; currency: string }[] = [
   840→      { pattern: /\$\s*([\d,]+(?:\.\d{1,2})?)/, currency: "USD" },
   841→      { pattern: /€\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EUR" },
   842→      { pattern: /£\s*([\d,]+(?:\.\d{1,2})?)/, currency: "GBP" },
   843→      { pattern: /OMR\s*([\d,]+(?:\.\d{1,3})?)/, currency: "OMR" },
   844→      { pattern: /BHD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "BHD" },
   845→      { pattern: /SAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "SAR" },
   846→      { pattern: /AED\s*([\d,]+(?:\.\d{1,2})?)/, currency: "AED" },
   847→      { pattern: /PKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PKR" },
   848→      { pattern: /EGP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "EGP" },
   849→      { pattern: /MAD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MAD" },
   850→      { pattern: /TND\s*([\d,]+(?:\.\d{1,3})?)/, currency: "TND" },
   851→      { pattern: /DZD\s*([\d,]+(?:\.\d{1,2})?)/, currency: "DZD" },
   852→      { pattern: /KWD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "KWD" },
   853→      { pattern: /QAR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "QAR" },
   854→      { pattern: /JOD\s*([\d,]+(?:\.\d{1,3})?)/, currency: "JOD" },
   855→      { pattern: /INR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "INR" },
   856→      { pattern: /BDT\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BDT" },
   857→      { pattern: /LKR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "LKR" },
   858→      { pattern: /NPR\s*([\d,]+(?:\.\d{1,2})?)/, currency: "NPR" },
   859→      { pattern: /PHP\s*([\d,]+(?:\.\d{1,2})?)/, currency: "PHP" },
   860→      { pattern: /BRL\s*([\d,]+(?:\.\d{1,2})?)/, currency: "BRL" },
   861→      { pattern: /MXN\s*([\d,]+(?:\.\d{1,2})?)/, currency: "MXN" },
   862→    ];
   863→
   864→    for (const { pattern, currency } of explicitPatterns) {
   865→      const match = snippet.match(pattern);
   866→      if (match) {
   867→        // For USD prices on European locale pages (like Ecuador), use comma-as-decimal parsing
   868→        const isEuroPrice = currency === "USD" && usesEuropeanComma;
   869→        const localPrice = parsePriceString(match[1], isEuroPrice);
   870→        if (localPrice > 0 && localPrice < 10000000) {
   871→          const rate = CURRENCY_TO_USD[currency];
   872→          if (rate) {
   873→            const usdPrice = Math.round(localPrice * rate * 100) / 100;
   874→            if (usdPrice >= 0.10 && usdPrice <= 5000) {
   875→              console.log(`[WebSearch] Found explicit price: ${localPrice} ${currency} (European format: ${isEuroPrice}) = $${usdPrice} USD`);
   876→              return {
   877→                price: usdPrice,
   878→                currency: "USD",
   879→                productName,
   880→                productDescription: null,
   881→                canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,
   882→                originalPrice: null,
   883→                image: null,
   884→                source: `web-search-snippet(${currency})`,
   885→                antiBotDetected: false,
   886→              };
   887→            }
   888→          }
   889→        }
   890→      }
   891→    }
   892→
   893→    // Also try European comma format pattern for $ prices (e.g., "$5,22" on Ecuador page)
   894→    // This catches prices where the comma is used as decimal separator
   895→    if (usesEuropeanComma) {
   896→      const euroPattern = /\$\s*(\d+,\d{1,2})\b/;
   897→      const euroMatch = snippet.match(euroPattern);
   898→      if (euroMatch) {
   899→        const localPrice = parseFloat(euroMatch[1].replace(",", "."));
   900→        if (localPrice > 0 && localPrice < 10000000) {
   901→          const usdPrice = Math.round(localPrice * 100) / 100; // Already USD
   902→          if (usdPrice >= 0.10 && usdPrice <= 5000) {
   903→            console.log(`[WebSearch] Found European format price: ${euroMatch[0]} = $${usdPrice} USD`);
   904→            return {
   905→              price: usdPrice,
   906→              currency: "USD",
   907→              productName,
   908→              productDescription: null,
   909→              canonicalUrl: foundGoodsId ? `https://www.temu.com/-g-${foundGoodsId}.html` : null,

  1411→}

  1413→/* ─── Extraction: focus on OG meta tags (always work) + JSON price fields ─── */
  1414→function extractProductInfo(html: string, originalUrl: string): TemuProductData {
  1415→  // 1. OG meta tags (work even on anti-bot page)
  1416→  const ogTitle = html.match(/<meta[^>]*property=["']og:title["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1417→  const ogDescription = html.match(/<meta[^>]*property=["']og:description["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1418→  const ogUrl = html.match(/<meta[^>]*property=["']og:url["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1419→  const ogImage = html.match(/<meta[^>]*property=["']og:image["'][^>]*content=["']([^"']+)["']/i)?.[1] || null;
  1420→  const canonical = html.match(/<link[^>]+rel=["']canonical["'][^>]*href=["']([^"']+)["']/i)?.[1] || ogUrl;
  1421→
  1422→  // Decode HTML entities in title
  1423→  const productName = ogTitle ? decodeHtmlEntities(ogTitle).replace(/\s*[-|]\s*Temu\s*$/i, "").trim() : null;
  1424→  const productDescription = ogDescription ? decodeHtmlEntities(ogDescription) : null;
  1425→
  1426→  // 2. Try to find price (rarely works due to Temu anti-bot, but try anyway)
  1427→  let price: number | null = null;
  1428→  let currency = "USD";
  1429→  let originalPrice: number | null = null;
  1430→  let priceSource = "";
  1431→
  1432→  // ─────────────────────────────────────────────────────────────
  1433→  // 2-preferred. URL-embedded price hint.
  1434→  // Temu product URLs often include `_oak_rec_ext_1=<base64>` which is
  1435→  // the base64-encoded price in MINOR UNITS (cents). E.g. "MTIz" → "123"
  1436→  // → 1.23 USD. This is the most reliable signal because it comes
  1437→  // straight from the share URL the user copied.
  1438→  // ─────────────────────────────────────────────────────────────
  1439→  try {
  1440→    const parsedUrl = new URL(originalUrl);
  1441→    const hint = parsedUrl.searchParams.get("_oak_rec_ext_1");
  1442→    if (hint) {
  1443→      // base64 decode (URL-safe variant — replace - with + and _ with /)
  1444→      const b64 = hint.replace(/-/g, "+").replace(/_/g, "/");
  1445→      const decoded = Buffer.from(b64, "base64").toString("utf-8").trim();
  1446→      // Strip non-digits
  1447→      const cents = parseInt(decoded.replace(/\D/g, ""), 10);
  1448→      if (cents > 0 && cents < 10000000) {
  1449→        // Heuristic: if the decoded value looks like cents (>= 10), divide by 100.
  1450→        // If it already looks like a dollar amount (>= 1000 → assume already in cents).
  1451→        const usd = cents / 100;
  1452→        if (usd >= 0.01 && usd < 100000) {
  1453→          price = usd;
  1454→          currency = "USD";
  1455→          priceSource = "url-hint";
  1456→        }
  1457→      }
  1458→    }
  1459→  } catch { /* not a parseable URL */ }
  1460→
  1461→  // 2a. JSON-LD structured data
  1462→  if (!price) {
  1463→    const jsonLdMatches = html.matchAll(/<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/gi);
  1464→    for (const match of jsonLdMatches) {
  1465→      try {
  1466→        const data = JSON.parse(match[1]);
  1467→        const schemas = Array.isArray(data) ? data : [data];
  1468→        for (const schema of schemas) {
  1469→          if (schema["@type"] === "Product" && schema.offers) {
  1470→            const offers = Array.isArray(schema.offers) ? schema.offers : [schema.offers];
  1471→            for (const offer of offers) {
  1472→              if (offer.price !== undefined) {
  1473→                price = parseFloat(offer.price);
  1474→                currency = offer.priceCurrency || "USD";
  1475→                priceSource = "json-ld";
  1476→                break;
  1477→              }
  1478→            }
  1479→          }
  1480→        }
  1481→      } catch { /* skip */ }
  1482→      if (price) break;
  1483→    }
  1484→  }
  1485→
  1486→  // 2b. OG product:price:amount meta
  1487→  if (!price) {
  1488→    const ogPrice = html.match(/<meta[^>]*property=["']product:price:amount["'][^>]*content=["']([^"']+)["']/i);
  1489→    if (ogPrice) {
  1490→      const p = parseFloat(ogPrice[1]);
  1491→      if (p > 0) {
  1492→        price = p;
  1493→        const ogCur = html.match(/<meta[^>]*property=["']product:price:currency["'][^>]*content=["']([^"']+)["']/i);
  1494→        currency = ogCur?.[1] || "USD";
  1495→        priceSource = "og-meta";
  1496→      }
  1497→    }
  1498→  }
  1499→
  1500→  // 2c. Embedded JSON priceInfo blocks (this is what Temu uses)
  1501→  //     IMPORTANT: Temu pages contain many priceInfo blocks (one per variant,
  1502→  //     one per "similar product", one per cross-sell, etc.). The previous
  1503→  //     implementation picked the FIRST match, which often returned a
  1504→  //     cross-sell product's price (e.g. $30) instead of the actual
  1505→  //     product's price (e.g. $1.23). We now collect ALL matches and
  1506→  //     pick the LOWEST plausible one, since the actual product the user
  1507→  //     is viewing is almost always the cheapest among the rendered
  1508→  //     priceInfo blocks.
  1509→  if (!price) {
  1510→    type Candidate = { usd: number; currency: string; marketUsd?: number };
  1511→    const candidates: Candidate[] = [];
  1512→
  1513→    // priceInfo with explicit priceStr (most reliable)
  1514→    const priceInfosWithStr = [...html.matchAll(
  1515→      /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"[^}]*?"priceStr"\s*:\s*"([^"]+)"/g
  1516→    )];
  1517→    for (const pi of priceInfosWithStr) {
  1518→      const cents = parseInt(pi[1]);
  1519→      const cur = pi[2];
  1520→      const priceStr = pi[3];
  1521→      // Skip i18n placeholder strings
  1522→      if (/OK|Btn|Label|Placeholder/i.test(priceStr)) continue;
  1523→      if (cents <= 0 || cents >= 100000000) continue;
  1524→      const usd = cents / 100;
  1525→      if (usd < 0.01 || usd > 100000) continue;
  1526→      const marketPriceMatch = pi[0].match(/"marketPrice"\s*:\s*(\d+)/);
  1527→      const marketUsd = marketPriceMatch ? parseInt(marketPriceMatch[1]) / 100 : undefined;
  1528→      candidates.push({ usd, currency: cur, marketUsd });
  1529→    }
  1530→
  1531→    // priceInfo without priceStr (broader fallback)
  1532→    if (candidates.length === 0) {
  1533→      const priceInfosAll = [...html.matchAll(
  1534→        /"priceInfo"\s*:\s*\{[^}]*?"price"\s*:\s*(\d+)[^}]*?"currency"\s*:\s*"([A-Z]{3})"/g
  1535→      )];
  1536→      for (const pi of priceInfosAll) {
  1537→        const cents = parseInt(pi[1]);
  1538→        const cur = pi[2];
  1539→        if (cents <= 50 || cents >= 100000000) continue;
  1540→        const usd = cents / 100;
  1541→        if (usd < 0.50 || usd > 100000) continue;
  1542→        candidates.push({ usd, currency: cur });
  1543→      }
  1544→    }
  1545→
  1546→    if (candidates.length > 0) {
  1547→      // Pick the LOWEST price — the user is always viewing the cheapest
  1548→      // variant of the actual product, not the cross-sells.
  1549→      candidates.sort((a, b) => a.usd - b.usd);
  1550→      const best = candidates[0];
  1551→      price = best.usd;
  1552→      currency = best.currency;
  1553→      if (best.marketUsd && best.marketUsd > best.usd) {
  1554→        originalPrice = best.marketUsd;
  1555→      }
  1556→      priceSource = `priceInfo(lowest of ${candidates.length})`;
  1557→    }
  1558→  }
  1559→
  1560→  // 2d. Embedded JSON price fields (broad regex sweep)
  1561→  if (!price) {
  1562→    const fields = ["salePrice", "minPrice", "minAppPrice", "appPrice", "displayPrice", "normalPrice"];
  1563→    const found: { value: number; field: string }[] = [];
  1564→    for (const f of fields) {
  1565→      const re = new RegExp(`"${f}"\\s*:\\s*"?([\\d.]+)"?`, "g");
  1566→      for (const m of html.matchAll(re)) {
  1567→        const v = parseFloat(m[1]);
  1568→        if (v > 0 && v < 100000) {
  1569→          found.push({ value: v, field: f });
  1570→        }
  1571→      }
  1572→    }
  1573→    if (found.length > 0) {
  1574→      // Sort and pick lowest (likely sale price)
  1575→      found.sort((a, b) => a.value - b.value);
  1576→      const best = found[0];
  1577→      // Heuristic: if value > 100, probably in cents → divide by 100
  1578→      price = best.value > 100 ? best.value / 100 : best.value;
  1579→      priceSource = `embedded-${best.field}`;
  1580→    }
  1581→  }
  1582→
  1583→  // 2e. Dollar/£/€ text (last resort — but only use prices > $1 to avoid UI elements)
  1584→  if (!price) {
  1585→    const text = html.replace(/<[^>]*>/g, " ");
  1586→    const matches = [...text.matchAll(/[£€$]\s*(\d{1,4}(?:[.,]\d{1,2})?)/g)];
  1587→    const prices: { value: number; currency: string }[] = [];
  1588→    for (const m of matches) {
  1589→      const raw = m[1].replace(/,/g, "");
  1590→      const v = parseFloat(raw);
  1591→      if (v > 1.5 && v < 5000) {
  1592→        const cur = m[0].includes("£") ? "GBP" : m[0].includes("€") ? "EUR" : "USD";
  1593→        prices.push({ value: v, currency: cur });
  1594→      }
  1595→    }
  1596→    if (prices.length > 0) {
  1597→      // Pick lowest non-suspicious price (avoid shipping credits like $5/$8/$13)
  1598→      const sorted = prices.sort((a, b) => a.value - b.value);
  1599→      // Skip the first if it's a suspicious "round" price (like $5.00, $8.00)
  1600→      const filtered = sorted.filter(p => !(p.value < 20 && p.value === Math.floor(p.value)));
  1601→      if (filtered.length > 0) {
  1602→        price = filtered[0].value;
  1603→        currency = filtered[0].currency;
  1604→        priceSource = "text";
  1605→      }
  1606→    }
  1607→  }
  1608→
  1609→  return {
  1610→    price,
  1611→    currency,
  1612→    productName,
  1613→    productDescription,
  1614→    canonicalUrl: canonical,
  1615→    originalPrice,
  1616→    image: ogImage,
  1617→    source: priceSource || (productName ? "og-only" : "none"),
  1618→  };
  1619→}

  1621→function decodeHtmlEntities(s: string): string {
  1622→  return s
  1623→    .replace(/&amp;/g, "&")
  1624→    .replace(/&lt;/g, "<")
  1625→    .replace(/&gt;/g, ">")

  1627→    .replace(/&#39;/g, "'")
  1628→    .replace(/&nbsp;/g, " ")
  1629→    .replace(/&#(\d+);/g, (_, n) => String.fromCharCode(parseInt(n)));
  1630→}

  1633→function extractProductNameFromUrl(url: string): string | null {
  1634→  try {

  1638→    const name = slug
  1639→      .replace(/-g-[a-zA-Z0-9]+\.html?$/i, "")
  1640→      .replace(/\.html?$/i, "")

  1642→      .trim();
  1643→    if (name && name.length > 3) {
  1644→      return name.replace(/\b\w/g, (l) => l.toUpperCase());

  1647→  return null;

  1650→function cleanProductName(name: string | null): string | null {
  1651→  if (!name) return null;
  1652→  let cleaned = name
  1653→    .replace(/\s*[-|]\s*(Temu|AliExpress)\s*/gi, "")
  1654→    .replace(/\s*[-|]\s*(Login|Sign In|Register)\s*/gi, "")
  1655→    // Remove Temu locale suffixes like "Algeria", "Mauritius", "Oman", etc.

  1657→    .replace(/\s+/g, " ")
  1658→    .trim();
  1659→  if (cleaned.length > 100) cleaned = cleaned.slice(0, 97) + "...";
  1660→  return cleaned || null;
  1661→}
